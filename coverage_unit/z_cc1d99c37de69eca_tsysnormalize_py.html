18def TsysAfterPowerChange(v_orig, v_new, tsys_orig=200.0, t_atm=270.0): 

19 # The maths to derive a new Tsys 

20 x = (v_new / v_orig) * (tsys_orig / (tsys_orig + t_atm)) 

21 return t_atm * x / (1.0 - x)

24def getPower(vis, scan, spw, duration, fromEnd=False, skipStartSecs=1.0, 

25 skipEndSecs=1.0, datacolname='data', verbose=True): 

26 """ 

27 Return a per-antenna list of total power values for the two polarizations of the specified scan and spw. 

28 duration: number of samples to use starting from the start of the scan 

29 I think the idea here is that the sky subscan is always the first subscan of a Tsys scan. If this ever 

30 changes to one of the loads, then the result will be less than optimal. It would likely result in very 

31 small changes from the original tsys table (i.e. it will not get normalized). - Todd 

32 """ 

33 

34 # Read in data from MS 

35 with casa_tools.MSReader(vis) as ms: 

36 ms.selectinit(datadescid=spw) 

37 ms.selecttaql('SCAN_NUMBER==%d AND DATA_DESC_ID==%d AND ANTENNA1==ANTENNA2'%(scan, spw)) 

38 d = ms.getdata([datacolname, 'axis_info'], ifraxis=True) 

39 

40 powers = d[datacolname] 

41 ants = d['axis_info']['ifr_axis']['ifr_name'] 

42 pols = list(d['axis_info']['corr_axis']) 

43 idxPol0 = pols.index("XX") 

44 idxPol1 = pols.index("YY") 

45 if verbose: 

46 LOG.info("Pol 0,1 indexes: %d, %d" % (idxPol0, idxPol1)) 

47 ts = d['axis_info']['time_axis']['MJDseconds'] 

48 t0 = ts[0] 

49 tf = ts[-1] 

50 ts -= t0 

51 if verbose: 

52 LOG.info("times:", ts) 

53 LOG.info("pols:", pols) 

54 # choose the time samples we want 

55 sampStart = -99 

56 sampEnd = -99 

57 for i in range(len(ts)): 

58 if ts[i] > skipStartSecs: 

59 sampStart = i 

60 break 

61 if sampStart >= len(ts): 

62 sampStart = len(ts) - 1 

63 for i in range(len(ts)-1, sampStart, -1): 

64 if tf - ts[i] > skipEndSecs: 

65 sampEnd = i 

66 break 

67 if sampEnd <= sampStart: 

68 sampEnd = sampStart + 1 

69 if not fromEnd: 

70 # take duration samples from start 

71 for i in range(sampStart+1, sampEnd, 1): 

72 if ts[i] - ts[sampStart] > duration: 

73 sampEnd = i 

74 break 

75 else: 

76 # instead from end 

77 for i in range(sampEnd-1, sampStart, -1): 

78 if ts[sampEnd] - ts[i] > duration: 

79 sampStart = i 

80 break 

81 if verbose: 

82 LOG.info("chosen sample range: %d to %d" % (sampStart, sampEnd)) 

83 # indexing is pol, baseline(=0), ant, sample 

84 if verbose: 

85 LOG.info("number of antennas to produce powers for:", len(ants)) 

86 result = [] 

87 for ant in range(len(ants)): 

88 powersPol0 = powers[idxPol0][0][ant][sampStart:sampEnd] 

89 powersPol1 = powers[idxPol1][0][ant][sampStart:sampEnd] 

90 medianP0 = np.median(powersPol0) 

91 medianP1 = np.median(powersPol1) 

92 result.append([medianP0, medianP1]) 

93 return result

96def scienceSpwForTsysSpw(mymsmd, tsysSpw): 

97 """ 

98 Automatically pick one science spw for the specified Tsys spw.  

99 Ideally it would be the one with the widest bandwidth, but it will likely make very little 

100 difference, so just take the first one if there are more than one. 

101 """ 

102 # FIXME: Use domain objects from context instead of msmd. 

103 baseband = mymsmd.baseband(tsysSpw) 

104 spws = np.intersect1d(mymsmd.almaspws(tdm=True, fdm=True), mymsmd.spwsforbaseband(baseband)) 

105 if 'OBSERVE_TARGET#ON_SOURCE' in mymsmd.intents(): 

106 spws = np.intersect1d(mymsmd.spwsforintent('OBSERVE_TARGET#ON_SOURCE'), spws) 

107 elif 'CALIBRATE_FLUX#ON_SOURCE' in mymsmd.intents(): 

108 spws = np.intersect1d(mymsmd.spwsforintent('CALIBRATE_FLUX#ON_SOURCE'), spws) 

109 elif 'CALIBRATE_AMPLI#ON_SOURCE' in mymsmd.intents(): 

110 spws = np.intersect1d(mymsmd.spwsforintent('CALIBRATE_AMPLI#ON_SOURCE'), spws) 

111 

112 if (len(spws) > 1): 

113 LOG.info("Multiple science spws for this tsys spw (%d). Picking the first one (%d)" % (tsysSpw, spws[0])) 

114 return(spws[0]) 

115 elif (len(spws) < 1): 

116 LOG.info("No science spws for this tsys spw (%d)" % (tsysSpw)) 

117 else: 

118 return(spws[0]) 

119 return -1

122def tsysNormalize(vis, tsysTable, newTsysTable, scaleSpws=[], verbose=False): 

123 """ 

124 Generate Tsys entries for one field from other fields, using autocorr 

125 (linear!) or SQLD data to determine the change in Tsys. 

126 Inputs: 

127 

128 vis the MS 

129 tsysTable: the tsys caltable (default = <vis>.tsys) 

130 newTsysTable: the new tsys caltable to create (default = <tsysTable>_normalized) 

131 """ 

132 

133 # intents likely to imply different attenuations or tuning to science-like 

134 # scans that we are applying Tsys to. 

135 badIntents = ['CALIBRATE_POINTING', 'CALIBRATE_FOCUS', 'CALIBRATE_SIDEBAND_RATIO', 'CALIBRATE_ATMOSPHERE'] 

136 

137 # Determine column name for data. This presently differs between 

138 # interferometry and single-dish, as they use different values  

139 # for the parameter ocorr_mode at the importasdm stage. 

140 # Note: using lower-case for column name, as ms.getdata will return 

141 # a dictionary that has lower-case keys, regardless of whether  

142 # the column was upper-case. 

143 # TODO: need a common solution for identifying data column. 

144 with casa_tools.TableReader(vis) as tb: 

145 # SD will contain "FLOAT_DATA" 

146 if 'FLOAT_DATA' in tb.colnames(): 

147 datacolname = 'float_data' 

148 # IF will contain "DATA" 

149 else: 

150 datacolname = 'data' 

151 

152 with casa_tools.MSMDReader(vis) as msmd: 

153 

154 with casa_tools.TableReader(tsysTable, nomodify=False) as table: 

155 

156 # For convenience squish the useful columns into unique lists 

157 tsysSpws = np.unique(table.getcol("SPECTRAL_WINDOW_ID")) 

158 tsysScans = np.unique(table.getcol("SCAN_NUMBER")) 

159 tsysTimes = np.unique(table.getcol("TIME")) 

160 tsysFields = np.unique(table.getcol("FIELD_ID")) 

161 tsysAntennas = np.unique(table.getcol("ANTENNA1")) 

162 

163 if len(scaleSpws) < len(tsysSpws): 

164 scaleSpws = [] 

165 for tsysSpw in tsysSpws: 

166 scaleSpws.append(scienceSpwForTsysSpw(msmd, tsysSpw)) 

167 LOG.info("Identified autocorrelation spws to use: {0}".format(scaleSpws)) 

168 

169 LOG.info("Tsys Spws ({0}): {1}".format(len(tsysSpws), tsysSpws)) 

170 LOG.info("Tsys Scans ({0}): {1}".format(len(tsysScans), tsysScans)) 

171 LOG.info("Tsys Times ({0}): {1}".format(len(tsysTimes), tsysTimes)) 

172 LOG.info("Tsys Fields ({0}): {1}".format(len(tsysFields), tsysFields)) 

173 LOG.info("Tsys Antennas ({0}): {1}".format(len(tsysAntennas), tsysAntennas)) 

174 

175 # Gather the power levels to use in the normalization process 

176 refPowers = {} 

177 refScans = {} 

178 for f in tsysFields: 

179 scanFieldsTab = table.query('FIELD_ID==%d'%f) 

180 fieldTsysScans = np.unique(scanFieldsTab.getcol("SCAN_NUMBER")) 

181 scanFieldsTab.close() 

182 fieldAllScans = msmd.scansforfield(f) 

183 fieldNonTsysScans = [x for x in fieldAllScans if x not in fieldTsysScans] 

184 if (len(fieldNonTsysScans) < 1): 

185 # Then there is no non-tsys scan for this field, e.g. 

186 # which can happen in a mosaic where the Tsys scan has a 

187 # different field ID, but in this case the field name will 

188 # have other scans with different field IDs, so revert to 

189 # using field names. Using field names might work from the 

190 # outset, but I have not tried it. 

191 fieldName = msmd.namesforfields(f)[0] 

192 fieldAllScans = msmd.scansforfield(fieldName) 

193 fieldNonTsysScans = [x for x in fieldAllScans if x not in fieldTsysScans] 

194 if (len(fieldNonTsysScans) < 1): 

195 LOG.info("This field (id={0}, name={1}) appears to have " 

196 "no non-Tsys-like-scans, and thus cannot be " 

197 "normalized.".format(f, fieldName)) 

198 return -1 

199 scienceLikeScans = [] 

200 for s in fieldNonTsysScans: 

201 intents = msmd.intentsforscan(s) 

202 good = True 

203 for i in intents: 

204 for b in badIntents: 

205 if i.startswith(b): 

206 good = False 

207 break 

208 if good: scienceLikeScans.append(s) 

209 powerRefScans = [] 

210 for s in fieldTsysScans: 

211 minDist = 9999999 

212 refScan = -1 

213 for r in scienceLikeScans: 

214 dist = abs(r - s) 

215 if dist < minDist: 

216 minDist = dist 

217 refScan = r 

218 powerRefScans.append(refScan) 

219 LOG.info("Field {0} Tsys scans: {1}, All scans: {2}, Non-Tsys scans: {3}," 

220 " Non-Tsys science-like scans: {4}".format(f, fieldTsysScans, fieldAllScans, fieldNonTsysScans, 

221 scienceLikeScans)) 

222 for i in range(len(fieldTsysScans)): 

223 LOG.info(" Tsys scan {0} power reference scan: {1}".format( 

224 fieldTsysScans[i], powerRefScans[i])) 

225 refScans[fieldTsysScans[i]] = powerRefScans[i] 

226 if verbose: 

227 LOG.info("populating powers corresponding to each Tsys scan...") 

228 for i in range(len(fieldTsysScans)): 

229 refPowers[fieldTsysScans[i]] = [] 

230 for spw in scaleSpws: 

231 if verbose: 

232 LOG.info("calling getPower(vis, {0}, {1}, 10.0, {2})".format( 

233 powerRefScans[i], spw, str(powerRefScans[i] < fieldTsysScans[i]))) 

234 p = getPower(vis, powerRefScans[i], spw, 10.0, powerRefScans[i] < fieldTsysScans[i], 

235 datacolname=datacolname, verbose=verbose) 

236 refPowers[fieldTsysScans[i]].append(p) 

237 if verbose: 

238 LOG.info("powers to use for Tsys scan {0}: {1}".format(fieldTsysScans[i], 

239 refPowers[fieldTsysScans[i]])) 

240 if verbose: 

241 LOG.info(refPowers) 

242 

243 # Create copy of the original Tsys caltable. 

244 copy = table.copy(newTsysTable) 

245 copy.close() 

246 

247 with casa_tools.TableReader(newTsysTable, nomodify=False) as table: 

248 startRefPower = refPowers[tsysScans[0]] 

249 for i in range(1, len(tsysScans)): 

250 # need to adjust each successive Tsys 

251 for ispw in range(len(tsysSpws)): 

252 spw = tsysSpws[ispw] 

253 for ant in range(len(tsysAntennas)): 

254 tsysSubTab1 = table.query("SCAN_NUMBER==%d AND SPECTRAL_WINDOW_ID==%d AND ANTENNA1==%d" % 

255 (tsysScans[i], tsysSpws[ispw], ant)) 

256 tsys1 = tsysSubTab1.getcell('FPARAM', 0) 

257 newTsys = tsysSubTab1.getcell('FPARAM', 0) 

258 for pol in range(len(tsys1)): 

259 for chan in range(len(tsys1[pol])): 

260 a = TsysAfterPowerChange(refPowers[tsysScans[i]][ispw][ant][pol], 

261 startRefPower[ispw][ant][pol], tsys1[pol][chan]) 

262 newTsys[pol][chan] = a 

263 LOG.info("Scan {0:2d} spw {1:2d} pol {2} mean {3:.1f} --> {4:.1f}".format( 

264 tsysScans[i], spw, pol, np.mean(tsys1[pol]), np.mean(newTsys[pol]))) 

265 tsysSubTab1.putcell('FPARAM', 0, newTsys) 

266 tsysSubTab1.close()
