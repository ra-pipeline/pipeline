161 def merge_with_context(self, context): 

162 target = context.observing_run 

163 

164 for ms in self.mses: 

165 LOG.info('Adding {0} to context'.format(ms.name)) 

166 target.add_measurement_set(ms) 

167 

168 if ms.antenna_array.name in ('EVLA', 'VLA', 'JVLA'): 

169 if not hasattr(context, 'evla'): 

170 context.evla = collections.defaultdict(dict) 

171 

172 msinfos = dict((ms.name, self._do_msinfo_heuristics(ms.name, context)) for ms in self.mses) 

173 context.evla['msinfo'].update(msinfos) 

174 context.project_summary.telescope = 'EVLA' 

175 context.project_summary.observatory = 'Karl G. Jansky Very Large Array' 

176 # context.evla['msinfo'] = { m.name : msinfo } 

177 

178 # Dictionaries for per band solution intervals 

179 m = context.observing_run.get_ms(ms.name) 

180 spw2band = m.get_vla_spw2band() 

181 

182 uniqueBands = set(spw2band.values()) 

183 

184 for bandname in uniqueBands: 

185 context.evla['msinfo'][m.name].gain_solint1[bandname] = 'int' 

186 context.evla['msinfo'][m.name].gain_solint2[bandname] = 'int' 

187 context.evla['msinfo'][m.name].shortsol1[bandname] = 0.0 

188 context.evla['msinfo'][m.name].shortsol2[bandname] = 0.0 

189 context.evla['msinfo'][m.name].longsolint[bandname] = 0.0 

190 context.evla['msinfo'][m.name].short_solint[bandname] = 0.0 

191 context.evla['msinfo'][m.name].new_gain_solint1[bandname] = '1.0s' 

192 context.evla['msinfo'][m.name].setjy_results = self.setjy_results 

193 if self.setjy_results: 

194 for result in self.setjy_results: 

195 result.merge_with_context(context)

197 def _do_msinfo_heuristics(self, ms, context): 

198 """Gets heuristics for VLA via original msinfo script 

199 """ 

200 

201 msinfo = VLAScanHeuristics(ms) 

202 msinfo.makescandict() 

203 msinfo.calibratorIntents() 

204 msinfo.determine3C84() 

205 

206 with casa_tools.TableReader(ms) as table: 

207 scanNums = sorted(numpy.unique(table.getcol('SCAN_NUMBER'))) 

208 

209 # Check for missing scans 

210 missingScans = 0 

211 missingScanStr = '' 

212 

213 for i in range(max(scanNums)): 

214 if scanNums.count(i + 1) == 1: 

215 pass 

216 else: 

217 LOG.warning("WARNING: Scan " + str(i + 1) + " is not present") 

218 missingScans += 1 

219 missingScanStr = missingScanStr + str(i + 1) + ', ' 

220 

221 if missingScans > 0: 

222 LOG.warning("WARNING: There were " + str(missingScans) + " missing scans in this MS") 

223 else: 

224 LOG.info("No missing scans found.") 

225 

226 return msinfo

235 def prepare(self, **parameters): 

236 # get results object by running super.prepare() 

237 results = super().prepare() 

238 

239 # create results object 

240 myresults = VLAImportDataResults(mses=results.mses, setjy_results=results.setjy_results) 

241 

242 myresults.origin = results.origin 

243 

244 for ms in myresults.origin: 

245 if myresults.origin[ms] == 'ASDM': 

246 myresults.origin[ms] = 'SDM' 

247 

248 PbandWarning = '' 

249 for ms in myresults.mses: 

250 for key, value in ms.get_vla_spw2band().items(): 

251 if 'P' in value: 

252 PbandWarning = 'P-band data detected in the raw data. ' \ 

253 'VLA P-band pipeline calibration has not yet been ' \ 

254 'commissioned and may even fail. Please inspect all P-band pipeline ' \ 

255 'products carefully.' 

256 

257 if PbandWarning: 

258 LOG.warning(PbandWarning) 

259 

260 # Spectral line detection tool 

261 for mset in myresults.mses: 

262 detect_spectral_lines(mset=mset, specline_spws=self.inputs.specline_spws) 

263 LOG.debug("Whether spectral window is designated as spectral line or continuum.") 

264 for spw in mset.get_all_spectral_windows(): 

265 LOG.debug("SPW ID {}: {}".format(spw.id, spw.specline_window)) 

266 

267 return myresults
