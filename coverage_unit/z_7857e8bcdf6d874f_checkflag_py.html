29 def __init__(self, context, vis=None, checkflagmode=None, overwrite_modelcol=None, growflags=None, usecontdat=None): 

30 """Initialize Inputs. 

31 

32 Args: 

33 context: Pipeline context object containing state information. 

34 

35 vis: The list of input MeasurementSets. Defaults to the list of MeasurementSets specified in the hifv_importdata task. 

36 

37 checkflagmode: 

38 - Standard VLA modes with improved RFI flagging heuristics: 'bpd-vla', 'allcals-vla', 'target-vla' 

39 - blank string default use of rflag on bandpass and delay calibrators 

40 - use string 'semi' after hifv_semiFinalBPdcals() for executing rflag on calibrators 

41 - use string 'bpd', for the bandpass and delay calibrators: 

42 execute rflag on all calibrated cross-hand corrected data; 

43 extend flags to all correlations 

44 execute rflag on all calibrated parallel-hand residual data; 

45 extend flags to all correlations 

46 execute tfcrop on all calibrated cross-hand corrected data, 

47 per visibility; extend flags to all correlations 

48 execute tfcrop on all calibrated parallel-hand corrected data, 

49 per visibility; extend flags to all correlations 

50 - use string 'allcals', for all the other calibrators, with delays and BPcal applied: 

51 similar procedure as 'bpd' mode, but uses corrected data throughout 

52 - use string 'target', for the target data: 

53 similar procedure as 'allcals' mode, but with a higher SNR cutoff 

54 for rflag to avoid flagging data due to source structure, and 

55 with an additional series of tfcrop executions to make up for 

56 the higher SNR cutoff in rflag 

57 - VLASS specific modes include 'bpd-vlass', 'allcals-vlass', and 'target-vlass' 

58 which calculate thresholds to use per spw/field/scan (action='calculate', then, 

59 per baseband/field/scan, replace all spw thresholds above the median with the median, 

60 before re-running rflag with the new thresholds. This has the effect of 

61 lowering the thresholds for spws with RFI to be closer to the RFI-free 

62 thresholds, and catches more of the RFI. 

63 - Mode 'vlass-imaging' is similar to 'target-vlass', except that it executes on the split off target 

64 data, intent='\\*TARGET', datacolumn='data' and uses a timedevscale of 4.0. 

65 

66 overwrite_modelcol: Always write the model column, even if it already exists. 

67 

68 growflags: Grow flags in time at the end of the following checkflagmodes: 

69 

70 - default=True, for 'bpd-vla', 'allcals-vla', 'bpd', and 'allcals.' 

71 - default=False, for '' and 'semi' 

72 

73 usecontdat: If True (default), use cont.dat file if present (only for checkflagmode='target-vla') 

74 to restrict flagging to specified spectral windows. 

75 If False, ignore any cont.dat file and apply flagging to all spectral windows. 

76 

77 """ 

78 super().__init__() 

79 self.context = context 

80 self.vis = vis 

81 self.checkflagmode = checkflagmode 

82 self.overwrite_modelcol = overwrite_modelcol 

83 if growflags is None: 

84 self.growflags = self.checkflagmode not in ('', 'semi') 

85 else: 

86 self.growflags = growflags 

87 self.usecontdat = usecontdat

415 def do_rfi_flag(self, fieldselect='', scanselect='', intentselect='', spwselect=''): 

416 """Do RFI flagging using multiple passes of rflag/tfcrop/extend.""" 

417 

418 rflag_standard, tfcrop_standard, growflag_standard = self._select_rfi_standard() 

419 flagbackup = False 

420 calcftdev = True 

421 

422 # set ignore_sedf=True/flagbackup=False to maintain the same behavior as the deprecated do_*vlass() methods 

423 ignore_sefd = self.inputs.checkflagmode in ('target-vlass', 'bpd-vlass', 'allcals-vlass', 'vlass-imaging') 

424 

425 # set calcftdev=False to turn off the new heuristic for some older modes 

426 calcftdev = self.inputs.checkflagmode not in ('semi', '', 'target', 'bpd', 'allcals') 

427 

428 if rflag_standard is not None: 

429 for datacolumn, correlation, scale, extendflags in rflag_standard: 

430 if '_' in correlation: 

431 polselect = correlation.split('_')[1] 

432 if not (polselect in self.corr_type_string or polselect == self.corrstring): 

433 continue 

434 if not mssel_valid(self.inputs.vis, field=fieldselect, spw=spwselect, correlation=polselect, 

435 scan=scanselect, intent=intentselect): 

436 continue 

437 if self.inputs.checkflagmode == "allcals-vla": 

438 ms = self.inputs.context.observing_run.get_ms(self.inputs.vis) 

439 fields = ms.get_fields(fieldselect) 

440 

441 for field in fields: 

442 field_scanselect = scanselect 

443 field_spwselect = spwselect 

444 field_intentselect = intentselect 

445 

446 if field_scanselect: 

447 scanlist = field_scanselect.split(",") 

448 scans = ms.get_scans(field=field.id) 

449 sub_scanlist = [] 

450 for scan in scans: 

451 if str(scan.id) in scanlist: 

452 sub_scanlist.append(str(scan.id)) 

453 field_scanselect = ",".join(sub_scanlist) 

454 

455 if field_spwselect: 

456 spwlist = field_spwselect.split(",") 

457 spws = field.valid_spws 

458 sub_spwlist = [] 

459 for spw in spws: 

460 if str(spw.id) in spwlist: 

461 sub_spwlist.append(str(spw.id)) 

462 field_spwselect = ",".join(sub_spwlist) 

463 

464 if field_intentselect: 

465 intentlist = field_intentselect.split(",") 

466 intents = field.intents 

467 sub_intentlist = [] 

468 for intent in intents: 

469 if intent in intentlist: 

470 sub_intentlist.append(intent) 

471 field_intentselect = ",".join(sub_intentlist) 

472 # PIPE-1729: checking if flux calibrator is bandpass calibrator or not 

473 if "BANDPASS" not in field.intents: 

474 # PIPE-1729: checking if model is 1jy point source or not, 

475 # if model is not a 1 Jy point source, setting datacolumn to 'residual' 

476 # otherwise setting it to 'corrected' 

477 if self._is_model_setjy(fieldselect=str(field.id), scanselect=field_scanselect, 

478 intentselect=field_intentselect, spwselect=field_spwselect): 

479 LOG.info(f" Model is not 1Jy point source, using data column residual for field {field.id}") 

480 datacolumn = "residual" 

481 else: 

482 LOG.info(f" MODEL_DATA not found or model is 1Jy point source, using data column corrected for field {field.id}") 

483 datacolumn = "corrected" 

484 else: 

485 LOG.info(f"Flux calibrator is a bandpass calibrator, using data column corrected for field {field.id}") 

486 datacolumn = "corrected" 

487 

488 method_args = {'mode': 'rflag', 

489 'field': str(field.id), 

490 'correlation': correlation, 

491 'scan': field_scanselect, 

492 'intent': field_intentselect, 

493 'spw': field_spwselect, 

494 'ntime': 'scan', 

495 'timedevscale': scale, 

496 'freqdevscale': scale, 

497 'datacolumn': datacolumn, 

498 'flagbackup': flagbackup, 

499 'savepars': False, 

500 'calcftdev': calcftdev, 

501 'useheuristic': True, 

502 'ignore_sefd': ignore_sefd, 

503 'extendflags': extendflags} 

504 self._do_rflag(**method_args) 

505 else: 

506 if datacolumn == 'residual': 

507 # PIPE-1256: determine if we can use the 'RESIDUAL' column in the 'bpd-vlass/vla' mode. 

508 # The usage of 'RESIDUAL' is only valid if the model of bpd source(s) is properly filled *AND* 

509 # the first-order gain/passband calibration has been applied in 'CORRECTED'. 

510 # Here we check each field from the data selection and see if they all meet the above requirements. 

511 # We only examine the parallel hand amplitude: 

512 # - setjy() has only I models for 3C48/3C138/3C286/3C147. 

513 # - setjy(fluxdensity=-1) will fill the cross-hand with zero values. 

514 LOG.info("Determining if we can use the RESIDUAL column for rflag:") 

515 if self._is_model_setjy(): 

516 LOG.info(" MODEL_DATA is present and none of the model(s) from selected data is a 1Jy point source.") 

517 else: 

518 datacolumn = 'corrected'

606 def _select_data(self): 

607 """Selects data according to the specified checkflagmode. 

608 

609 This method constructs selection strings for fields, scans, and intents based on the 

610 `checkflagmode` input. It also determines the appropriate data column to use 

611 ('corrected' or 'data'). 

612 

613 Returns: 

614 tuple: A tuple containing: 

615 - field_select_string (str): Comma-separated list of field IDs. 

616 - scan_select_string (str): Comma-separated list of scan IDs. 

617 - intent_select_string (str): String representing the intent selection. 

618 - column_select_string (str): String representing the data column selection. 

619 """ 

620 

621 # start with default 

622 fieldselect = scanselect = intentselect = '' 

623 columnselect = 'corrected' 

624 

625 ms = self.inputs.context.observing_run.get_ms(self.inputs.vis) 

626 msinfo = self.inputs.context.evla['msinfo'].get(ms.name, None) 

627 sci_spw_list = [spw.id for spw in ms.get_spectral_windows(science_windows_only=True)] 

628 

629 # Select Bandpass/Delay (BPD) calibrators 

630 if self.inputs.checkflagmode in ('bpd-vla', 'bpd-vlass', '', 'bpd'): 

631 fieldselect = msinfo.checkflagfields 

632 # PIPE-1335: down-select scans using both msinfo.checkflagfields and msinfo.testgainscans. 

633 # msinfo.testgainscans alone may include scans of fields not in msinfo.checkflagfields 

634 # (e.g., second calibrators with bandpass or delay intents). See the 21A-311 case from PIPE-1335. 

635 if msinfo.testgainscans: 

636 testpbd_scans = { 

637 s.id for s in ms.get_scans( 

638 scan_id=list(map(int, msinfo.testgainscans.split(','))), 

639 field=msinfo.checkflagfields, spw=sci_spw_list)} 

640 else: 

641 testpbd_scans = set() 

642 scanselect = ','.join(map(str, sorted(testpbd_scans))) 

643 

644 # Select all calibrators excluding BPD calibrators 

645 if self.inputs.checkflagmode in ('allcals-vla', 'allcals-vlass', 'allcals'): 

646 if msinfo.testgainscans: 

647 testpbd_scans = { 

648 s.id for s in ms.get_scans( 

649 scan_id=list(map(int, msinfo.testgainscans.split(','))), 

650 field=msinfo.checkflagfields, spw=sci_spw_list)} 

651 else: 

652 testpbd_scans = set() 

653 allcals_scans = { 

654 s.id for s in ms.get_scans( 

655 scan_id=list(map(int, msinfo.calibrator_scan_select_string.split(','))), 

656 field=msinfo.calibrator_field_select_string, spw=sci_spw_list)} 

657 scanselect = ','.join(map(str, sorted(allcals_scans-testpbd_scans))) 

658 

659 # PIPE-1335: only construct the field selection string if the scan selection string is not empty. 

660 # Note that an exclusion based on msinfo.checkflagfield might accidentaly reject fields observed 

661 # in different intents across scans. See the 21B-136 case from PIPE-1335. 

662 if scanselect: 

663 fields_in_scans = { 

664 f.id for scan in ms.get_scans( 

665 scan_id=list(allcals_scans - testpbd_scans), 

666 field=msinfo.calibrator_field_select_string, spw=sci_spw_list) 

667 for f in scan.fields} 

668 fieldselect = ','.join(map(str, sorted(fields_in_scans))) 

669 

670 # select targets 

671 if self.inputs.checkflagmode in ('target-vla', 'target-vlass', 'vlass-imaging', 'target'): 

672 fieldids = [field.id for field in ms.get_fields(intent='TARGET')] 

673 fieldselect = ','.join([str(fieldid) for fieldid in fieldids]) 

674 intentselect = '*TARGET*' 

675 

676 # select all calibrators 

677 if self.inputs.checkflagmode == 'semi': 

678 fieldselect = msinfo.calibrator_field_select_string 

679 scanselect = msinfo.calibrator_scan_select_string 

680 

681 if self.inputs.checkflagmode == 'vlass-imaging': 

682 # use the 'data' column by default as 'vlass-imaging' is working on target-only MS. 

683 columnselect = 'data' 

684 

685 LOG.debug('FieldSelect: {}'.format(repr(fieldselect))) 

686 LOG.debug('ScanSelect: {}'.format(repr(scanselect))) 

687 LOG.debug('IntentSelect: {}'.format(repr(intentselect))) 

688 LOG.debug('ColumnSelect: {}'.format(repr(columnselect))) 

689 

690 return fieldselect, scanselect, intentselect, columnselect
