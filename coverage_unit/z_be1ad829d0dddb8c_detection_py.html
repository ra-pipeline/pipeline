499class LineWindowParser: 

500 """LineWindowParser is a parser for line window parameter. 

501 

502 Supported format is as follows: 

503 

504 [Single window list] -- apply to all spectral windows 

505 - integer list [chmin, chmax] 

506 - float list [fmin, fmax] 

507 - string list ['XGHz', 'YGHz'] 

508 

509 [Multiple window list] -- apply to all spectral windows

765 def _freq2chan(self, 

766 spwid: int, 

767 window: list[str] | list[float] | list[int] | None) -> list[int] | None: 

768 """Convert frequency selection into channel selection. 

769 

770 If float values are given, they are interpreted as the value in Hz. 

771 Input frequency values should be in LSRK frame. LSRK frequencies are 

772 converted to the frame in which spw is defined. 

773 

774 If int values are given, input window list is simply sorted and 

775 returned as it is. 

776 

777 Args: 

778 spwid: spw id to process 

779 window: Line window list in frequency domain 

780 

781 Returns: 

782 Line window list in channel domain 

783 """ 

784 if window is None: 

785 return window 

786 

787 # window must be a list 

788 assert isinstance(window, list), "Unexpected value for 'window', must be a list." 

789 

790 # return without conversion if empty list 

791 if len(window) == 0: 

792 return window 

793 

794 item_type = type(window[0]) 

795 

796 # process recursively if item is a list 

797 if item_type in (list, np.ndarray): 

798 converted = [] 

799 for w in window: 

800 LOG.trace('_freq2chan: w=%s type %s', w, type(w)) 

801 _w = self._freq2chan(spwid, w) 

802 LOG.trace('_freq2chan: _w=%s type %s', _w, type(_w)) 

803 if len(_w) == 2: 803 ↛ 799line 803 didn't jump to line 799 because the condition on line 803 was always true

804 converted.append(_w) 

805 

806 return converted 

807 

808 # return without conversion if item is an integer 

809 if issubclass(item_type, numbers.Integral): 809 ↛ 814line 809 didn't jump to line 814 because the condition on line 809 was always true

810 window.sort() 

811 return window 

812 

813 # convert floating-point value to quantity string 

814 if issubclass(item_type, numbers.Real): 

815 return self._freq2chan(spwid, ['{0}Hz'.format(x) for x in window]) 

816 

817 # now list item should be a quantity string 

818 assert item_type == str, 'unexpected item type {0}'.format(item_type) 

819 

820 # also, length of the window should be 2 

821 assert len(window) == 2 

822 

823 # frequency conversion from LSRK to TOPO 

824 new_window = self._lsrk2topo(spwid, window) 

825 

826 # construct ms channel selection syntax 

827 spwsel = self._construct_msselection(spwid, new_window) 

828 

829 # channel mapping using ms tool 

830 processed = self._string2dict(spwsel) 

831 

832 # target spwid should exist 

833 assert spwid in processed 

834 

835 new_window = sorted(processed[spwid]) 

836 LOG.trace('_freq2chan: new_window=%s type %s', new_window, type(new_window)) 

837 if len(new_window) == 0: 

838 return [] 

839 assert len(new_window) == 1 

840 return new_window[0]

842 def _lsrk2topo(self, spwid: int, window: list[str]) -> list[str]: 

843 """Apply frame conversion to line window frequencies in LSRK as needed. 

844 

845 Args: 

846 spwid: spw id 

847 window: Line window list in LSRK frequency 

848 

849 Returns: 

850 Line window list in the frame that spw is defined. In the case 

851 of ALMA data, spw is defined in TOPO frame so the returned 

852 frequency values are the ones in TOPO. 

853 """ 

854 # if frequency frame for target spw is LSRK, just return input window 

855 spw = self.ms.get_spectral_window(spwid) 

856 frame = spw.frame 

857 if frame == 'LSRK': 

858 return window 

859 

860 # assuming that measure tool is properly initialized 

861 qa = casa_tools.quanta 

862 qfreq = [qa.quantity(x) for x in window] 

863 if qa.gt(qfreq[0], qfreq[1]): 

864 qfreq = [qfreq[1], qfreq[0]] 

865 mfreq = [self.me.frequency(rf='LSRK', v0=x) for x in qfreq] 

866 new_mfreq = [self.me.measure(v=x, rf=frame) for x in mfreq] 

867 new_window = ['{value}{unit}'.format(**x['m0']) for x in new_mfreq] 

868 return new_window
