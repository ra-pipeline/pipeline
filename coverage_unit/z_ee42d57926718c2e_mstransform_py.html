107 @vdp.VisDependentProperty 

108 def field(self): 

109 

110 # Find fields in the current ms that have been observed 

111 # with the desired intent 

112 fields = self.ms.get_fields(intent=self.intent) 

113 

114 # When an observation is terminated the final scan can be a TARGET scan but may 

115 # not contain any TARGET data for the science spectral windows, e.g. there is square 

116 # law detector data but nothing else. The following code removes fields that do not contain 

117 # data for the requested spectral windows - the science spectral windows by default. 

118 

119 if fields: 

120 fields_by_id = sorted(fields, key=operator.attrgetter('id')) 

121 last_field = fields_by_id[-1] 

122 

123 # While we're here, remove any fields that are not linked with a source. This should 

124 # not occur since the CAS-9499 tablereader bug was fixed, but check anyway. 

125 if getattr(last_field, 'source', None) is None: 

126 fields.remove(last_field) 

127 LOG.info('Truncated observation detected (no source for field): ' 

128 'removing Field {!s}'.format(last_field.id)) 

129 else: 

130 # .. and then any fields that do not contain the 

131 # requested spectral windows. This should prevent 

132 # aborted scans from being split into the TARGET 

133 # measurement set. 

134 requested_spws = set(self.ms.get_spectral_windows(self.spw)) 

135 if last_field.valid_spws.isdisjoint(requested_spws): 

136 LOG.info('Truncated observation detected (missing spws): ' 

137 'removing Field {!s}'.format(last_field.id)) 

138 fields.remove(last_field) 

139 

140 unique_field_names = {f.name for f in fields} 

141 field_ids = {f.id for f in fields} 

142 

143 # Fields with different intents may have the same name. Check for this 

144 # and return the ids rather than the names if necessary to resolve any 

145 # ambiguities 

146 if len(unique_field_names) == len(field_ids): 

147 return ','.join(unique_field_names) 

148 else: 

149 return ','.join([str(i) for i in field_ids])

154 # Find all the spws with TARGET intent. These may be a subset of the 

155 # science spws which include calibration spws. 

156 @vdp.VisDependentProperty 

157 def spw(self): 

158 science_target_intents = set(self.intent.split(',')) 

159 science_target_spws = [] 

160 

161 science_spws = [spw for spw in self.ms.get_spectral_windows(science_windows_only=True)] 

162 for spw in science_spws: 

163 if spw.intents.intersection(science_target_intents): 

164 science_target_spws.append(spw) 

165 

166 return ','.join([str(spw.id) for spw in science_target_spws])

455 def _get_task_args_list(self): 

456 

457 valid_args_list = [] 

458 inputs = self.inputs 

459 original_vis = inputs.vis 

460 original_input_data_type = inputs.input_data_type 

461 original_all_eph_obj = inputs.all_eph_obj 

462 

463 # hif_mstransform needs to work on multiple datatypes at once. 

464 # Therefore the auto-selected vis list needs to be modified. 

465 data_types_groups = [ 

466 [DataType.SELFCAL_LINE_SCIENCE, DataType.SELFCAL_CONTLINE_SCIENCE], 

467 [DataType.REGCAL_LINE_SCIENCE, DataType.REGCAL_CONTLINE_SCIENCE], 

468 [DataType.REGCAL_CONTLINE_ALL] 

469 ] 

470 vis_list = [] 

471 data_type_list = [] 

472 found_data_type_group = False 

473 for data_type_group in data_types_groups: 

474 for data_type in data_type_group: 

475 ms_objects = inputs.context.observing_run.get_measurement_sets_of_type(dtypes=[data_type], msonly=True) 

476 if ms_objects: 

477 found_data_type_group = True 

478 vis_list.extend([ms_object.name for ms_object in ms_objects]) 

479 data_type_list.extend(len(ms_objects)*[data_type]) 

480 if found_data_type_group: 

481 break 

482 

483 if not found_data_type_group: 

484 return [] 

485 

486 if not self._frames_consistent(vis_list): 

487 raise Exception('Cannot handle mixture of ephemeris and non-ephemeris objects.') 

488 

489 if not self._conversion_possible(vis_list, data_type_list): 

490 return [] 

491 

492 try: 

493 for vis, data_type in zip(vis_list, data_type_list): 

494 inputs.vis = vis 

495 inputs.input_data_type = data_type 

496 inputs.all_eph_obj = all(self._get_eph_frames([vis])) 

497 task_args = inputs.as_dict() 

498 valid_args_list.append(task_args) 

499 finally: 

500 inputs.vis = original_vis 

501 inputs.input_data_type = original_input_data_type 

502 inputs.all_eph_obj = original_all_eph_obj 

503 

504 if inputs.per_spw: 

505 valid_args_list_per_spw = [] 

506 for args in valid_args_list: 

507 spw_value = args.get('spw') 

508 # If no SPW is specified, keep the original args unchanged for this entry. 

509 if spw_value is None: 

510 valid_args_list_per_spw.append(args) 

511 continue 

512 

513 spw_list = [s.strip() for s in spw_value.split(',') if s.strip()] 

514 for spw in spw_list: 

515 args_spw = args.copy() 

516 args_spw['spw'] = spw 

517 base, ext = os.path.splitext(args_spw['outputvis']) 

518 args_spw['outputvis'] = f'{base}.spw{spw}{ext}' 

519 valid_args_list_per_spw.append(args_spw) 

520 

521 valid_args_list = valid_args_list_per_spw 

522 

523 if inputs.per_spw: 

524 LOG.info( 

525 'Mstransform will be executed per SPW; the number of tasks to be executed is %d', 

526 len(valid_args_list), 

527 ) 

528 else: 

529 LOG.info( 

530 'Mstransform will be executed; the number of tasks to be executed is %d', 

531 len(valid_args_list), 

532 ) 

533 

534 return valid_args_list

536 def prepare(self): 

537 

538 assessed = [] 

539 parallel = mpihelpers.parse_parallel_input_parameter(self.inputs.parallel) 

540 task_args_list = self._get_task_args_list() 

541

544 emptyResults.task = self.__class__ 

545 emptyResults.inputs = self.inputs.as_dict() 

546 emptyResults.stage_number = self.inputs.context.task_counter

551 with TaskQueue(parallel=taskqueue_parallel_request, executor=self._executor) as tq:

554 tq.add_pipelinetask(SerialMstransform, task_args, self.inputs.context)

558 vis = task_args['vis']

566 else:
