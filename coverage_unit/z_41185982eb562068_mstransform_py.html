105class VlaMstransform(mst.SerialMstransform): 

106 Inputs = VlaMstransformInputs

108 def prepare(self): 

109 inputs = self.inputs 

110 

111 # Run CASA task to create the output MS for continuum data 

112 mstransform_args = inputs.to_casa_args() 

113 

114 # Remove input member variables that don't belong as input to casatasks/mstransform 

115 for key in ('outputvis_for_line', 'spw_line', 'omit_contline_ms'): 

116 mstransform_args.pop(key, None) 

117 

118 mstransform_job = casa_tasks.mstransform(**mstransform_args) 

119 

120 try: 

121 self._executor.execute(mstransform_job) 

122 except OSError as ee: 

123 LOG.warning(f"Caught mstransform exception: {ee}") 

124 

125 # Copy across requisite XML files. 

126 mst.SerialMstransform._copy_xml_files(inputs.vis, inputs.outputvis) 

127 

128 # Update ms history. 

129 mst.SerialMstransform._update_history(inputs.vis, inputs.outputvis) 

130 

131 if not self.inputs.omit_contline_ms: 

132 # Create output MS for cont+line data (_target.ms) 

133 self._create_targets_ms(inputs, mstransform_args) 

134 

135 # Create the results structure 

136 result = VlaMstransformResults(vis=inputs.vis, outputvis=inputs.outputvis, 

137 outputvis_for_line=inputs.outputvis_for_line) 

138 

139 return result 

140 

141 def analyse(self, result): 

142 

143 # Check for existence of the output vis. 

144 if not os.path.exists(result.outputvis): 

145 LOG.debug('Could not create science targets cont+line MS for continuum: %s' % (os.path.basename(result.outputvis))) 

146 return result 

147 

148 # Check for existence of the output vis for line processing. 

149 if not os.path.exists(result.outputvis_for_line): 

150 LOG.info('Did not create science targets cont+line MS for line imaging: %s. Subsequent stages will not do line imaging.' % (os.path.basename(result.outputvis_for_line))) 

151 

152 # Import the new measurement sets. 

153 try: 

154 to_import = os.path.relpath(result.outputvis) 

155 self._import_new_ms(result, to_import, datatype=DataType.REGCAL_CONT_SCIENCE) 

156 if os.path.exists(result.outputvis_for_line): 

157 to_import_for_line = os.path.relpath(result.outputvis_for_line) 

158 self._import_new_ms(result, to_import_for_line, datatype=DataType.REGCAL_CONTLINE_SCIENCE) 

159 except Exception: 

160 traceback.print_exc() 

161 msg = "Failed to import new measurement sets." 

162 raise Exception(msg) 

163 

164 return result

166 def _create_targets_ms(self, inputs, mstransform_args) -> bool: 

167 """Create _targets.ms for line imaging. 

168 

169 This will be created if pre-RFI flags exist and there are spectral lines in 

170 any spws. 

171 

172 Returns True if targets.ms was created, else False. 

173 """ 

174 produce_lines_ms = False 

175 

176 # Split off non-RFI flagged target data 

177 # The main goal is to get an MS with the same shape as the _target.ms to 

178 # get the flags for non-RFI flagged data 

179 

180 # Identify flags from before RFI flagging was applied 

181 pre_rfi_flagversion_name = None 

182 flags_list_task = casa_tasks.flagmanager(vis=inputs.vis, mode="list") 

183 flags_dict = self._executor.execute(flags_list_task) 

184 for value in flags_dict.values(): 

185 if 'name' in value: 

186 if 'hifv_checkflag_target-vla' in value['name']: 

187 pre_rfi_flagversion_name = value['name'] 

188 

189 if pre_rfi_flagversion_name is None: 

190 msg = "For {}: could not locate the pre-RFI flags to restore, so no targets.ms file will be created".format(inputs.vis) 

191 LOG.warning(msg) 

192 return False 

193 

194 # Restore flags from before RFI flagging was applied 

195 task = casa_tasks.flagmanager(vis=inputs.vis, mode='restore', versionname=pre_rfi_flagversion_name) 

196 self._executor.execute(task) 

197 

198 # Run CASA task to create the output MS for the line data 

199 mstransform_args['outputvis'] = inputs.outputvis_for_line 

200 

201 if inputs.spw_line: 

202 mstransform_args['spw'] = inputs.spw_line 

203 produce_lines_ms = True 

204 else: # check to see if any spws have been identified as spectral lines for this MS 

205 specline_spws = [] 

206 for spw in inputs.ms.get_spectral_windows(science_windows_only=True): 

207 if spw.specline_window: 

208 specline_spws.append(spw) 

209 if specline_spws: 

210 mstransform_args['spw'] = ','.join([str(spw.id) for spw in specline_spws]) 

211 produce_lines_ms = True 

212 

213 if produce_lines_ms: 

214 mstransform_job = casa_tasks.mstransform(**mstransform_args) 

215 

216 try: 

217 self._executor.execute(mstransform_job) 

218 except OSError as ee: 

219 LOG.warning(f"Caught mstransform exception: {ee}") 

220 

221 # Save flags from line MS without rfi flagging 

222 task = casa_tasks.flagmanager(vis=inputs.outputvis_for_line, mode='save', versionname=pre_rfi_flagversion_name) 

223 self._executor.execute(task) 

224 

225 # Copy across requisite XML files. 

226 mst.SerialMstransform._copy_xml_files(inputs.vis, inputs.outputvis_for_line) 

227 

228 # Update ms history 

229 mst.SerialMstransform._update_history(inputs.vis, inputs.outputvis_for_line) 

230 

231 # Restore RFI flags to main MS 

232 task = casa_tasks.flagmanager(vis=inputs.vis, mode='restore', versionname='rfi_flagged_statwt') 

233 self._executor.execute(task) 

234 

235 return produce_lines_ms

237 def _import_new_ms(self, result, to_import, datatype): 

238 observing_run = tablereader.ObservingRunReader.get_observing_run(to_import) 

239 

240 # Adopt same session as source measurement set 

241 for ms in observing_run.measurement_sets: 

242 LOG.debug('Setting session to %s for %s', self.inputs.ms.session, ms.basename) 

243 ms.session = self.inputs.ms.session 

244 LOG.debug('Setting data_column and origin_ms.') 

245 ms.origin_ms = self.inputs.ms.origin_ms 

246 ms.set_data_column(datatype, 'DATA') 

247 # Propagate spectral line spw designation from source MS 

248 for spw in ms.get_all_spectral_windows(): 

249 spw.specline_window = self.inputs.ms.get_spectral_window(spw.id).specline_window 

250 result.mses.extend(observing_run.measurement_sets)

262 return

264 target = context.observing_run 

265 

266 # Adding mses to context

269 target.add_measurement_set(ms)

272 for ms in self.mses: 

273 template_flagsfile = os.path.join( 

274 self.inputs['output_dir'], os.path.splitext(os.path.basename(self.vis))[0] + '.flagtargetstemplate.txt') 

275 self._make_template_flagfile(template_flagsfile, 'User flagging commands file for the imaging pipeline') 

276 

277 # Initialize callibrary 

278 for ms in self.mses: 

279 calto = callibrary.CalTo(vis=ms.name) 

280 LOG.info('Registering {} with callibrary'.format(ms.name)) 

281 context.callibrary.add(calto, [])
