141 flagpct_per_field = np.zeros((len(flags_by_field), 3))

184 def _plot_grid(self, x, y, z, nx=200, ny=200): 

185 

186 x[x < 0] = x[x < 0] + 360. 

187 

188 xi, dx = np.linspace(np.max(x), np.min(x), nx, retstep=True) 

189 yi, dy = np.linspace(np.min(y), np.max(y), ny, retstep=True) 

190 zi = griddata((x, y), z, (xi[None, :], yi[:, None]), method='cubic') 

191 with np.errstate(invalid='ignore'): 

192 zi[zi > 100] = 100. 

193 

194 ax = plt.gca() 

195 im = ax.imshow(zi, origin='lower', extent=[np.max(x)-0.5*dx, np.min(x) + 0.5*dx, 

196 np.min(y)-0.5*dy, np.max(y)+0.5*dy], aspect='equal') 

197 ax.plot(x, y, 'k.') 

198 

199 dxy = min((np.max(x)-np.min(x))*0.05, (np.max(y)-np.min(y))*0.05) 

200 ax.set_xlim(np.max(x)+dxy, np.min(x) - dxy) 

201 ax.set_ylim(np.min(y)-dxy, np.max(y)+dxy) 

202 ax.get_xaxis().get_major_formatter().set_useOffset(False) 

203 cax = make_axes_locatable(ax).append_axes("right", size="5%", pad=0.05)
