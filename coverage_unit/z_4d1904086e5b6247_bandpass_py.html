17def removeRows(caltable, spwids): 

18 """Remove rows from specified spwid from a CASA caltable.""" 

19 

20 with casa_tools.TableReader(caltable, nomodify=False) as tb: 

21 for spwid in spwids: 

22 subtb = tb.query('SPECTRAL_WINDOW_ID == '+str(spwid)) 

23 flaggedrows = subtb.rownumbers() 

24 if len(flaggedrows) > 0: 

25 LOG.debug('removing rows from table '+caltable+' for spw='+str(spwid)) 

26 tb.removerows(flaggedrows) 

27 subtb.close() 

28

30 for spwid in spwids: 

31 temparray = tb.getcol('FLAG_ROW')

33 tb.putcol('FLAG_ROW', temparray)

105def _compute_median_snr(caltable: str) -> dict: 

106 """Compute the median S/N for each SPW in the given caltable, ignoring flagged 

107 and NaN values. Returns a dictionary mapping SPW ID to median S/N. 

108 """ 

109 

110 median_snr = {} 

111 

112 with casa_tools.TableReader(caltable) as table: 

113 # Get all unique SPWs 

114 uniq_spws = np.unique(table.getcol('SPECTRAL_WINDOW_ID')) 

115 

116 for spw in uniq_spws: 

117 # Query the subtable for this SPW 

118 query_str = f'SPECTRAL_WINDOW_ID=={spw}' 

119 subtable = table.query(query=query_str) 

120 # Read SNR and FLAG columns 

121 snr = subtable.getcol('SNR') 

122 flag = subtable.getcol('FLAG') 

123 # Compute median SNR 

124 if snr.size == 0: 

125 median_snr[spw] = 0 

126 subtable.close() 

127 continue 

128 valid_snr = snr[(~flag) & (~np.isnan(snr))].ravel() 

129 # if there are no valid SNR values, set median to 0 

130 if valid_snr.size: 

131 median_snr[spw] = np.median(valid_snr) 

132 else: 

133 median_snr[spw] = 0 

134 subtable.close() 

135 

136 return median_snr

139def do_bandpass(vis, caltable, context=None, RefAntOutput=None, spw=None, ktypecaltable=None, 

140 bpdgain_touse=None, solint=None, append=None, executor=None, solnorm=None): 

141 """Run CASA task bandpass.""" 

142 m = context.observing_run.get_ms(vis) 

143 bandpass_field_select_string = context.evla['msinfo'][m.name].bandpass_field_select_string 

144 bandpass_scan_select_string = context.evla['msinfo'][m.name].bandpass_scan_select_string 

145 minBL_for_cal = vla_minbaselineforcal() 

146 

147 try: 

148 setjy_results = context.results[0].read()[0].setjy_results 

149 except Exception as e: 

150 setjy_results = context.results[0].read().setjy_results 

151 

152 BPGainTables = sorted(context.callibrary.active.get_caltable()) 

153 BPGainTables.append(ktypecaltable) 

154 BPGainTables.append(bpdgain_touse) 

155 

156 bandpass_task_args = {'vis': vis, 

157 'caltable': caltable, 

158 'field': '', 

159 'spw': spw, 

160 'intent': '', 

161 'selectdata': True, 

162 'uvrange': '', 

163 'scan': bandpass_scan_select_string, 

164 'solint': solint, 

165 'combine': 'scan', 

166 'refant': ','.join(RefAntOutput), 

167 'minblperant': minBL_for_cal, 

168 'minsnr': 5.0, 

169 'solnorm': solnorm, 

170 'bandtype': 'B', 

171 'fillgaps': 0, 

172 'smodel': [], 

173 'append': append, 

174 'docallib': False, 

175 'gaintable': BPGainTables, 

176 'gainfield': [''], 

177 'interp': [''], 

178 'spwmap': [], 

179 'parang': True} 

180 

181 bpscanslist = list(map(int, bandpass_scan_select_string.split(','))) 

182 scanobjlist = m.get_scans(scan_id=bpscanslist) 

183 allfieldidlist = [] 

184 for scanobj in scanobjlist: 

185 fieldobj, = scanobj.fields 

186 if str(fieldobj.id) not in allfieldidlist: 

187 allfieldidlist.append(str(fieldobj.id)) 

188 

189 # See vlascanheuristics - only use the first bandpass calibrator 

190 fieldidlist = [fieldid for fieldid in allfieldidlist if fieldid in bandpass_field_select_string] 

191 

192 for fieldidstring in fieldidlist: 

193 fieldid = int(fieldidstring) 

194 uvrangestring = uvrange(setjy_results, fieldid) 

195 bandpass_task_args['field'] = fieldidstring 

196 bandpass_task_args['uvrange'] = uvrangestring 

197 if os.path.exists(caltable): 

198 bandpass_task_args['append'] = True 

199 

200 job = casa_tasks.bandpass(**bandpass_task_args) 

201 

202 executor.execute(job) 

203 

204 # PIPE-2512: Re-run bandpass for low-S/N SPWs with smoothing 

205 median_snrs = _compute_median_snr(caltable) 

206 low_snr_spws = [spw for spw, snr in median_snrs.items() if snr < 50.0] 

207 good_snr_spws = [spw for spw, snr in median_snrs.items() if snr >= 50.0] 

208 if not low_snr_spws: 

209 LOG.info("All SPWs have median S/N ≥ 50 — no rerun needed.") 

210 return {} 

211 

212 LOG.info(f"Re-running bandpass for SPWs with low S/N: {low_snr_spws}") 

213 # rename the old caltable 

214 if os.path.exists(caltable): 

215 timestamp = time.strftime("%Y%m%d_%H%M%S") 

216 os.rename(caltable, f"{caltable}.allspws.{timestamp}.bak") 

217 if len(good_snr_spws) != 0: 

218 # Run bandpass using high-SNR SPWs only 

219 bandpass_task_args['spw'] = ','.join(map(str, good_snr_spws)) 

220 bandpass_task_args['append'] = False 

221 job = casa_tasks.bandpass(**bandpass_task_args) 

222 executor.execute(job) 

223 else: 

224 LOG.warning("No SPWs with median S/N ≥ 50, so skipping bandpass run.") 

225 spw_solint = {} 

226 # re-run the low-SNR SPWs individually with smoothing 

227 for bad_spw in low_snr_spws: 

228 snr = median_snrs[bad_spw] 

229 nchan = m.get_spectral_window(bad_spw).num_channels 

230 if nchan < 16: 

231 continue 

232 Nbin = nchan / 16 if snr <= 0 else min((50.0 / snr) ** 2, nchan / 16) 

233 start = int(math.ceil(Nbin)) 

234 

235 for i in range(start, nchan+1): 

236 if nchan % i == 0: 

237 Nbin = i 

238 break 

239 spw_solint[bad_spw] = Nbin 

240 solint_smooth = f"inf,{int(Nbin)}ch" 

241 

242 LOG.info(f"SPW {bad_spw}: median S/N={snr:.1f}, rerunning with solint='{solint_smooth}'") 

243 

244 bandpass_task_args.update({ 

245 'spw': str(bad_spw), 

246 'solint': solint_smooth, 

247 'append': True, 

248 }) 

249 

250 job = casa_tasks.bandpass(**bandpass_task_args) 

251 executor.execute(job) 

252 

253 return spw_solint

298def weakbp(vis, caltable, context=None, RefAntOutput=None, ktypecaltable=None, 

299 bpdgain_touse=None, solint=None, append=None, executor=None, spw='', solnorm=None): 

300 

301 m = context.observing_run.get_ms(vis) 

302 channels = m.get_vla_numchan() # Number of channels before averaging 

303 

304 bpjob = do_bandpassweakbp(vis, caltable, context=context, spw=spw, RefAntOutput=RefAntOutput, 

305 ktypecaltable=ktypecaltable, bpdgain_touse=bpdgain_touse, solint='inf', append=False, solnorm=solnorm) 

306 executor.execute(bpjob) 

307 (largechunk, spwids) = computeChanFlag(vis, caltable, context) 

308 # print largechunk, spwids 

309 if not largechunk and spwids == []: 

310 # All solutions found - proceed as normal with the pipeline 

311 interp = '' 

312 return interp 

313 

314 LOG.warning("Solutions for all channels not obtained. Using weak bandpass calibration heuristic.") 

315 cpa = 2 # Channel pre-averaging 

316 while largechunk: 

317 

318 LOG.info("Removing rows in table " + caltable + " for spws="+','.join([str(i) for i in spwids])) 

319 removeRows(caltable, spwids) 

320 solint = 'inf,' + str(cpa) + 'ch' 

321 LOG.warning("Largest contiguous set of channels with no BP solution is greater than maximum " + 

322 "allowable 1/32 fractional bandwidth for spw="+','.join([str(i) for i in spwids]) 

323 + "." + " Using solint=" + solint) 

324 LOG.info('Weak bandpass calibration heuristic. Using solint='+solint) 

325 bpjob = do_bandpassweakbp(vis, caltable, context=context, RefAntOutput=RefAntOutput, 

326 spw=','.join([str(i) for i in spwids]), 

327 ktypecaltable=ktypecaltable, bpdgain_touse=bpdgain_touse, solint=solint, append=True) 

328 executor.execute(bpjob) 

329 (largechunk, spwids) = computeChanFlag(vis, caltable, context) 

330 for spw in spwids: 

331 preavgnchan = channels[spw]/float(cpa) 

332 LOG.debug("CPA: " + str(cpa) + " NCHAN: "+str(preavgnchan)+" NCHAN/32: "+str(preavgnchan/32.0)) 

333 if cpa > preavgnchan/32.0: 

334 LOG.warning("Limiting pre-averaging to maximum 1/32 fractional bandwidth for spw="+str(spw) 

335 + ". Interpolation in applycal will need to extend over greater " + 

336 "than 1/32 fractional bandwidth, which may fail to capture significant bandpass structure.") 

337 largechunk = False # This will break the while loop and move onto applycal 

338 cpa = cpa * 2 

339 

340 LOG.warning("Channel gaps in bandpass solutions will be linearly interpolated over in applycal.") 

341 LOG.warning("Accuracy of bandpass solutions will be slightly degraded at interpolated channels, " + 

342 "particularly if these fall at spectral window edges where applycal will " + 

343 "perform 'nearest' extrapolation.") 

344 interp = 'nearest' 

345 return interp
