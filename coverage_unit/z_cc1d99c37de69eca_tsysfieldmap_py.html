18def get_intent_to_tsysfield_map(ms: MeasurementSet, is_single_dish: bool) -> dict[str, str]: 

19 """ 

20 Get the mapping of observing intent to gainfield parameter for a 

21 measurement set. 

22 

23 The mapping follows the observing intent to gainfield intent defined in 

24 CAS-12213. 

25 

26 Args: 

27 ms: MS to analyse. 

28 is_single_dish: boolean for if SD data or not. 

29 

30 Returns: 

31 Dictionary of {observing intent: gainfield}. 

32 """ 

33 soln_map = get_solution_map(ms, is_single_dish) 

34 final_map = {s.intent: s.preferred if s.preferred else s.fallback for s in soln_map} 

35 

36 # Detect cases where there's no preferred or fallback gainfield mapping, 

37 # e.g., if there are no Tsys scans on a target or phase calibrator. 

38 undefined_intents = [k for k, v in final_map.items() 

39 if not v # gainfield mapping is empty.. 

40 and k in ms.intents] # ..for a valid intent in the MS 

41 if undefined_intents: 

42 msg = 'Undefined Tsys gainfield mapping for {} intents: {}'.format(ms.basename, undefined_intents) 

43 LOG.error(msg) 

44 raise AssertionError(msg) 

45 

46 # convert magic string back to empty string 

47 converted = {k: v.replace('___EMPTY_STRING___', '') for k, v in final_map.items()} 

48 

49 return converted

52def get_tsys_fields_for_intent(ms: MeasurementSet, intent: str, exclude_intents: str | None = None) -> list[str]: 

53 """Return the identity of the Tsys field(s) for an intent. 

54 

55 Args: 

56 ms: MS to analyse. 

57 intent: Intent to retrieve fields for. 

58 exclude_intents: String of intent(s) (comma-separated) that should not 

59 be covered by the Tsys field. 

60 

61 Returns: 

62 Insertion-ordered list of unique field identifiers corresponding to given intent, while not 

63 associated with intents (optionally) given by ``exclude_intents``. 

64 """ 

65 # In addition to the science intent scan, a field must also have a Tsys 

66 # scan observed for a Tsys solution to be considered present. The 

67 # exception is science mosaics, which are handled as a special case. 

68 

69 # We need to know which science intent scans have Tsys scans; the ones 

70 # that don't will be checked for science mosaics separately. This lets 

71 # us handle single field, single pointing science targets alongside mosaic 

72 # targets mixed together in the same EB. Theoretically, at least... 

73 intent_fields = ms.get_fields(intent=intent) 

74 

75 # PIPE-2394: If requested, avoid matching fields that cover any of the 

76 # intents that are to be excluded. 

77 if exclude_intents is not None: 

78 intent_fields = [f for f in intent_fields if f.intents.isdisjoint(set(exclude_intents.split(',')))] 

79 

80 # contains fields of this intent that also have a companion Tsys scan 

81 intent_fields_with_tsys = [f for f in intent_fields if 'ATMOSPHERE' in f.intents] 

82 

83 # contains fields without a companion Tsys scan. These might be science 

84 # mosaics. 

85 intent_fields_without_tsys = [f for f in intent_fields if f not in intent_fields_with_tsys] 

86 

87 tsys_fields_for_mosaics = [] 

88 if intent == 'TARGET': 

89 # In science mosaics, the fields comprising the TARGET pointings do 

90 # not have Tsys scans observed on those fields. Instead, there is a 

91 # Tsys-only field roughly at the centre of the mosaic that is 

92 # referenced by the same parent source as the TARGET pointing fields. 

93 

94 # Double check that the fields without Tsys scans are indeed science 

95 # mosaics with a separate Tsys field. Note that a mosaic consisting of 

96 # a source with a single TARGET pointing and a single Tsys scan would 

97 # also be classified as a mosaic by this logic. 

98 mosaic_fields = [f for f in intent_fields_without_tsys if 'ATMOSPHERE' in f.source.intents] 

99 

100 # Collect the Tsys fields referenced by the parent source of the 

101 # science mosaic fields missing Tsys scans. 

102 tsys_fields_for_mosaics = [f 

103 for pointing in mosaic_fields 

104 for f in pointing.source.fields if 'ATMOSPHERE' in f.intents] 

105 

106 # when field names are not unique, as is usually the case for science 

107 # mosaics, then we must reference the numeric field ID instead 

108 field_identifiers = utils.get_field_identifiers(ms) 

109 

110 # Preserve insertion order while deduplicating 

111 all_fields = [field_identifiers[field.id] for field in intent_fields_with_tsys] 

112 all_fields.extend(field_identifiers[field.id] for field in tsys_fields_for_mosaics) 

113 return utils.deduplicate(all_fields)

116def get_solution_map(ms: MeasurementSet, is_single_dish: bool) -> list[GainfieldMapping]: 

117 """ 

118 Get gainfield solution map. Different solution maps are returned for 

119 single dish and interferometric data. 

120 

121 Args: 

122 ms: MS to analyse. 

123 is_single_dish: True if MS is single dish data. 

124 

125 Returns: 

126 List of GainfieldMappings. 

127 """ 

128 # define function to get Tsys fields for intent 

129 def f(intent, exclude: str | None = None) -> str: 

130 if ',' in intent: 

131 head, tail = intent.split(',', 1) 

132 # Split comma-separated results into individual fields and deduplicate 

133 # at the field ID level (not at the string level) 

134 all_fields = [] 

135 for field_str in (f(head, exclude=exclude), f(tail, exclude=exclude)): 

136 if field_str: 

137 all_fields.extend(field_str.split(',')) 

138 unique_fields = utils.deduplicate(all_fields) 

139 return ','.join(unique_fields) 

140 return ','.join(str(s) for s in get_tsys_fields_for_intent(ms, intent, exclude_intents=exclude)) 

141 

142 # return different gainfield maps for single dish and interferometric 

143 if is_single_dish: 

144 # PIPE-2869: added fallback of ATMOSPHERE for TARGET intent 

145 # 

146 # Intent to be calibrated: 

147 # - BANDPASS cal 

148 # * Preferred: all BANDPASS cals. 

149 # * Fallback: 'nearest'. 

150 # - AMPLITUDE cal 

151 # * Preferred: all AMPLITUDE cals. 

152 # * Fallback: 'nearest'. 

153 # - TARGET 

154 # * Preferred: TARGET and/or ATMOSPHERE sources. 

155 # * Fallback: null fallback 

156 return [ 

157 GainfieldMapping(intent='BANDPASS', preferred=f('BANDPASS'), fallback='nearest'), 

158 GainfieldMapping(intent='AMPLITUDE', preferred=f('AMPLITUDE'), fallback='nearest'), 

159 # non-empty magic string to differentiate between no field found and a null fallback 

160 GainfieldMapping(intent='TARGET', preferred=f('TARGET,ATMOSPHERE'), fallback='___EMPTY_STRING___') 

161 ] 

162 

163 else: 

164 # CAS-12213: original intent mapping. 

165 # PIPE-2080: updated to add mapping for DIFFGAINREF, DIFFGAINSRC intent. 

166 # PIPE-2394: updated mapping for PHASE, TARGET, CHECK 

167 # 

168 # Intent to be calibrated: 

169 # - BANDPASS cal 

170 # * Preferred: all BANDPASS cals. 

171 # * Fallback: 'nearest'. 

172 # - AMPLITUDE cal 

173 # * Preferred: all AMPLITUDE cals. 

174 # * Fallback: 'nearest'. 

175 # - DIFFGAIN[REF|SRC] 

176 # * Preferred: all DIFFGAIN cals. 

177 # * Fallback to BANDPASS. 

178 # - PHASE cal 

179 # * Preferred: ATMOSPHERE cals, but excluding AMP, BP, DIFFGAIN*, POL* cals, and TARGET. 

180 # * Fallback: ATMOSPHERE cals, but excluding AMP, BP, DIFFGAIN*, POL* cals. 

181 # - TARGET 

182 # * Preferred: ATMOSPHERE cals, but excluding AMP, BP, DIFFGAIN*, POL* cals, and PHASE. 

183 # * Fallback: ATMOSPHERE cals, but excluding AMP, BP, DIFFGAIN*, POL* cals. 

184 # - CHECK_SOURCE 

185 # * Preferred: ATMOSPHERE cals, but excluding AMP, BP, DIFFGAIN*, POL* cals, and PHASE. 

186 # * Fallback: ATMOSPHERE cals, but excluding AMP, BP, DIFFGAIN*, POL* cals. 

187 

188 # PIPE-2394: typical calibrator intents to avoid (all but PHASE) 

189 # matching searching for nearby Tsys field for PHASE, TARGET, and/or 

190 # CHECK. 

191 exclude_intents = 'AMPLITUDE,BANDPASS,DIFFGAINREF,DIFFGAINSRC,POLARIZATION,POLANGLE,POLLEAKAGE' 

192 

193 return [ 

194 GainfieldMapping(intent='BANDPASS', preferred=f('BANDPASS'), fallback='nearest'), 

195 GainfieldMapping(intent='AMPLITUDE', preferred=f('AMPLITUDE'), fallback='nearest'), 

196 GainfieldMapping(intent='DIFFGAINREF', preferred=f('DIFFGAINREF'), fallback=f('BANDPASS')), 

197 GainfieldMapping(intent='DIFFGAINSRC', preferred=f('DIFFGAINSRC'), fallback=f('BANDPASS')), 

198 GainfieldMapping(intent='PHASE', preferred=f('ATMOSPHERE', exclude=f'{exclude_intents},TARGET'), 

199 fallback=f('ATMOSPHERE', exclude=exclude_intents)), 

200 GainfieldMapping(intent='TARGET', preferred=f('ATMOSPHERE', exclude=f'{exclude_intents},PHASE'), 

201 fallback=f('ATMOSPHERE', exclude=exclude_intents)), 

202 GainfieldMapping(intent='CHECK', preferred=f('ATMOSPHERE', exclude=f'{exclude_intents},PHASE'), 

203 fallback=f('ATMOSPHERE', exclude=exclude_intents)), 

204 ]
