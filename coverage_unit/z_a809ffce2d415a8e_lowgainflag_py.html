351 def _do_initial_bandpass(self): 

352 """ 

353 This method is responsible for doing the  

354 calls to make a bandpass table. The method will call 

355 either of the PhcorBandpass, or SerialALMAPhcorBandpass 

356 processes for non-ALMA and ALMA data respecitvely. 

357 

358 With ALMA data, the PIPE-2442 process is used, i.e. an  

359 SNR check is used to combine spw and change solint time for  

360 the pre-bandpass phase-up, and thereafter the bandpass solver  

361 will also bin channels for a suitable SNR bandpass solutions 

362  

363 For other data, the default process (as <=PL2025) will  

364 not combine spw, and uses int for all phase-up solves 

365 while using a default 7.8125MHz channel binning for bandpass 

366 """ 

367 

368 inputs = self.inputs 

369 

370 # PIPE-2601 check ALMA or not 

371 if inputs.ms.antenna_array.name == 'ALMA': 

372 bpcal_inputs = bandpass_alma.SerialALMAPhcorBandpass.Inputs( 

373 context=inputs.context, vis=inputs.vis, intent=inputs.intent, 

374 spw=inputs.spw, refant=inputs.refant, phaseupsnr=5, bpsnr=20) 

375 bpcal_task = bandpass_alma.SerialALMAPhcorBandpass(bpcal_inputs) 

376 bpcal_result = self._executor.execute(bpcal_task, merge=False) 

377 else: 

378 # Not ALMA 

379 bpcal_inputs = bandpass.PhcorBandpass.Inputs( 

380 context=inputs.context, vis=inputs.vis, intent=inputs.intent, 

381 spw=inputs.spw, refant=inputs.refant, solint='inf,7.8125MHz') 

382 bpcal_task = bandpass.PhcorBandpass(bpcal_inputs) 

383 bpcal_result = self._executor.execute(bpcal_task, merge=False) 

384 

385 # Default fallback used when bandpass produces no final solution. 

386 solint, combine = 'int', '' 

387 if not bpcal_result.final: 

388 LOG.warning("No bandpass solution computed for %s", inputs.ms.basename) 

389 else: 

390 bpcal_result.accept(inputs.context) 

391 # PIPE-2601 get the phase-up solution interval and combine parameters 

392 # as these need to be passed to the subsequent gaincal to avoid loss of data. 

393 # The function returns 'int', '' for non-ALMA data. 

394 solint, combine = self._get_prebandpass_phaseup_params(bpcal_result) 

395 

396 return bpcal_result, solint, combine

398 def _do_postbandpass_phasediff(self): 

399 """  

400 This method is responsible for doing the post 

401 bandpass solve for the phasediff in the case if it  

402 needed prior to postbandpass_phaseup. It is required 

403 only if combine='spw', because a phasediff is required 

404 to align the phases of all SpW to allow combination. 

405 The bandpass in local context will be applied and then a phase 

406 gain caltable is made. 

407  

408 returns the gpdiff result but it is registered here 

409 """ 

410 

411 inputs = self.inputs 

412 

413 # Calculate gain phase offsets (PIPE-3139) 

414 # PIPE-2601 added the use of combine, so a phasediff (single somve per spw) 

415 # is required to align all SpWs before postbandpass_phaseup 

416 gpdiff_inputs = gaincal.GTypeGaincal.Inputs(context=inputs.context, vis=inputs.vis, intent=inputs.intent, 

417 spw=inputs.spw, refant=inputs.refant, calmode='p', minsnr=2.0, 

418 solint='inf') 

419 gpdiff_task = gaincal.GTypeGaincal(gpdiff_inputs) 

420 gpdiff_result = self._executor.execute(gpdiff_task, merge=False) 

421 if not gpdiff_result.final: 

422 LOG.warning("No phase offset solution computed for {}".format(inputs.ms.basename)) 

423 else: 

424 gpdiff_result.accept(inputs.context) 

425 

426 # pass the result back in case any subsequent or future code want this 

427 return gpdiff_result

429 def _do_postbandpass_phaseup(self, solint, combine): 

430 """  

431 This method is responsible for doing the post 

432 bandpass solve phaseup. The bandpass in local context 

433 will be applied and then a phase gain caltable is made. 

434 Inputs are solint and combine, as to use values previously  

435 found that will get a sufficient SNR and not cause  

436 undue lost solutions due to low SNR for e.g. High Freq and 

437 ACA data where SNR is intrinsically low and an assumed per 

438 SpW, solint = 'int' solution will not gurantee enough SNR 

439  

440 returns the gpcal result but it is registered here 

441 """ 

442 

443 inputs = self.inputs 

444 

445 # Calculate gain phases 

446 # PIPE-2601 added to use solint and combine found in pre-phaseup done before bandpass 

447 # required to have high enough SNR and avoid flags due to this solve 

448 # if combine used need to add spw mapping for ALMA data. 

449 # Otherwise, 'int' and '' are passes for solint and combine such as to make 

450 # the default phase-up 

451 gpcal_inputs = gaincal.GTypeGaincal.Inputs(context=inputs.context, vis=inputs.vis, intent=inputs.intent, 

452 spw=inputs.spw, refant=inputs.refant, calmode='p', minsnr=2.0, 

453 solint=solint, combine=combine) 

454 gpcal_task = gaincal.GTypeGaincal(gpcal_inputs) 

455 gpcal_result = self._executor.execute(gpcal_task, merge=False) 

456 if not gpcal_result.final: 

457 LOG.warning("No phase time solution computed for {}".format(inputs.ms.basename)) 

458 

459 else: 

460 # PIPE-2601 before accepting to context 

461 # need to set the spwmap such that the pre-apply when solving amp gains 

462 # will map correctly. Can only trigger for ALMA data where combine has an entry 

463 if combine: 

464 spwmap = inputs.ms.get_spectral_windows(task_arg=inputs.spw, science_windows_only=True) 

465 combspwmapheur = combine_spwmap(spwmap) 

466 

467 LOG.info('Using combined spw map %s for ALMA phaseup', combspwmapheur) 

468 

469 # and interpolation for any solves after should be linearPD 

470 interp = 'linearPD,linear' 

471 modified_calapp = callibrary.copy_calapplication(gpcal_result.pool[0], spwmap=combspwmapheur, interp=interp) 

472 gpcal_result.pool[0] = modified_calapp 

473 gpcal_result.final[0] = modified_calapp 

474 gpcal_result.accept(inputs.context) 

475 

476 # pass the result back in case any subsequent or future code want this 

477 return gpcal_result

480 def _do_postbandpass_ampgains(self): 

481 """ 

482 This method is responsible for creating the amp gain 

483 caltable for the BANDPASS intent and making a fixed 'inf'  

484 solution amplitude gains solve. 

485  

486 The resulting caltable will be locally registered and  

487 investigated for low gain values 

488 """ 

489 

490 inputs = self.inputs 

491 

492 # Calculate gain amplitudes 

493 gacal_inputs = gaincal.GTypeGaincal.Inputs( 

494 context=inputs.context, vis=inputs.vis, 

495 intent=inputs.intent, spw=inputs.spw, 

496 refant=inputs.refant, 

497 calmode='a', minsnr=2.0, solint='inf', gaintype='T') 

498 gacal_task = gaincal.GTypeGaincal(gacal_inputs) 

499 gacal_result = self._executor.execute(gacal_task, merge=False) 

500 if not gacal_result.final: 

501 gatable = list(gacal_result.error) 

502 gatable = gatable[0].gaintable 

503 LOG.warning("No amplitude time solution computed for {}".format(inputs.ms.basename)) 

504 table_available = False 

505 else: 

506 gacal_result.accept(inputs.context) 

507 gatable = gacal_result.final 

508 gatable = gatable[0].gaintable 

509 table_available = True 

510 

511 return gacal_result, gatable, table_available

513 def _get_prebandpass_phaseup_params(self, result): 

514 """ 

515 Read the result object passed from the bandpass task for ALMA data 

516 as to loop the pre-applied gaintables, and to extract the input args used 

517 for solint and combine so these can be output, and will later propagate to  

518 the phase-up _post_ bandpass and _pre_ amp gains 

519 

520 :param: result, the bpcal result from bandpass.SerialALMAPhCorBandpass 

521 :return: solint, combine as strings to gaincal input 

522 """ 

523 

524 # Default parameters for ALMA and non-ALMA data 

525 solint_use, combine_use = 'int', '' 

526 

527 # PIPE-2601 protection here to only work on ALMA data 

528 if self.inputs.ms.antenna_array.name == 'ALMA': 

529 phaseup_calapps = [] 

530 # loop applies that 'preceded' the bandpass solve, i.e. the phaseup/phase-offset 

531 for previous_result in result.preceding: 

532 for calapp in previous_result: 

533 gaincal_froms = [cf for cf in calapp.calfrom if cf.caltype == 'gaincal'] 

534 if gaincal_froms and calapp not in phaseup_calapps: 

535 phaseup_calapps.append(calapp) 

536 

537 # extract solint and combine from the calapp applied 

538 # there will be a maximum of two calapps for ALMA; 

539 # if spw combination was triggered in the ALMAPhcorBandpass method 

540 # the first calapp is the phase offset with solint=inf, 

541 # the second is the phase-up with solint = 'time', this is what we extract here. 

542 for calapp in phaseup_calapps: 

543 solint = utils.get_origin_input_arg(calapp, 'solint') 

544 combine = utils.get_origin_input_arg(calapp, 'combine') 

545 if solint != 'inf': 

546 solint_use = solint 

547 combine_use = combine 

548 

549 return solint_use, combine_use

581 def calculate_view(self, table): 

582 """ 

583 table -- Name of gain table to be analysed. 

584 """ 

585 

586 # Open gains caltable. 

587 gtable = caltableaccess.CalibrationTableDataFiller.getcal(table) 

588 

589 # Get range of scans covered. 

590 scans = set() 

591 for row in gtable.rows: 

592 # The gain table is T, should be no pol dimension 

593 npols = np.shape(row.get('CPARAM'))[0] 

594 if npols != 1: 

595 raise Exception('table has polarization results') 

596 scans.update([row.get('SCAN_NUMBER')]) 

597 scans = np.sort(list(scans)) 

598 

599 # Create translation of scan ID to flagging view axis ID. 

600 scanid_to_axisid = {scan_id: axis_id for axis_id, scan_id in enumerate(scans)} 

601 

602 # Get the MS domain object. 

603 ms = self.context.observing_run.get_ms(name=self.vis) 

604 

605 # Get spw IDs from MS: the task input "spw" arg may contain channel 

606 # specification, so let MS parse input and read spw ID from the 

607 # SpectralWindow domain objects that are returned. 

608 spwids = [spw.id for spw in ms.get_spectral_windows(self.spw)] 

609 

610 # Identify set of unique antenna diameters present in MS. 

611 ant_diameters = {antenna.diameter for antenna in ms.antennas} 

612 

613 # Create separate flagging view for each set of antennas with 

614 # same diameter. 

615 for antdiam in ant_diameters: 

616 # Identify antennas with current diameter. 

617 antenna_ids = sorted([antenna.id for antenna in ms.antennas if antenna.diameter == antdiam]) 

618 

619 # Create translation of antenna ID to flagging view axis ID. 

620 antid_to_axisid = {ant_id: axis_id for axis_id, ant_id in enumerate(antenna_ids)} 

621 nants = len(antenna_ids) 

622 

623 # If the number of antennas is below the threshold, then skip 

624 # flagging for the set with current diameter. 

625 if nants < self.min_nants_threshold: 

626 LOG.warning( 

627 "Number of antennas with diameter of {:.1f} m is" 

628 " below the minimum threshold ({}), skipping" 

629 " flagging for these antennas (no flagging view" 

630 " created).".format(antdiam, self.min_nants_threshold)) 

631 else: 

632 # Create flagging view for each spwid. 

633 for spwid in spwids: 

634 

635 # Initialize arrays for flagging view. 

636 data = np.zeros([nants, len(scans)]) 

637 flag = np.ones([nants, len(scans)], bool) 

638 

639 for row in gtable.rows: 

640 ant = row.get('ANTENNA1') 

641 if row.get('SPECTRAL_WINDOW_ID') == spwid and ant in antenna_ids: 

642 gain = row.get('CPARAM')[0][0] 

643 gainflag = row.get('FLAG')[0][0] 

644 scan = row.get('SCAN_NUMBER') 

645 if not gainflag: 

646 data[antid_to_axisid[ant], scanid_to_axisid[scan]] = np.asarray(np.abs(gain)).item() 

647 flag[antid_to_axisid[ant], scanid_to_axisid[scan]] = 0 

648 

649 axes = [ 

650 commonresultobjects.ResultAxis(name='Antenna1', units='id', data=np.asarray(antenna_ids)), 

651 commonresultobjects.ResultAxis(name='Scan', units='id', data=[str(scan) for scan in scans], 

652 channel_width=1) 

653 ] 

654 

655 # associate the result with a generic filename - using 

656 # specific names gives confusing duplicates on the weblog 

657 # display 

658 ants = ','.join([str(ant) for ant in antenna_ids]) 

659 viewresult = commonresultobjects.ImageResult( 

660 filename='%s(gtable)' % os.path.basename(gtable.vis), 

661 intent=self.intent, data=data, flag=flag, axes=axes, 

662 datatype='gain amplitude', spw=spwid, 

663 ant=ants) 

664 

665 # add the view results and their children results to the 

666 # class result structure 

667 self.result.addview(viewresult.description, viewresult)
