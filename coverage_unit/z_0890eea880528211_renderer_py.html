233 def get_skycal_applications(self, context: Context, result: skycal_task.SDSkyCalResults, ms: MeasurementSet) -> list[dict]: 

234 """Get application information from SDSkyCalResults instance and set them into a list. 

235 

236 Args: 

237 context: Pipeline context object containing state information. 

238 result: SDSkyCalResults instance. 

239 ms: MeasurementSet domain object. 

240 

241 Returns: 

242 A list "application" containing dictionary; they are used to a table in skycal.mako file. 

243 """ 

244 applications = [] 

245 

246 calmode_map = {'ps': 'Position-switch', 

247 'otfraster': 'OTF raster edge'} 

248 

249 calapps = result.final 

250 for calapp in calapps: 

251 caltype = calmode_map[utils.get_origin_input_arg(calapp, 'calmode')] 

252 gaintable = os.path.basename(calapp.gaintable) 

253 spw = calapp.spw.replace(',', ', ') 

254 intent = calapp.intent.replace(',', ', ') 

255 antenna = calapp.antenna 

256 if antenna == '': 

257 antenna = ', '.join([a.name for a in ms.antennas]) 

258 field = ms.get_fields(calapp.field)[0].name 

259 

260 applications.append({'ms': ms.basename, 

261 'gaintable': gaintable, 

262 'spw': spw, 

263 'intent': intent, 

264 'field': field, 

265 'antenna': antenna, 

266 'caltype': caltype}) 

267 

268 return applications

270 def _get_reference_coord(self, context: Context, ms: MeasurementSet, field: Field) -> str: 

271 """Get celestial coordinates and the reference. 

272 

273 Args: 

274 context: Pipeline context object containing state information. 

275 ms: MeasurementSet domain object. 

276 field: field domain object. 

277 

278 Returns: 

279 Reference, RA and Declination. 

280 

281 Raises: 

282 RuntimeError: No OFF source data exist for given field in the ms. 

283 """ 

284 LOG.debug('_get_reference_coord({ms}, {field})'.format(ms=ms.basename, field=field.name)) 

285 spws = ms.get_spectral_windows(science_windows_only=True) 

286 data_desc_ids = [ms.get_data_description(spw=spw.id).id for spw in spws] 

287 reference_states = [state for state in ms.states if 'REFERENCE' in state.intents] 

288 state_ids = [state.id for state in reference_states] 

289 field_id = field.id 

290 with casa_tools.TableReader(ms.name) as tb: 

291 t = tb.query(f'ANTENNA1==ANTENNA2 && FIELD_ID=={field_id} && DATA_DESC_ID IN {utils.list_to_str(data_desc_ids)} && STATE_ID IN {utils.list_to_str(state_ids)}') 

292 if t.nrows() == 0: 

293 t.close() 

294 raise RuntimeError(f'No OFF source data for field "{field.name}" (id {field_id}) in {ms.basename}') 

295 

296 rownumbers = t.rownumbers() 

297 antenna_id = t.getcell('ANTENNA1', 0) 

298 timestamp = t.getcell('TIME', 0) 

299 t.close() 

300 timeref = tb.getcolkeyword('TIME', 'MEASINFO')['Ref'] 

301 timeunit = tb.getcolkeyword('TIME', 'QuantumUnits')[0] 

302 with casa_tools.MSMDReader(ms.name) as msmd: 

303 pointing_direction = msmd.pointingdirection(rownumbers[0]) 

304 antenna_position = msmd.antennaposition(antenna_id) 

305 qa = casa_tools.quanta 

306 me = casa_tools.measures 

307 epoch = me.epoch(rf=timeref, v0=qa.quantity(timestamp, timeunit)) 

308 

309 LOG.debug('pointing_direction=%s' % pointing_direction) 

310 

311 direction = pointing_direction['antenna1']['pointingdirection'] 

312 me.doframe(antenna_position) 

313 me.doframe(epoch) 

314 # 2018/04/18 TN 

315 # CAS-10874 single dish pipeline should use a direction reference 

316 # taken from input MS 

317 origin_basename = os.path.basename(ms.origin_ms) 

318 datatable_name = os.path.join(context.observing_run.ms_datatable_name, origin_basename) 

319 datatable = DataTable() 

320 datatable.importdata(datatable_name, minimal=False, readonly=True) 

321 outref = datatable.direction_ref 

322 

323 converted = me.measure(direction, rf=outref) 

324 LOG.debug('converted direction=%s' % converted) 

325 if outref.upper() == 'GALACTIC': 

326 xformat = 'dms' 

327 else: 

328 xformat = 'hms' 

329 coord = '{ref} {ra} {dec}'.format(ref=outref, 

330 ra=qa.formxxx(converted['m0'], format=xformat), 

331 dec=qa.formxxx(converted['m1'], format='dms')) 

332 

333 return coord
