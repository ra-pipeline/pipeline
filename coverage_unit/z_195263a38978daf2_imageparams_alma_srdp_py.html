23 def uvtaper(self, beam_natural=None, protect_long=3, beam_user=None, tapering_limit=None, repr_freq=None): 

24 """ 

25 This code will take a given beam and a desired beam size and calculate the necessary 

26 UV-tapering parameters needed for tclean to recreate that beam. 

27 

28 UV-tapering parameter larger than the 80 percentile baseline is not allowed. 

29 

30 :param beam_natural: natural beam, dictionary with major, minor and positionangle keywords 

31 :param beam_user: desired beam, dictionary with major, minor and positionangle keywords 

32 :param tapering_limit: 190th baseline in meters. uvtaper larger than this baseline is not allowed 

33 :param repr_freq: representative frequency, dictionary with unit and value keywords. 

34 :return: uv_taper needed to recreate user_beam in tclean 

35 """ 

36 if beam_natural is None: 

37 return [] 

38 if beam_user is None: 

39 return [] 

40 if tapering_limit is None: 

41 return [] 

42 if repr_freq is None: 

43 return [] 

44 

45 # Determine uvtaper based on equations from Ryan Loomis, 

46 # https://open-confluence.nrao.edu/display/NAASC/Data+Processing%3A+Imaging+Tips 

47 # See PIPE-704. 

48 cqa = casa_tools.quanta 

49 

50 bmajor = 1.0 / cqa.getvalue(cqa.convert(beam_natural['major'], 'arcsec')) 

51 bminor = 1.0 / cqa.getvalue(cqa.convert(beam_natural['minor'], 'arcsec')) 

52 

53 des_bmajor = 1.0 / cqa.getvalue(cqa.convert(beam_user['major'], 'arcsec')) 

54 des_bminor = 1.0 / cqa.getvalue(cqa.convert(beam_user['minor'], 'arcsec')) 

55 

56 if (des_bmajor > bmajor) or (des_bminor > bminor): 

57 LOG.warning('uvtaper cannot be calculated for beam_user (%.2farcsec) larger than beam_natural (%.2farcsec)' % ( 

58 1.0 / des_bmajor, 1.0 / bmajor)) 

59 return [] 

60 

61 tap_bmajor = 1.0 / (bmajor * des_bmajor / (math.sqrt(bmajor ** 2 - des_bmajor ** 2))) 

62 tap_bminor = 1.0 / (bminor * des_bminor / (math.sqrt(bminor ** 2 - des_bminor ** 2))) 

63 

64 # Assume symmetric beam 

65 tap_angle = math.sqrt((tap_bmajor ** 2 + tap_bminor ** 2) / 2.0) 

66 

67 # Convert angle to baseline 

68 ARCSEC_PER_RAD = 206264.80624709636 

69 uvtaper_value = ARCSEC_PER_RAD / tap_angle 

70 LOG.info('uvtaper needed to achive user specified angular resolution is %.2fklambda' % 

71 utils.round_half_up(uvtaper_value / 1000., 2)) 

72 

73 # Determine maximum allowed uvtaper 

74 # PIPE-1104: Limit such that the image includes the baselines from at least 20 antennas. 

75 # Limit should be set to: (N*(N-1)/2)=the length of the 190th baseline. 

76 uvtaper_limit = tapering_limit / cqa.getvalue(cqa.convert(cqa.constants('c'), 'm/s'))[0] * \ 

77 cqa.getvalue(cqa.convert(repr_freq, 'Hz'))[0] 

78 # Limit uvtaper 

79 if uvtaper_value < uvtaper_limit: 

80 uvtaper_value = uvtaper_limit 

81 LOG.warning('uvtaper is smaller than allowed limit of %.2fklambda, the length of the 190th baseline, using the limit value' % 

82 utils.round_half_up(uvtaper_limit / 1000., 2)) 

83 return ['%.2fklambda' % utils.round_half_up(uvtaper_value / 1000., 2)]
