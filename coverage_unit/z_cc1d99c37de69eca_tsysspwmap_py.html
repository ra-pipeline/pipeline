29def trim_spw_map(spw_map: list[int]) -> list[int]: 

30 """ 

31 Trim a spw map (list of SpW IDs) to remove tail end where SpWs are mapped to 

32 themselves. 

33 

34 E.g. 

35 [0, 1, 2, 2, 4, 4, 6, 7, 8, 9] 

36 gets trimmed to: 

37 [0, 1, 2, 2, 4, 4] 

38 

39 This assumes that CASA will by default map SpWs to themselves if they do not 

40 appear in the spw map. 

41 

42 Args: 

43 spw_map: SpW map to trim. 

44 

45 Returns: 

46 Trimmed SpW map. 

47 """ 

48 for i in reversed(range(len(spw_map))): 

49 if spw_map[i] != i: 

50 return spw_map[:i + 1] 

51 return []

54def tsysspwmap(ms: MeasurementSet, tsystable: str | None = None, channel_tolerance: int = 1, trim: bool = True)\ 

55 -> tuple[list[int], list[int]]: 

56 """ 

57 Generate default spectral window map for ALMA Tsys, including TDM->FDM 

58 associations. 

59 

60 This function generates an "applycal-ready" SpW-to-Tsys-SpW mapping that 

61 provides the appropriate information regarding the transfer of the Tsys 

62 calibration from TDM spectral windows to FDM spectral windows. 

63 

64 If provided a Tsys Caltable that you wish to apply to a MeasurementSet, then 

65 it filters to only use Tsys SpW(s) that appear in the caltable; otherwise, 

66 it will use all Tsys SpWs present in MS and assume these are present in the 

67 caltable. 

68 

69 The resulting tsys spwmap can then be supplied to the applycal spwmap 

70 parameter to ensure proper Tsys calibration application. 

71 

72 Args: 

73 ms: The measurement set (object) to process. 

74 tsystable: Optional, the input Tsys caltable (w/ TDM Tsys measurements), 

75 used to identify which Tsys SpW(s) are available. 

76 channel_tolerance: Frequency tolerance in number of channels to use in 

77 matching Tsys SpW freq. range to SpW freq. range. 

78 trim: If True (the default), return minimum-length spwmap; otherwise the 

79 spwmap will be exhaustive and include the high-numbered (and usually 

80 irrelevant) WVR SpWs. 

81 

82 Returns: 

83 2-tuple containing: 

84 - list of unmatched SpW IDs. 

85 - list representing the SpW-to-Tsys-SpW mapping. 

86 """ 

87 # Case of Nobeyama(=NRO) data 

88 if ms.antenna_array.name == "NRO": 

89 LOG.debug('Mapping process between Tsys windows and Science windows is not specially needed.') 

90 spw_mapping = [spw.id for spw in ms.get_spectral_windows()] 

91 return [], spw_mapping 

92 

93 # Retrieve all spectral windows. 

94 spectral_windows = ms.get_spectral_windows(science_windows_only=False) 

95 

96 # Identify the Tsys spectral windows. 

97 # 

98 # If a Tsys caltable is provided, then select spectral windows with entries 

99 # in the Tsys solutions table. 

100 if tsystable is not None: 

101 with casa_tools.TableReader(tsystable) as table: 

102 spwids_in_tsyscaltable = set(table.getcol("SPECTRAL_WINDOW_ID")) 

103 tsys_spectral_windows = [spw for spw in spectral_windows if spw.id in spwids_in_tsyscaltable] 

104 # Otherwise, select spectral windows in MS that cover ATMOSPHERE intent and 

105 # are of type "TDM". 

106 else: 

107 tsys_spectral_windows = [spw for spw in spectral_windows if 'ATMOSPHERE' in spw.intents and spw.type == 'TDM'] 

108 

109 # Precompute frequency tolerance bounds for each Tsys SpW. 

110 tsys_bounds = { 

111 spw.id: ( 

112 spw.min_frequency - channel_tolerance * spw.channels[0].getWidth(), 

113 spw.max_frequency + channel_tolerance * spw.channels[-1].getWidth() 

114 ) 

115 for spw in tsys_spectral_windows 

116 } 

117 

118 # Initialize SpW-to-Tsys-Spw-mapping with one-to-one mapping as default. 

119 spw_mapping = list(range(len(spectral_windows))) 

120 unmatched_spws = [] 

121 

122 # Match each SpW to a suitable Tsys SpW with matching frequency range, same 

123 # baseband, and same spectral spec. 

124 for spw in spectral_windows: 

125 for tsys_spw in tsys_spectral_windows: 

126 min_bound, max_bound = tsys_bounds[tsys_spw.id] 

127 if ( 

128 spw.baseband == tsys_spw.baseband 

129 and spw.spectralspec == tsys_spw.spectralspec 

130 and spw.min_frequency >= min_bound 

131 and spw.max_frequency <= max_bound 

132 ): 

133 spw_mapping[spw.id] = tsys_spw.id 

134 break 

135 else: 

136 unmatched_spws.append(spw.id) 

137 

138 # Log in case any science SpWs had no matching Tsys SpW. 

139 unmatched_scispws = [spw.id for spw in ms.get_spectral_windows() if spw.id in unmatched_spws] 

140 if unmatched_scispws: 

141 LOG.info(f"No Tsys match found for science SpW(s) {utils.commafy(unmatched_scispws, False)}.") 

142 

143 # Trim off "excess SpWs" (SpWs mapped to themselves) from map if asked. 

144 if trim: 

145 spw_mapping = trim_spw_map(spw_mapping) 

146 

147 LOG.info(f"Computed tsysspwmap is: {spw_mapping}") 

148 

149 return unmatched_scispws, spw_mapping
