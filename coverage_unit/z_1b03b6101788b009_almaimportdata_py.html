178 def prepare(self, **parameters) -> Results: 

179 results = super().prepare() 

180 

181 # If online flux services were requested but unavailable, signal an error 

182 # after weblog generation to stop the pipeline before the next task. 

183 if getattr(self.inputs, 'dbservice', False) and results.fluxservice == 'FAIL': 

184 results.tb = ( 

185 'Online flux catalog unavailable; fell back to local Source.xml fluxes. ' 

186 'Stopping after weblog export.' 

187 ) 

188 

189 return results

191 def _get_fluxes( 

192 self, 

193 context: Context, 

194 observing_run: ObservingRun, 

195 ) -> tuple[str | None, list[FluxCalibrationResults], list[dict[str, str | None]] | None]: 

196 # get the flux measurements from Source.xml for each MS 

197 

198 if self.inputs.dbservice: 

199 testquery = '?DATE=27-March-2013&FREQUENCY=86837309056.169219970703125&WEIGHTED=true&RESULT=1&NAME=J1427-4206&VERBOSE=1' 

200 # Test for service response 

201 flux_url, backup_flux_url = dbfluxes.get_flux_urls() 

202 url = flux_url + testquery 

203 

204 try: 

205 ssl_context = ssl.create_default_context(cafile=certifi.where()) 

206 LOG.info('Attempting test query at: %s', url) 

207 urllib.request.urlopen(url, context=ssl_context, timeout=60.0) 

208 xml_results, qastatus = dbfluxes.get_setjy_results(observing_run.measurement_sets) 

209 fluxservice = 'FIRSTURL' 

210 except Exception: 

211 try: 

212 LOG.warning('Unable to execute initial test query with primary flux service.') 

213 traceback_msg = traceback.format_exc() 

214 LOG.debug(traceback_msg) 

215 ssl_context = ssl.create_default_context(cafile=certifi.where()) 

216 url = backup_flux_url + testquery 

217 LOG.info('Attempting test query at backup: %s', url) 

218 urllib.request.urlopen(url, context=ssl_context, timeout=60.0) 

219 xml_results, qastatus = dbfluxes.get_setjy_results(observing_run.measurement_sets) 

220 fluxservice='BACKUPURL' 

221 except Exception as e2: 

222 LOG.warning(('Unable to execute backup test query with flux service.\n' 

223 'Proceeding without using the online flux catalog service.')) 

224 traceback_msg = traceback.format_exc() 

225 LOG.debug(traceback_msg) 

226 xml_results = fluxes.get_setjy_results(observing_run.measurement_sets) 

227 fluxservice = 'FAIL' 

228 qastatus = None 

229 else: 

230 xml_results = fluxes.get_setjy_results(observing_run.measurement_sets) 

231 fluxservice = None 

232 qastatus = None 

233 # write/append them to flux.csv 

234 

235 # Cycle 1 hack for exporting the field intents to the CSV file: 

236 # export_flux_from_result queries the context, so we pseudo-register 

237 # the mses with the context by replacing the original observing run 

238 orig_observing_run = context.observing_run 

239 context.observing_run = observing_run 

240 try: 

241 fluxes.export_flux_from_result(xml_results, context) 

242 finally: 

243 context.observing_run = orig_observing_run 

244 

245 # re-read from flux.csv, which will include any user-coded values 

246 combined_results = fluxes.import_flux(context.output_dir, observing_run) 

247 

248 return fluxservice, combined_results, qastatus
