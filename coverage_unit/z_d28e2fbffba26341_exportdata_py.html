152

155 oussid : OUS status ID 

156 session_names : list of session names 

157 session_vislists : list of MeasurementSet names associated with each session 

158 products_dir : path of products directory

297 def _export_final_baseline_applylist(self, context: Context, vis: str, 

298 products_dir: str) -> str: 

299 """Save commands to apply the final baseline table to a file. 

300 

301 For now this is a text file. Eventually it will be the CASA callibrary 

302 file. 

303 

304 Args: 

305 context : pipeline context 

306 vis : MeasurementSet name 

307 products_dir : path of products directory 

308 

309 Returns: 

310 the name of calibration apply file 

311 """ 

312 applyfile_name = self.NameBuilder.calapply_list(os.path.basename(vis), 

313 aux_product=True) 

314 # applyfile_name = os.path.basename(vis) + '.auxcalapply.txt' 

315 LOG.info('Storing calibration apply list for %s in %s', 

316 os.path.basename(vis), applyfile_name) 

317 

318 try: 

319 # Log the list in human readable form. Better way to do this ? 

320 cmd = string.Template("sdbaseline(infile='${infile}', " 

321 "datacolumn='corrected', spw='${spw}', " 

322 "blmode='apply', bltable='${bltable}', " 

323 "blfunc='poly', outfile='${outfile}', " 

324 "overwrite=True)") 

325 

326 # Create the list of baseline caltable for that vis 

327 name = self.__get_last_baseline_table(vis) 

328 ms = context.observing_run.get_ms(vis) 

329 science_spws = ms.get_spectral_windows(science_windows_only=True) 

330 spw = ','.join([str(s.id) for s in science_spws]) 

331 if name is not None: 

332 applied_calstate = \ 

333 cmd.safe_substitute(infile=vis, bltable=name, spw=spw, 

334 outfile=vis.rstrip('/') + '_bl') 

335 

336 # Open the file. 

337 with open(os.path.join(products_dir, applyfile_name), "w") \ 

338 as applyfile: 

339 applyfile.write('# Apply file for %s\n' 

340 % (os.path.basename(vis))) 

341 applyfile.write(applied_calstate) 

342 except Exception: 

343 applyfile_name = 'Undefined' 

344 LOG.info('No calibrations for MS %s' % os.path.basename(vis)) 

345 

346 return applyfile_name

348 def _detect_jyperk(self, context: Context) -> str: 

349 """Detect K2Jy file and return it. 

350 

351 Args: 

352 context : pipeline context 

353 

354 Raises: 

355 RuntimeError: raise if multiple K2Jy files are detected 

356 

357 Returns: 

358 path of K2Jy file 

359 """ 

360 reffile_list = set(self.__get_reffile(context.results)) 

361 

362 if len(reffile_list) == 0: 

363 # if no reffile is found, return None 

364 LOG.debug('No K2Jy factor file found.') 

365 return None 

366 if len(reffile_list) > 1: 

367 raise RuntimeError("K2Jy conversion file must be only one. %s found." 

368 % (len(reffile_list))) 

369 

370 jyperk = reffile_list.pop() 

371 

372 if not os.path.exists(jyperk): 

373 # if reffile doesn't exist, return None 

374 LOG.debug('K2Jy file \'%s\' not found' % jyperk) 

375 return None 

376 

377 LOG.info('Exporting {0} as a product'.format(jyperk)) 

378 return absolute_path(jyperk)

381 def __get_reffile(results: Generator[str, None, None]): 

382 """Find SDK2JyCalResults and yield K2JY reference file. 

383 

384 Args: 

385 results : a list of ResultsProxy 

386 (contains Results object of every tasks) 

387 

388 Yields: 

389 K2JY reference file, default: jyperk.csv 

390 """ 

391 for proxy in results: 

392 result = proxy.read() 

393 if not isinstance(result, basetask.ResultsList): 

394 result = [result] 

395 for r in result: 

396 if str(r).find('SDK2JyCalResults') != -1 and hasattr(r, 'reffile'): 

397 reffile = r.reffile 

398 if reffile is not None and os.path.exists(reffile): 

399 yield reffile

401 def _do_auxiliary_products(self, context: Context, oussid: str, 

402 output_dir: str, products_dir: str, 

403 pipeline_stats_file: str) -> str: 

404 """Save a K2JY reference file and flag files into tarball. 

405 

406 Args: 

407 context : pipeline context 

408 oussid : OUS Status UID 

409 output_dir : path of output directory 

410 products_dir : path of products directory 

411 pipeline_stats_file: pipeline stats file 

412 

413 Returns: 

414 tarball file name 

415 """ 

416 # Track whether any auxiliary products exist to be exported. 

417 aux_prod_exists = False 

418 

419 # Get the jyperk file, check whether it exists. 

420 jyperk = self._detect_jyperk(context) 

421 if jyperk and os.path.exists(jyperk): 

422 aux_prod_exists = True 

423 

424 # Export the general and target source template flagging files 

425 # The general template flagging files are not required 

426 # for the restore but are informative to the user. 

427 # Whether or not the target template files should be exported to 

428 # the archive depends on the final place of the target flagging 

429 # step in the work flow and how flags will or will not be stored 

430 # back into the ASDM. 

431 flags_file_list = [os.path.join(output_dir, fname) 

432 for fname in glob.glob('*.flag*template.txt')] 

433 if flags_file_list: 

434 aux_prod_exists = True 

435 

436 # If no auxiliary product was found, skip creation of tarfile 

437 # and return early. 

438 if not aux_prod_exists: 

439 return None 

440 

441 # Create the tarfile. 

442 # Define the name of the output tarfile. 

443 tarfilename = self.NameBuilder.auxiliary_products( 

444 'auxproducts.tgz', ousstatus_entity_id=oussid) 

445 # tarfilename = '{}.auxproducts.tgz'.format(oussid) 

446 LOG.info('Saving auxiliary data products in %s', tarfilename) 

447 

448 # Open tarfile. 

449 with tarfile.open(os.path.join(products_dir, tarfilename), 'w:gz') \ 

450 as tar: 

451 

452 # Save flux file. 

453 if jyperk and os.path.exists(jyperk): 

454 tar.add(jyperk, arcname=os.path.basename(jyperk)) 

455 LOG.info('Saving auxiliary data product ' 

456 '{} in {}'.format(os.path.basename(jyperk), tarfilename)) 

457 elif isinstance(jyperk, str): 

458 LOG.info('Auxiliary data product ' 

459 '{} does not exist'.format(os.path.basename(jyperk))) 

460 

461 # Save flag files. 

462 for flags_file in flags_file_list: 

463 if os.path.exists(flags_file): 

464 tar.add(flags_file, arcname=os.path.basename(flags_file)) 

465 LOG.info('Saving auxiliary data product ' 

466 '{} in {}'.format(os.path.basename(flags_file), tarfilename)) 

467 else: 

468 LOG.info('Auxiliary data product ' 

469 '{} does not exist'.format(os.path.basename(flags_file))) 

470 

471 # PIPE-2380: Save the pipeline statistics file 

472 if pipeline_stats_file and os.path.exists(pipeline_stats_file): 

473 tar.add(pipeline_stats_file, arcname=pipeline_stats_file) 

474 LOG.info('Saving pipeline statistics file %s in %s', pipeline_stats_file, tarfilename) 

475 else: 

476 LOG.info("Pipeline statistics file does not exist.") 

477 tar.close() 

478 

479 return tarfilename

513 def _get_hm_rasterscan_value(self, context: Context) -> str: 

514 """Retrieve hm_rasterscan value from Results object. 

515 

516 This method checks if either SDImportDataResults or 

517 SDRestoreDataResults object is registered to Pipeline context. 

518 If exists, hm_rasterscan value is retrieved from the results 

519 object. If no SDImportDataResults nor SDRestoreDataResults 

520 object is registered, 'time' will be returned. 

521 

522 Args: 

523 context: Pipeline context 

524 

525 Returns: 

526 The hm_rasterscan value. 

527 """ 

528 results_filter = filter( 

529 lambda x: isinstance(x, basetask.ResultsList) and isinstance(x[0], (SDImportDataResults, SDRestoreDataResults)), 

530 map(lambda x: x.read(), context.results) 

531 ) 

532 importdata_results = next(results_filter, None) 

533 if importdata_results: 

534 hm_rasterscan = importdata_results.inputs.get('hm_rasterscan', 'time') 

535 else: 

536 hm_rasterscan = 'time' 

537 

538 return hm_rasterscan
