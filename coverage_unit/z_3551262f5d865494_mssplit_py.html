147 # Check for existence of the output vis. 

148 if not os.path.exists(result.outputvis): 

149 return result

151 inputs = self.inputs 

152 

153 # There seems to be a rerendering issue with replace. Fir now just

174 def _set_data_column_to_ms(self, msobj: MeasurementSet): 

175 """ 

176 Set data_column to input MeasurementSet domain object. 

177 

178 This method sets data_column information of output MS depending on 

179 intent and datacolumn selection of the Inputs class. 

180 

181 Args: 

182 msobj: MS domain object to set data_column information. 

183 """ 

184 datacolumn = self.inputs.datacolumn 

185 in_column = datacolumn.upper() if datacolumn != 'corrected' else 'CORRECTED_DATA' 

186 LOG.debug('in_column = %s' % in_column) 

187 #if self.inputs.replace and not os.path.exists(msobj.origin_ms): 

188 # # Replace RAW column if original MS was replaced. 

189 # data_type = DataType.RAW 

190 #el 

191 

192 data_type = None 

193 if self.inputs.intent == 'TARGET': 

194 data_type = DataType.REGCAL_CONTLINE_SCIENCE 

195 else: 

196 for t, c in self.inputs.ms.data_column.items(): 

197 if c == in_column: 

198 data_type = t 

199 LOG.debug(f'Identified data type {data_type}') 

200 break 

201 

202 if data_type is None: 

203 data_type = DataType.RAW 

204 LOG.warning( 

205 f'The datatype of the requested datacolumn is unknown, and a fallback value of {data_type} is used.') 

206 

207 out_column = in_column if datacolumn != 'corrected' else 'DATA' 

208 LOG.info(f'Setting {data_type} to {out_column}') 

209 msobj.set_data_column(data_type, out_column)

218 def merge_with_context(self, context): 

219 # Check for an output vis 

220 if not self.ms: 

221 LOG.error('No h_mssplit results to merge') 

222 return 

223 

224 target = context.observing_run 

225 parentms = None 

226 # The parent MS has been removed. 

227 if not os.path.exists(self.vis): 

228 for index, ms in enumerate(target.get_measurement_sets()): 

229 if ms.name == self.vis: 

230 parentms = index 

231 break 

232 

233 if self.ms: 

234 if parentms is not None: 

235 LOG.info('Replace {} in context'.format(self.ms.name)) 

236 del target.measurement_sets[parentms] 

237 target.add_measurement_set(self.ms) 

238 

239 else: 

240 LOG.info('Adding {} to context'.format(self.ms.name)) 

241 target.add_measurement_set(self.ms)
