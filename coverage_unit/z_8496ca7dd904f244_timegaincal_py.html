179 def prepare(self, **parameters) -> GaincalResults: 

180 inputs = self.inputs 

181 

182 # Create a results object. 

183 result = GaincalResults() 

184 result.phasecal_for_phase_plot = [] 

185 

186 # PIPE-2268: update intents to exclude those that are not present, to 

187 # avoid these appearing in CalApplications / weblog. 

188 inputs.intent = utils.filter_intents_for_ms(inputs.ms, inputs.intent) 

189 

190 # Compute the phase solutions for the science target, check source, 

191 # and phase calibrator. This caltable will be registered as applicable 

192 # to the target, check source, and phase calibrator when the result for 

193 # this task is accepted. 

194 LOG.info('Computing phase gain table(s) for target(s), check source(s), and phase calibrator(s).') 

195 target_phasecal_calapps = self._do_phasecal_for_target() 

196 # Add the solutions to final task result, for adopting into context 

197 # callibrary, but do not merge them into local task context, so they 

198 # are not used in pre-apply of subsequent gaincal calls. 

199 result.pool.extend(target_phasecal_calapps) 

200 result.final.extend(target_phasecal_calapps) 

201 

202 # Compute the phase solutions for all calibrators in inputs.intent. 

203 # These phase cal results include solutions for the PHASE calibrator 

204 # field(s), and will be temporarily accepted into the local context to 

205 # have these available as pre-apply in subsequent gaincals (both for 

206 # amplitude solves and for computing residual phase offsets). But for 

207 # the final task result, these phase solutions will only be registered 

208 # as applicable to the bandpass, flux, differential gain, and 

209 # polarization calibrators. 

210 LOG.info('Computing phase gain table(s) for bandpass, flux, diffgain, and polarization calibrator(s).') 

211 cal_phase_results, max_phase_solint = self._do_phasecal_for_calibrators() 

212 

213 # Merge the phase solutions for the calibrators into the local task 

214 # context so that it is marked for inclusion in pre-apply (gaintable) 

215 # in subsequent gaincal calls during this task. 

216 for cpres in cal_phase_results: 

217 if not cpres.final: 

218 continue 

219 cpres.accept(inputs.context) 

220 

221 # Look through calibrator phasecal results for any CalApplications for 

222 # caltables that are applicable to non-PHASE calibrators (i.e. 

223 # AMPLITUDE, BANDPASS, POL*, and DIFFGAIN*). Add these CalApps to the 

224 # final task result, to be merged into the final context / callibrary. 

225 for cpres in cal_phase_results: 

226 if not cpres.final: 

227 continue 

228 cp_calapp = cpres.final[0] 

229 if cp_calapp.intent != 'PHASE': 

230 result.final.append(cp_calapp) 

231 result.pool.append(cp_calapp) 

232 

233 # PIPE-1377: add all results to the list to be plotted in the 

234 # phase vs. time diagnostic plots in the renderer. 

235 result.phasecal_for_phase_plot.append(cp_calapp) 

236 

237 # Compute the amplitude calibration. 

238 LOG.info('Computing the final amplitude gain table.') 

239 amplitude_calapps = self._do_target_ampcal() 

240 

241 # Accept the amplitude calibration into the final results. 

242 result.pool.extend(amplitude_calapps) 

243 result.final.extend(amplitude_calapps) 

244 

245 # Produce the diagnostic table for displaying amplitude vs time plots. 

246 # For the solint, use the maximum SpW-mapping-mode-based solint that 

247 # was used earlier for phase solutions for the PHASE calibrator(s). 

248 # This table is not applied to the data, and no special mapping is 

249 # required here. 

250 LOG.info('Computing diagnostic amplitude gain table for displaying amplitude vs time plots.') 

251 amp_diagnostic_result = self._do_caltarget_ampcal(solint=max_phase_solint) 

252 result.calampresult = amp_diagnostic_result 

253 

254 # To ensure that a diagnostic phase offsets caltable (to be created in 

255 # an upcoming step) will be sensitive to residual phase offsets, it is 

256 # necessary to enforce that the phase caltable for PHASE calibrator 

257 # fields used in pre-apply was solved with combine='spw'. But the 

258 # caltables for PHASE calibrator fields that earlier got solved and 

259 # merged into local context (for pre-apply in amplitude solves) may not 

260 # always have used combine='spw' (since it used optimal parameters 

261 # based on SpW mapping). Hence, first unregister any phase caltable for 

262 # PHASE calibrator fields where the SpW mapping recommended combine=''. 

263 LOG.info("Prior to computing diagnostic residual phase offsets, unregistering any phase gain table(s) for" 

264 " phase calibrator(s) that were derived with combine='', earlier in this stage.") 

265 self._unregister_phasecal_with_no_combine() 

266 

267 # Next, compute a new phase solutions solve for those PHASE calibrator 

268 # fields where the recommended combine was '', while this time 

269 # enforcing combine='spw'. 

270 LOG.info("Prior to computing diagnostic residual phase offsets, where necessary (caltable was unregistered)" 

271 " re-computing phase gain table(s) for phase calibrator(s) while enforcing combine='spw'.") 

272 phasecal_phase_results = self._do_phasecal_for_phase_calibrators_forcing_combine() 

273 

274 # Merge the new phase solutions for PHASE calibrator fields into the 

275 # local task context, so they are marked for inclusion in pre-apply in 

276 # subsequent diagnostic phase offsets solve. 

277 for res in phasecal_phase_results: 

278 res.accept(inputs.context) 

279 

280 # Now compute a new phase solutions caltable while using previous phase 

281 # caltables in pre-apply. Assuming that inputs.intent included 'PHASE' 

282 # (true by default, but can be overridden by user), then the merger of 

283 # the previous phase solutions result into the local task context will 

284 # mean that the initial phase corrections will be included in pre-apply 

285 # during this gaincal, and thus this new caltable will represent the 

286 # residual phase offsets. 

287 LOG.info('Computing diagnostic residual phase offsets gain table.') 

288 phase_offsets_result = self._do_offsets_phasecal() 

289 result.phaseoffsetresult = phase_offsets_result 

290 

291 return result

335 def _do_phasecal_for_target(self) -> list[callibrary.CalApplication]: 

336 """ 

337 This method is responsible for creating phase gain caltable(s) that 

338 will be applicable to the TARGET, the CHECK source, and the PHASE 

339 calibrator. 

340 

341 Separate phase solutions are created for each PHASE field, and for each 

342 SpectralSpec when a "combine" SpW mapping is used. 

343 

344 The resulting caltable(s) will be part of the final task result, with 

345 separate CalApplications to register the caltable(s) to be applicable 

346 to PHASE, as well as to TARGET/CHECK. 

347 """ 

348 inputs = self.inputs 

349 p_intent = 'PHASE' 

350 

351 # Initialize output list of CalApplications. 

352 calapp_list = [] 

353 

354 # Determine non-phase calibrator fields, to be added to gaincal solve 

355 # for plotting purposes. 

356 np_intents = ','.join(dict.fromkeys(item for item in inputs.intent.split(',') if item != p_intent)) 

357 np_fields = None 

358 if np_intents: 

359 # PIPE-2752: Identify fields that have non-phase intents but lack the PHASE intent. 

360 candidates = inputs.ms.get_fields(intent=np_intents) 

361 exclusive_fields = [f.name for f in candidates if p_intent not in f.intents] 

362 if exclusive_fields: 

363 np_fields = ','.join(dict.fromkeys(exclusive_fields)) 

364 else: 

365 LOG.debug('No exclusive non-phase fields for intents=%s selection on %s', np_intents, inputs.ms) 

366 

367 # Determine which SpWs to solve for, which SpWs the solutions should 

368 # apply to, and whether to override refantmode. By default, use all 

369 # input SpW, do not restrict what SpWs the solutions apply to, and do 

370 # not override the refantmode. 

371 spw_to_solve = inputs.spw 

372 refantmode = None 

373 apply_to_spw = None 

374 if inputs.ms.is_band_to_band: 

375 # PIPE-2087: for BandToBand, restrict the solve to the diffgain 

376 # reference SpWs, use refantmode strict for the solve, and register 

377 # the solutions to be applied to the diffgain on-source SpWs. 

378 dg_refspws = inputs.ms.get_spectral_windows(task_arg=inputs.spw, intent='DIFFGAINREF') 

379 dg_srcspws = inputs.ms.get_spectral_windows(task_arg=inputs.spw, intent='DIFFGAINSRC') 

380 spw_to_solve = ','.join(str(s.id) for s in dg_refspws) 

381 refantmode = 'strict' 

382 apply_to_spw = ','.join(str(s.id) for s in dg_srcspws) 

383 

384 # Create separate phase solutions for each PHASE field. 

385 for field in inputs.ms.get_fields(intent=p_intent): 

386 # Retrieve from MS which TARGET/CHECK fields the gain solutions for 

387 # the current PHASE field should be applied to. 

388 tc_fields = ','.join(inputs.ms.phasecal_mapping.get(field.name, {})) 

389 

390 # If the user specified a filename, then add the field name, to 

391 # ensure the filenames remain unique in case of multiple fields. 

392 caltable = None 

393 if inputs.targetphasetable: 

394 root, ext = os.path.splitext(inputs.targetphasetable) 

395 caltable = f'{root}.{field.name}{ext}' 

396 

397 # Get optimal phase solution parameters for current PHASE field, 

398 # based on spw mapping info in MS. No need to catch the value for 

399 # optimal solint as solint will be fixed to inputs.targetsolint. 

400 combine, gaintype, interp, _, spwmap = self._get_phasecal_params(p_intent, field.name) 

401 

402 # PIPE-2087: for BandToBand override interp, for these phase 

403 # solutions that will apply to the science target. 

404 if inputs.ms.is_band_to_band: 

405 interp = 'linearPD,linear' 

406 

407 # PIPE-390: if not combining across spw, then no need to deal with 

408 # SpectralSpec, so create a gaincal solution for all SpWs, using 

409 # provided gaintype, spwmap, and interp. 

410 if not combine: 

411 calapp_list.extend(self._do_target_phasecal(caltable=caltable, field=field.name, spw=spw_to_solve, 

412 gaintype=gaintype, combine=combine, spwmap=spwmap, 

413 interp=interp, apply_to_field=tc_fields, 

414 apply_to_spw=apply_to_spw, include_field=np_fields, 

415 refantmode=refantmode)) 

416 

417 # Otherwise, a combined SpW solution is expected, and we need to 

418 # create separate solutions for each SpectralSpec grouping of Spws. 

419 else: 

420 # Group the input SpWs by SpectralSpec. 

421 spw_groups = self._get_spw_groupings(inputs.ms, spw_to_solve, spwmap) 

422 if not spw_groups: 

423 raise ValueError('Invalid SpW grouping input.') 

424 

425 # Loop through each grouping of spws. 

426 for _, sspec, spw_sel in spw_groups: 

427 LOG.info(f'Processing spectral spec {sspec} with spws {spw_sel}') 

428 

429 # Check if there are scans for current intent and SpWs. 

430 selected_scans = inputs.ms.get_scans(scan_intent=p_intent, spw=spw_sel) 

431 if len(selected_scans) == 0: 

432 LOG.info(f'Skipping table generation for empty selection: spw={spw_sel}, intent={p_intent}') 

433 continue 

434 

435 # If an explicit output filename is defined, then add the 

436 # spw selection to ensure the filenames remain unique. 

437 if caltable: 

438 root, ext = os.path.splitext(caltable) 

439 caltable = f'{root}.{spw_sel}{ext}' 

440 

441 # Run phase calibration. 

442 calapp_list.extend(self._do_target_phasecal(caltable=caltable, field=field.name, spw=spw_sel, 

443 gaintype=gaintype, combine=combine, spwmap=spwmap, 

444 interp=interp, apply_to_field=tc_fields, 

445 apply_to_spw=apply_to_spw, include_field=np_fields, 

446 refantmode=refantmode)) 

447 

448 return calapp_list

450 def _do_target_phasecal(self, caltable: str = None, field: str = None, spw: str = None, gaintype: str = None, 

451 combine: str = None, interp: str = None, spwmap: list[int] = None, 

452 apply_to_field: str = None, apply_to_spw: str = None, include_field: str = None, 

453 refantmode: str | None = None)\ 

454 -> list[callibrary.CalApplication]: 

455 """ 

456 This runs the gaincal for creating phase solutions intended for TARGET, 

457 CHECK, and PHASE. The result contains two CalApplications, one for 

458 how to apply the caltable to the PHASE calibrator, and a second one for 

459 how to apply the caltable to the TARGET and CHECK source(s). 

460 """ 

461 inputs = self.inputs 

462 

463 gc_fields = field 

464 if include_field: 

465 gc_field_list = gc_fields.split(',') 

466 gc_field_list.extend(include_field.split(',')) 

467 gc_fields = ','.join(dict.fromkeys(gc_field_list)) 

468 

469 # PIPE-1154: for phase solutions of target, check, phase, always use 

470 # solint=inputs.targetsolint. 

471 task_args = { 

472 'output_dir': inputs.output_dir, 

473 'vis': inputs.vis, 

474 'caltable': caltable, 

475 'field': gc_fields, 

476 'intent': inputs.intent, 

477 'spw': spw, 

478 'solint': inputs.targetsolint, 

479 'gaintype': gaintype, 

480 'calmode': 'p', 

481 'minsnr': inputs.targetminsnr, 

482 'combine': combine, 

483 'refant': inputs.refant, 

484 'refantmode': refantmode, 

485 'minblperant': inputs.minblperant, 

486 'solnorm': inputs.solnorm 

487 } 

488 result = do_gtype_gaincal(inputs.context, self._executor, task_args) 

489 

490 # Create new CalApplications to correctly register caltable. 

491 new_calapps = [] 

492 

493 # Define what overrides should be included in the cal application. 

494 # Phase solution caltables should always be registered with 

495 # calwt=False (PIPE-1154). Adjust the interpolation and SpW mapping if 

496 # provided. 

497 calapp_overrides = {'calwt': False} 

498 if interp: 

499 calapp_overrides['interp'] = interp 

500 if spwmap: 

501 calapp_overrides['spwmap'] = spwmap 

502 

503 # Create a modified CalApplication to register this caltable against 

504 # the PHASE calibrator field itself. 

505 new_calapps.append(callibrary.copy_calapplication(result.final[0], intent='PHASE', field=field, 

506 gainfield=field, **calapp_overrides)) 

507 

508 # If the current PHASE field was mapped to TARGET/CHECK field(s), then 

509 # create a modified CalApplication to register this caltable against 

510 # those TARGET/CHECK fields, where present in the MS (PIPE-2268). 

511 if apply_to_field: 

512 # Adjust what SpWs to apply to, if provided. 

513 if apply_to_spw: 

514 calapp_overrides['spw'] = apply_to_spw 

515 intents_for_calapp = utils.filter_intents_for_ms(inputs.ms, 'CHECK,TARGET') 

516 if intents_for_calapp: 

517 new_calapps.append( 

518 callibrary.copy_calapplication( 

519 result.final[0], 

520 intent=intents_for_calapp, 

521 field=apply_to_field, 

522 gainfield=field, 

523 **calapp_overrides, 

524 ) 

525 ) 

526 

527 return new_calapps

529 def _do_phasecal_for_calibrators(self) -> tuple[list[GaincalResults], float | None]: 

530 """ 

531 This method is responsible for creating phase gain caltable(s) that 

532 are applicable to all calibrators specified in inputs.intent, typically: 

533 phase, amplitude, bandpass, diffgain(ref/src), and polarization. 

534 """ 

535 inputs = self.inputs 

536 phasecal_results = [] 

537 

538 # Split intents by PHASE and non-PHASE calibrators. 

539 p_intent = 'PHASE' 

540 np_intents = set(inputs.intent.split(',')) - {p_intent} 

541 

542 # PIPE-1154: first create a phase caltable for the non-PHASE 

543 # calibrators. 

544 if np_intents: 

545 phasecal_results.extend(self._do_phasecal_for_non_phase_calibrators(np_intents)) 

546 

547 # PIPE-1154: next, compute the phase gain solutions for the PHASE 

548 # calibrator fields. These solutions for the PHASE fields are not 

549 # intended for the final result, but will be merged into the local task 

550 # context, so that they are used in pre-apply when computing the 

551 # residual phase offsets. Track the maximum phase solint that gets 

552 # used, so the same value can be when computing the diagnostic 

553 # amplitude caltable. 

554 max_phase_solint = None 

555 if p_intent in inputs.intent: 

556 pcal_results, max_phase_solint = self._do_phasecal_for_phase_calibrators(p_intent) 

557 phasecal_results.extend(pcal_results) 

558 

559 return phasecal_results, max_phase_solint

561 def _do_phasecal_for_non_phase_calibrators(self, intents: set) -> list[GaincalResults]: 

562 """ 

563 Compute phase gain caltable(s) for the non-PHASE calibrators, typically: 

564 amplitude, bandpass, diffgain(ref/src), and polarization. 

565 """ 

566 inputs = self.inputs 

567 phasecal_results = [] 

568 

569 # PIPE-645: for bandpass, amplitude, diffgain, and polarisation intents, 

570 # always use minsnr set to 3. 

571 minsnr = 3.0 

572 

573 # Create separate gaincal for each non-phase calibrator field. 

574 fields = [f for f in inputs.ms.get_fields(intent=','.join(intents))] 

575 for field in fields: 

576 # Matching intents in current field. 

577 fld_intents = field.intents.intersection(intents) 

578 fld_intents_str = ",".join(fld_intents) 

579 LOG.info(f'Compute phase gaincal table for intent={fld_intents_str}, field={field.name}.') 

580 

581 # If this field is used as a bandpass calibrator, then retrieve the 

582 # optimal phase cal parameters for BANDPASS, and solve for all 

583 # matching intents at once. This case can cover a number of cases 

584 # of overlapping calibrators: 

585 # - BP == AMP: should use optimal parameters for BANDPASS. 

586 # - BP == DIFFGAIN*: should use optimal parameters for BANDPASS. 

587 # - BP == POL*: in this case, hifa_spwphaseup will not have 

588 # derived any optimal parameters for this field (explicitly 

589 # skips fields with POL*), so the look-up should return the 

590 # default gaincal parameters, as is required for polarization 

591 # calibrators. 

592 if "BANDPASS" in fld_intents: 

593 combine, gaintype, interp, solint, spwmap = self._get_phasecal_params('BANDPASS', field.name) 

594 phasecal_results.append( 

595 self._do_calibrator_phasecal(field=field.name, intent=fld_intents_str, spw=inputs.spw, 

596 gaintype=gaintype, combine=combine, solint=solint, minsnr=minsnr, 

597 spwmap=spwmap, interp=interp)) 

598 # If this field is a diffgain calibrator (while no overlap with 

599 # bandpass), then assume that hifa_spwphaseup will have stored 

600 # separate SpwMapping info for DIFFGAINREF and DIFFGAINSRC and 

601 # create separate solves for those. It is assumed here that if 

602 # DIFFGAINSRC is present, DIFFGAINREF must be present as well. 

603 # It is further assumed that there is no support for band-to-band 

604 # polarization, so this field should not have also POL* intents. 

605 # It is further assumed that the diffgain calibrator cannot also be 

606 # the amplitude calibrator. If it was, then the amplitude scans 

607 # would not get a phase-up solution here. 

608 elif "DIFFGAINSRC" in fld_intents: 

609 for dg_intent in {"DIFFGAINREF", "DIFFGAINSRC"}: 

610 combine, gaintype, interp, solint, spwmap = self._get_phasecal_params(dg_intent, field.name) 

611 phasecal_results.append( 

612 self._do_calibrator_phasecal(field=field.name, intent=dg_intent, spw=inputs.spw, 

613 gaintype=gaintype, combine=combine, solint=solint, minsnr=minsnr, 

614 spwmap=spwmap, interp=interp)) 

615 # For all other cases, use all intents of current field to retrieve 

616 # optimal parameters and compute phase solutions. 

617 # Typically, this would cover amplitude and/or polarization 

618 # calibrators. 

619 # Note: if this field covers both AMP and POL*, then hifa_spwphaseup 

620 # would not have derived optimal parameters as it currently skips 

621 # polarization fields; so instead this step would use default 

622 # phasecal parameters. 

623 else: 

624 combine, gaintype, interp, solint, spwmap = self._get_phasecal_params(fld_intents_str, field.name) 

625 phasecal_results.append( 

626 self._do_calibrator_phasecal(field=field.name, intent=fld_intents_str, spw=inputs.spw, 

627 gaintype=gaintype, combine=combine, solint=solint, minsnr=minsnr, 

628 spwmap=spwmap, interp=interp)) 

629 

630 return phasecal_results

632 def _do_phasecal_for_phase_calibrators(self, intent: str) -> tuple[list[GaincalResults], float]: 

633 """ 

634 This method is responsible for creating phase gain caltable(s) for each 

635 field that covers a PHASE calibrator, using optimal gaincal parameters 

636 based on the SpW mapping registered in the measurement set. 

637 """ 

638 inputs = self.inputs 

639 

640 # Initialize list of phase gaincal results and solints used. 

641 phasecal_results = [] 

642 solints = [] 

643 

644 # Determine which SpWs to solve for: use all input SpWs, filtered for 

645 # phase calibrator intent. 

646 spw_to_solve = ','.join(str(s.id) for s in inputs.ms.get_spectral_windows(task_arg=inputs.spw, intent=intent)) 

647 

648 # Create separate phase solutions for each PHASE field. These solutions 

649 # are intended to be used as a temporary pre-apply when generating the 

650 # final amplitude caltable and the phase offsets caltable. 

651 for field in inputs.ms.get_fields(intent=intent): 

652 # Get optimal phase solution parameters for current PHASE field, 

653 # based on spw mapping info in MS. 

654 combine, gaintype, interp, solint, spwmap = self._get_phasecal_params(intent, field.name) 

655 

656 # PIPE-390: if not combining across spw, then no need to deal with 

657 # SpectralSpec for solint considerations, so create a gaincal 

658 # solution for all SpWs, using provided solint, gaintype, and 

659 # interp. 

660 if not combine: 

661 phasecal_results.append(self._do_calibrator_phasecal(field=field.name, intent=intent, spw=spw_to_solve, 

662 gaintype=gaintype, combine=combine, solint=solint, 

663 minsnr=inputs.calminsnr, interp=interp, 

664 spwmap=spwmap)) 

665 solints.append(solint) 

666 

667 # Otherwise, a combined SpW solution is expected, and we need to 

668 # create separate solutions for each SpectralSpec grouping of SpWs. 

669 else: 

670 # Group the input SpWs by SpectralSpec. 

671 spw_groups = self._get_spw_groupings(inputs.ms, spw_to_solve, spwmap) 

672 if not spw_groups: 

673 raise ValueError('Invalid SpW grouping input.') 

674 

675 # Loop through each grouping of SpWs. 

676 for ref_spw, sspec, spw_sel in spw_groups: 

677 LOG.info(f'Processing spectral spec {sspec} with SpWs {spw_sel}') 

678 phasecal_results.append(self._do_calibrator_phasecal(field=field.name, intent=intent, spw=spw_sel, 

679 gaintype=gaintype, combine=combine, 

680 solint=solint, minsnr=inputs.calminsnr, 

681 interp=interp, spwmap=spwmap)) 

682 solints.append(solint) 

683 

684 # PIPE-1154: determine which was the longest solint used for any of the 

685 # PHASE fields; this will be re-used for the diagnostic amplitude 

686 # caltable. 

687 max_solint = max(solints) 

688 

689 return phasecal_results, max_solint

691 def _do_phasecal_for_phase_calibrators_forcing_combine(self) -> list[GaincalResults]: 

692 """ 

693 This method will create phase gain caltable(s) for each field that 

694 both a.) covers a PHASE calibrator, and b.) for which the SpW mapping 

695 registered in the measurement set recommended to use combine='' in the 

696 gaincal solve. For these fields, an appropriate "combine" SpW map is 

697 generated locally, then a new gaincal solve is performed using this SpW 

698 map and enforcing combine='spw'. 

699 

700 The resulting caltable is used later during this task for computing the 

701 residual phase offsets. 

702 """ 

703 inputs = self.inputs 

704 intent = "PHASE" 

705 

706 # Initialize list of phase gaincal results. 

707 phasecal_results = [] 

708 

709 # Determine which SpWs to solve for: use all input SpWs, filtered for 

710 # phase calibrator intent. 

711 spw_to_solve = ','.join(str(s.id) for s in inputs.ms.get_spectral_windows(task_arg=inputs.spw, intent=intent)) 

712 

713 # Create separate phase solutions for each PHASE field. 

714 for field in inputs.ms.get_fields(intent="PHASE"): 

715 # Get optimal phase solution parameters for current PHASE field, 

716 # based on spw mapping info in MS. 

717 combine, gaintype, interp, solint, spwmap = self._get_phasecal_params(intent, field.name) 

718 

719 # Skip any field where the recommended combine was already 'spw'. 

720 if combine == 'spw': 

721 continue 

722 

723 # For PHASE calibrator fields where the SpW mapping recommended to 

724 # use combine='', continue with creating a new caltable while 

725 # forcing the combination of SpWs, and generate a corresponding 

726 # combine spwmap with which this caltable should be registered. 

727 # This is done to ensure that the subsequent phase offsets 

728 # caltable is always sensitive to residual offsets on the PHASE 

729 # calibrator and the corresponding plots would display 

730 # meaningful information. 

731 

732 # Create a "combine" spwmap for input SpWs. 

733 spws = inputs.ms.get_spectral_windows(inputs.spw) 

734 spwmap = combine_spwmap(spws) 

735 

736 # Run the phase calibration, forcing combination of SpWs with 

737 # appropriate values for interp and spwmap. 

738 phasecal_results.append(self._do_calibrator_phasecal(field=field.name, intent=intent, spw=spw_to_solve, 

739 gaintype=gaintype, combine='spw', solint=solint, 

740 minsnr=inputs.calminsnr, interp='linearPD,linear', 

741 spwmap=spwmap)) 

742 

743 return phasecal_results

746 def _do_calibrator_phasecal(self, field: str = None, intent: str = None, spw: str = None, gaintype: str = 'G', 

747 combine: str = None, solint: str = None, minsnr: float = None, 

748 interp: str = None, spwmap: list[int] = None) -> GaincalResults: 

749 """ 

750 This runs the gaincal for creating phase solutions intended for the 

751 calibrators (amplitude, bandpass, polarization, phase, diffgain(ref/src)). 

752 """ 

753 inputs = self.inputs 

754 

755 # Construct filename of output caltable: 

756 # If provided, use the "calphasetable" parameter from top-level task 

757 # inputs as a basename, but modify to ensure the filename is unique. 

758 caltable = None 

759 if inputs.calphasetable: 

760 root, ext = os.path.splitext(inputs.calphasetable) 

761 # Always add intent, and if the intent is PHASE then also add field 

762 # name. 

763 field_str = f'.{field}' if intent == 'PHASE' else '' 

764 caltable = f'{root}.{intent}{field_str}{ext}' 

765 

766 # Filter provided SpWs to only use science SpWs that were covered by 

767 # current intent(s) and field(s). In principle, it should suffice to 

768 # only provide intent and field to gaincal, but restricting SpWs here 

769 # ensures that the filename of the output caltable only contains the 

770 # SpWs for which solutions are computed. 

771 fieldlist = inputs.ms.get_fields(task_arg=field) 

772 sci_spws = set(inputs.ms.get_spectral_windows(task_arg=spw, intent=intent)) 

773 spws_to_solve = ','.join({str(spw.id) for fld in fieldlist for spw in fld.valid_spws.intersection(sci_spws)}) 

774 

775 task_args = { 

776 'output_dir': inputs.output_dir, 

777 'vis': inputs.vis, 

778 'caltable': caltable, 

779 'field': field, 

780 'intent': intent, 

781 'spw': spws_to_solve, 

782 'solint': solint, 

783 'gaintype': gaintype, 

784 'calmode': 'p', 

785 'minsnr': minsnr, 

786 'combine': combine, 

787 'refant': inputs.refant, 

788 'minblperant': inputs.minblperant, 

789 'solnorm': inputs.solnorm, 

790 } 

791 result = do_gtype_gaincal(inputs.context, self._executor, task_args) 

792 

793 # Modify the cal application for this caltable based on overrides. 

794 # Phase solution caltables should always be registered to be applied 

795 # with calwt=False (PIPE-1154). Register the caltable to be applied 

796 # only to the calibrator intent(s). Adjust the interpolation and SpW 

797 # mapping if provided. 

798 calapp_overrides = { 

799 'calwt': False, 

800 'intent': intent, 

801 } 

802 if interp: 

803 calapp_overrides['interp'] = interp 

804 if spwmap: 

805 calapp_overrides['spwmap'] = spwmap 

806 

807 # PIPE-1154: if adding solutions for a field with PHASE intent, then 

808 # modify the CalApplication to ensure that solutions from this PHASE 

809 # field are only applied to itself. 

810 if intent == 'PHASE': 

811 calapp_overrides['field'] = field 

812 calapp_overrides['gainfield'] = field 

813 

814 # Create a modified CalApplication and replace CalApp in result with 

815 # this new one. 

816 result.pool = [callibrary.copy_calapplication(c, **calapp_overrides) for c in result.pool] 

817 result.final = [callibrary.copy_calapplication(c, **calapp_overrides) for c in result.final] 

818 

819 return result

887 def _do_target_ampcal(self) -> list[callibrary.CalApplication]: 

888 """ 

889 This method computes the amplitude caltable intended for TARGET, 

890 CHECK, and all calibrators. It returns a list of two CalApplications, 

891 one for how to apply the caltable to all the calibrators, and a second 

892 one for how to apply the caltable to the TARGET and CHECK source(s). 

893 """ 

894 inputs = self.inputs 

895 

896 task_args = { 

897 'output_dir': inputs.output_dir, 

898 'vis': inputs.vis, 

899 'caltable': inputs.amptable, 

900 'field': inputs.field, 

901 'intent': inputs.intent, 

902 'spw': inputs.spw, 

903 'solint': 'inf', 

904 'gaintype': 'T', 

905 'calmode': 'a', 

906 'minsnr': inputs.targetminsnr, 

907 'combine': inputs.combine, 

908 'refant': inputs.refant, 

909 'minblperant': inputs.minblperant, 

910 'solnorm': inputs.solnorm 

911 } 

912 result = do_gtype_gaincal(inputs.context, self._executor, task_args) 

913 

914 # Create modified CalApplications for registering this amplitude 

915 # caltable in the callibrary/context, once for TARGET,CHECK, and once 

916 # for the calibrators. 

917 # 

918 # Spec for gainfield application from CAS-12214: 

919 # 

920 # 1) When applying phase and/or amp gain tables to calibrators, 

921 # continue to use gainfield='nearest'. This will ensure that time 

922 # interpolations from one group's complex gain calibrator's 

923 # solutions to the next group's complex gain calibrator solutions 

924 # will never influence how either calibrator gets calibrated. 

925 # 

926 # 2) When applying phase and/or amp gain tables to science targets, 

927 # leave gainfield blank (instead of 'nearest'), so that time 

928 # interpolation across all gain calibrators will be used to 

929 # calibrate science targets. This will automagically take care of 

930 # the multiple science groups case. Specifically, since it is 

931 # assumed that all science target groups are bracketed on both 

932 # sides by "their" complex gain calibrator, there will never be a 

933 # case where the interpolated solution between two complex gain 

934 # calibrators needs to be used. 

935 result_calapp = result.final[0] 

936 

937 # Create CalApplication for the calibrators. 

938 cal_calapp = callibrary.copy_calapplication(result_calapp, intent=inputs.intent, gainfield='nearest', 

939 interp='nearest,linear') 

940 

941 # Create CalApplication for the TARGET/CHECK sources, where present in 

942 # the MS (PIPE-2268). 

943 calapp_overrides = {'intent': utils.filter_intents_for_ms(inputs.ms, 'CHECK,TARGET'), 'gainfield': ''} 

944 fields_targets = inputs.ms.get_fields(intent=calapp_overrides['intent']) 

945 fields_not_targets = inputs.ms.get_fields( 

946 intent='BANDPASS,PHASE,AMPLITUDE,POLARIZATION,DIFFGAINREF,DIFFGAINSRC' 

947 ) 

948 fields_checktarget_only = set(fields_targets) - set(fields_not_targets) 

949 if not calapp_overrides['intent'] or not fields_checktarget_only: 

950 LOG.debug( 

951 'No CHECK-only or TARGET-only intent scans found in %s and we will skip the creation of CalApp for intents="CHECK,TARGET".', 

952 inputs.ms.name, 

953 ) 

954 return [cal_calapp] 

955 

956 # PIPE-2087: for BandToBand, register the solutions to be applied to the 

957 # diffgain on-source SpWs, and use the amplitude solutions from the 

958 # BANDPASS intent. 

959 if inputs.ms.is_band_to_band: 

960 dg_srcspws = inputs.ms.get_spectral_windows(task_arg=inputs.spw, intent='DIFFGAINSRC') 

961 calapp_overrides['spw'] = ','.join(str(s.id) for s in dg_srcspws) 

962 calapp_overrides['gainfield'] = ','.join(f.name for f in inputs.ms.get_fields(intent='BANDPASS')) 

963 

964 target_calapp = callibrary.copy_calapplication(result_calapp, **calapp_overrides) 

965 

966 return [cal_calapp, target_calapp]

968 def _get_phasecal_params(self, intent: str, field: str) -> tuple[str, str, str | None, str, list[int]]: 

969 inputs = self.inputs 

970 

971 # By default, no spw mapping or combining, no interp, gaintype='G', 

972 # and use solint set by "calsolint" input parameter. 

973 combine = '' 

974 gaintype = 'G' 

975 interp = None 

976 solint = inputs.calsolint 

977 spwmap = [] 

978 

979 # Try to fetch spwmapping info from MS for requested intent and field. 

980 spwmapping = inputs.ms.spwmaps.get((intent, field), None) 

981 

982 # If a mapping was found, use the spwmap, and update further parameters 

983 # depending on whether it is a combine spw mapping. 

984 if spwmapping: 

985 spwmap = spwmapping.spwmap 

986 solint = spwmapping.solint 

987 gaintype = spwmapping.gaintype 

988 # If the spwmap is for combining spws, then override combine and 

989 # interp accordingly. 

990 if spwmapping.combine: 

991 combine = 'spw' 

992 interp = 'linearPD,linear' 

993 else: 

994 # PIPE-1154: when using a phase up spw mapping, ensure that 

995 # interp = 'linear,linear'; though this may need to be changed 

996 # in the future, see PIPEREQ-85. 

997 interp = 'linear,linear' 

998 

999 return combine, gaintype, interp, solint, spwmap
