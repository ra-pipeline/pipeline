254 def prepare(self): 

255 inputs = self.inputs 

256 

257 # Identify number and fraction of antennas with WVR. 

258 nr_ants_wvr, frac_ants_wvr = self._identify_ants_with_wvr() 

259 

260 # Evaluate whether there are enough antennas with WVR by number and 

261 # fraction to be worth proceeding with calibration and flagging. 

262 if (nr_ants_wvr < inputs.ants_with_wvr_nr_thresh or 

263 frac_ants_wvr < self.inputs.ants_with_wvr_thresh): 

264 # If there are too few antennas, return an empty result, with 

265 # flag set to indicate that there were too few WVR antennas. 

266 flaggerresult = viewflaggers.MatrixFlaggerResults(vis=inputs.vis) 

267 result = WvrgcalflagResults(vis=inputs.vis, 

268 flaggerresult=flaggerresult, 

269 too_few_wvr=True) 

270 # Log info message. 

271 if nr_ants_wvr < inputs.ants_with_wvr_nr_thresh: 

272 LOG.info("Number of antennas with WVR ({}) below the" 

273 " threshold ({}), skipping WVR calibration." 

274 "".format(nr_ants_wvr, inputs.ants_with_wvr_nr_thresh)) 

275 elif frac_ants_wvr < self.inputs.ants_with_wvr_thresh: 

276 LOG.info("Fraction of antennas with WVR ({:.2f}) below the" 

277 " threshold of {}, skipping WVR calibration." 

278 "".format(frac_ants_wvr, self.inputs.ants_with_wvr_thresh)) 

279 else: 

280 # If there are enough WVR antennas initially, run calibration 

281 # and flagging. 

282 flaggerresult = self._run_flagger() 

283 

284 # Attach flagging result to final result. 

285 result = WvrgcalflagResults(vis=inputs.vis, 

286 flaggerresult=flaggerresult) 

287 

288 return result

290 def analyse(self, result): 

291 inputs = self.inputs 

292 

293 # If flagging views are available, then evaluate the flagger result to 

294 # check whether the WVR correction should still be applied. 

295 if result.flaggerresult.viewresult and result.flaggerresult.viewresult.descriptions(): 

296 # Identify number and fraction of unflagged antennas with WVR. 

297 nr_ants_wvr, frac_ants_wvr = self._identify_unflagged_ants_with_wvr(result) 

298 

299 # If too few unflagged antennas with WVR remain (by number or 

300 # fraction), then remove the calfile from the result so that it 

301 # cannot be accepted into the context, and set flag in result 

302 # to indicate too few WVR antennas remained. 

303 if (nr_ants_wvr < inputs.ants_with_wvr_nr_thresh or 

304 frac_ants_wvr < inputs.ants_with_wvr_thresh): 

305 result.flaggerresult.dataresult.final = [] 

306 result.too_few_wvr_post_flagging = True 

307 

308 # Log warning. 

309 if nr_ants_wvr < inputs.ants_with_wvr_nr_thresh: 

310 LOG.warning( 

311 "After flagging, the number of unflagged antennas" 

312 " with WVR ({}) is below the threshold of {}, skipping" 

313 " WVR calibration.".format(nr_ants_wvr, inputs.ants_with_wvr_nr_thresh)) 

314 elif frac_ants_wvr < inputs.ants_with_wvr_thresh: 

315 LOG.warning( 

316 "After flagging, the fraction of unflagged antennas" 

317 " with WVR ({:.2f}) is below the threshold of {};" 

318 " wvrgcal file will not be applied." 

319 "".format(frac_ants_wvr, inputs.ants_with_wvr_thresh)) 

320 

321 # If the associated qa score indicates that applying the WVR 

322 # calibration will make things worse then remove the calfile from 

323 # the result so that it cannot be accepted into the context. 

324 elif result.flaggerresult.dataresult.qa_wvr.overall_score is not None \ 

325 and result.flaggerresult.dataresult.qa_wvr.overall_score < inputs.accept_threshold: 

326 result.flaggerresult.dataresult.final = [] 

327 

328 return result

330 def _identify_ants_with_wvr(self): 

331 """ 

332 Identify antennas with WVR, return as number and as fraction of all 

333 antennas. 

334 """ 

335 # Get the MS domain object. 

336 ms = self.inputs.context.observing_run.get_ms(name=self.inputs.vis) 

337 

338 # Identify antennas with WVR as those whose name does not start with 'CM'. 

339 ants_with_wvr = [antenna.name 

340 for antenna in ms.antennas 

341 if not antenna.name.startswith('CM')] 

342 nr_ants_wvr = len(ants_with_wvr) 

343 frac_ants_wvr = nr_ants_wvr / float(len(ms.antennas)) 

344 

345 return nr_ants_wvr, frac_ants_wvr

347 def _identify_unflagged_ants_with_wvr(self, result): 

348 """ 

349 Evaluate flagger result to identify remaining unflagged antennas with 

350 WVR, return as a number and as a fraction of all antennas. 

351 

352 :param result: WvrgcalflagResults object 

353 :return: (int, float): number of antennas, fraction of antennas. 

354 """ 

355 # Get the MS domain object. 

356 ms = self.inputs.context.observing_run.get_ms(name=self.inputs.vis) 

357 

358 # Only expect 1 flagging view for Wvrgcalflag. 

359 view = result.flaggerresult.last(result.flaggerresult.descriptions()[0]) 

360 

361 # Identify antennas fully flagged for all timestamps, mapping the 

362 # array indices to the original antenna IDs using the flagging view 

363 # x-axis data. 

364 antids_fully_flagged = view.axes[0].data[ 

365 np.where(np.all(view.flag, axis=1))[0]] 

366 

367 # Identify antennas with WVR that also remain unflagged. 

368 ants_with_wvr_and_unflagged = [ 

369 antenna.name 

370 for antenna in ms.antennas 

371 if not antenna.name.startswith('CM') 

372 and antenna.id not in antids_fully_flagged 

373 ] 

374 

375 nr_ants_wvr = len(ants_with_wvr_and_unflagged) 

376 frac_ants_wvr = nr_ants_wvr / float(len(ms.antennas)) 

377 

378 return nr_ants_wvr, frac_ants_wvr

380 def _run_flagger(self): 

381 inputs = self.inputs 

382 

383 # Construct the task that will prepare/read the data necessary to 

384 # create a flagging view. 

385 datainputs = WvrgcalflagData.Inputs( 

386 context=inputs.context, output_dir=inputs.output_dir, 

387 vis=inputs.vis, caltable=inputs.caltable, 

388 offsetstable=inputs.offsetstable, 

389 hm_toffset=inputs.hm_toffset, toffset=inputs.toffset, 

390 segsource=inputs.segsource, hm_tie=inputs.hm_tie, 

391 tie=inputs.tie, sourceflag=inputs.sourceflag, nsol=inputs.nsol, 

392 disperse=inputs.disperse, wvrflag=inputs.wvrflag, 

393 hm_smooth=inputs.hm_smooth, smooth=inputs.smooth, 

394 scale=inputs.scale, maxdistm=inputs.maxdistm, 

395 minnumants=inputs.minnumants, mingoodfrac=inputs.mingoodfrac, 

396 refant=inputs.refant, qa_intent=inputs.qa_intent, 

397 qa_bandpass_intent=inputs.qa_bandpass_intent) 

398 datatask = WvrgcalflagData(datainputs) 

399 

400 # Construct the generator that will create the views of the data 

401 # that is the basis for flagging. 

402 viewtask = WvrgcalflagView(context=inputs.context, 

403 flag_intent=inputs.flag_intent) 

404 

405 # Construct the "flagsetter" task: Wvrgcalflag has its own dedicated 

406 # flagsetter that does not really set flags in underlying data, but 

407 # rather updates the "pre-existing flags" parameter (wvrflag) of the 

408 # datatask, such that a subsequent run of the datatask takes the newly 

409 # found flags into account. 

410 flagsetterinputs = wvrgcalflagsetter.WvrgcalFlagSetterInputs( 

411 context=inputs.context, table=inputs.vis, vis=inputs.vis, 

412 datatask=datatask) 

413 flagsettertask = wvrgcalflagsetter.WvrgcalFlagSetter(flagsetterinputs) 

414 

415 # Define which type of flagger to use. 

416 flagger = viewflaggers.MatrixFlagger 

417 

418 # Translate the input flagging parameters to a more compact 

419 # list of rules. 

420 rules = flagger.make_flag_rules( 

421 flag_hi=inputs.flag_hi, 

422 fhi_limit=inputs.fhi_limit, 

423 fhi_minsample=inputs.fhi_minsample) 

424 

425 # Construct the flagger task around the data, view, and flagsetter 

426 # tasks. Wvrgcalflag passes its own Results class to be used by the 

427 # flagger, and the flagger produces the final result. 

428 flaggerinputs = flagger.Inputs( 

429 context=inputs.context, output_dir=inputs.output_dir, 

430 vis=inputs.vis, datatask=datatask, viewtask=viewtask, 

431 flagsettertask=flagsettertask, rules=rules, niter=1, 

432 iter_datatask=True) 

433 flaggertask = flagger(flaggerinputs) 

434 

435 # Execute the flagger task, and wrap this child task in a 

436 # SuspendCapturingLogger so that warnings emitted by the child task 

437 # do not make it to the web log page; see CAS-7795. 

438 with logging.SuspendCapturingLogger(): 

439 flaggerresult = self._executor.execute(flaggertask) 

440 

441 return flaggerresult

579class WvrgcalflagView: 

580 def __init__(self, context, flag_intent=None): 

581 """ 

582 Creates an WvrgcalflagView instance. 

583 """ 

584 self.context = context 

585 self.flag_intent = flag_intent 

586 

587 def __call__(self, dataresult): 

588 """ 

589 When called, the WvrgcalflagView object calculates flagging views 

590 based on the provided dataresult (from Wvrgcal). For the views matching 

591 the intent specified by "inputs.flag_intent", it retrieves the view, 

592 updates the flags in this view to reflect the flagging found by 

593 Wvrgcal, and then stores the resulting view in WvrgcalflagViewResults. 

594 

595 dataresult -- WvrgcalResult object. 

596 

597 Returns: 

598 WvrgcalflagViewResults object containing the flagging view. 

599 """ 

600 

601 # Initialize result structure. 

602 result = WvrgcalflagViewResults(vis=dataresult.vis) 

603 

604 # As part of the data task, Wvrgcal will have created flagging views, 

605 # Wvrgcalflag will start from a copy of the views stored in dataresult. 

606 LOG.info('Retrieving flagging views for vis {0}'.format( 

607 dataresult.vis)) 

608 

609 # Get a list of available intents from the MS. 

610 ms = self.context.observing_run.get_ms(name=dataresult.vis) 

611 ms_intent_list = ms.intents 

612 

613 # Test if the flagging intents are present in the MS. 

614 intent_available = [] 

615 flag_intent_list = self.flag_intent.split(',') 

616 for flag_intent in flag_intent_list: 

617 intent_available = [intent for intent in ms_intent_list if flag_intent in intent] 

618 if intent_available: 

619 LOG.info('flagging views will use %s data' % flag_intent) 

620 else: 

621 LOG.warning('no data for intent %s' % flag_intent) 

622 if not intent_available: 

623 LOG.warning('no data fits flag_intent {0}, no flagging will be' 

624 ' done'.format(self.flag_intent)) 

625 

626 # From the QA section of the dataresult (from Wvrgcal) copy those views 

627 # for which the intent matches the flag_intent. 

628 for description in dataresult.qa_wvr.descriptions(): 

629 for flag_intent in flag_intent_list: 

630 if 'Intent:%s' % flag_intent in description: 

631 # Translate list of WVR flagged antenna names back to list 

632 # of antenna IDs. 

633 ant_names, ant_ids = commonhelpermethods.get_antenna_names(ms) 

634 wvrflagids = [] 

635 for ant_name in dataresult.wvrflag: 

636 ant_id = [idx for idx in ant_names if 

637 ant_names[idx] == ant_name] 

638 wvrflagids += ant_id 

639 

640 # Set antennas specified by the wvrflag parameter in the 

641 # flagging image to show that these data are already 'flagged' 

642 # (i.e. interpolated). 

643 image = dataresult.qa_wvr.last(description) 

644 image.setflags(axisname='Antenna', indices=wvrflagids) 

645 

646 # Add updated view to result. 

647 result.addview(description, image) 

648 

649 return result
