93def robuststats(A): 

94 '''Return median and estimate standard deviation 

95 of numpy array A using median statistics. 

96 ''' 

97 #correction factor for obtaining sigma from MAD 

98 madfactor = 1.482602218505602 

99 n = len(A) 

100 #Fitting parameters for sample size correction factor b(n) 

101 if (n%2 == 0): 

102 alpha = 1.32 

103 beta = -1.5 

104 else: 

105 alpha = 1.32 

106 beta = -0.9 

107 bn = 1.0 - 1.0/(alpha*n + beta) 

108 mu = np.ma.median(A) 

109 sigma = (1.0/bn)*madfactor*np.ma.median(np.ma.abs(A - mu)) 

110 return (mu, sigma)

113def segmentEdges(seq, gap, label, sortdata = True): 

114 '''Return edges of sequence of values with gaps 

115 Assumes 1D array "seq" is ordered. Otherwise one should sort it first. 

116 seq: Sequence of values to search gaps in. 

117 gap: Gap size to consider. 

118 label: Label to put in the numpy array returned. 

119 sortdata: Whether to sort data in the seq vector. 

120 ''' 

121 

122 if sortdata: 

123 seq = np.sort(seq) 

124 n = len(seq) 

125 diff = seq[1:] - seq[0:n-1] 

126 if np.any(diff < 0): 

127 LOG.info('Array is not sorted, use sortdata = True') 

128 return None 

129 isgap = (diff > gap) 

130 if np.sum(isgap) > 0: 

131 startseg = seq[np.append([True],isgap)] 

132 endseg = seq[np.append(isgap,[True])] 

133 output = np.array([(startseg[i], endseg[i], label) for i in range(len(startseg))], np.dtype([('tstart',np.float64),('tend',np.float64),('intent',np.str_,40)])) 

134 else: 

135 output = np.array([(seq[0], seq[-1], label)], np.dtype([('tstart',np.float64),('tend',np.float64),('intent',np.str_,40)])) 

136 

137 return output

151def sigclipfit(x, A, Asig, fitdeg, nclips, nsigma, bordertokeep = 0.05, progdegree = False, smoothselbox = 0.05): 

152 '''Return polynomial model parameters done to vector A and residuals from the fit. 

153 x: x coordinate 

154 A: Data set vector. 

155 Asig: Data sigma vector. Should include variance due to skylines and science target emission. 

156 fitdeg: Fitting polynomial degree. 

157 nclips: Number of sigma clipping iterations. 

158 nsigma: Number of sigmas in sigma clipping iterations. Positive float or two-value tuple. 

159 bordertokeep: Fraction of channels to keep in the borders 

160 progdegree: Whether to increase polynomial fit progressively in each iteration. 

161 smoothselbox: Enlarge each outlier detection with a box of this size (as fraction of SPW). 

162 ''' 

163 if type(nsigma) == float: 

164 nsigmahigh = nsigma 

165 nsigmalow = nsigma 

166 elif ((type(nsigma) == tuple) or (type(nsigma) == list)) and (len(nsigma) == 2): 

167 nsigmahigh = nsigma[1] 

168 nsigmalow = nsigma[0] 

169 output = {} 

170 gooddata = np.ma.MaskedArray(np.ones(len(A)), mask=A.mask) 

171 #Start data selection with all good data 

172 sel = gooddata 

173 minx = np.ma.min(x) 

174 maxx = np.ma.max(x) 

175 bordermin = minx + bordertokeep*(maxx - minx) 

176 bordermax = maxx - bordertokeep*(maxx - minx) 

177 #Set border selection, avoiding any bad data 

178 border = np.ma.MaskedArray(np.ma.all([np.ma.any([x < bordermin, x > bordermax], axis=0), gooddata], axis=0), mask=A.mask) 

179 if progdegree: 

180 degree = 0 

181 else: 

182 degree = fitdeg 

183 if smoothselbox > 0.0: 

184 box = max(2, int(smoothselbox*len(A))) 

185 else: 

186 box = 1 

187 while (degree <= fitdeg): 

188 for i in range(nclips): 

189 #Include constant border, if demanded 

190 thissel = np.ma.any([sel, border], axis=0) 

191 #If everything is flagged, continue with a default selection of everything 

192 if np.sum(~A.mask[thissel]) <= fitdeg: 

193 thissel = np.ma.any([gooddata, border], axis=0) 

194 try: 

195 coefs = np.polyfit(x[thissel], A[thissel], fitdeg, w=1/Asig[thissel]) 

196 except: 

197 coefs = [0.0 for i in range(degree+1)] 

198 LOG.info('No data!!!\nx='+str(x[thissel])+'\nA='+str(A[thissel])+'\nAsig='+str(Asig[thissel])) 

199 model = np.ma.sum([coefs[i]*(x**(degree-i)) for i in range(degree+1)], axis=0) 

200 diff = A - model 

201 (mu, sigma) = robuststats(diff) 

202 #Select pixels to use in the next iteration from pixels within nsigma of mean 

203 #including only non-masked data, but not in the border. 

204 sel = np.ma.all([((diff - mu)/sigma >= -nsigmalow), ((diff - mu)/sigma <= nsigmahigh), gooddata], axis=0) 

205 #If boxcar smoothing of selection is chosen (smoothselbox > 0), broadening non-selected pixels 

206 if box > 1: 

207 sel = np.ma.logical_not(enlargesel(np.ma.logical_not(sel), box)) 

208 degree += 1 

209 output['coefs'] = coefs 

210 output['residuals'] = diff 

211 output['sel'] = sel 

212 output['border'] = border 

213 return output

217 '''Enlarge selection by "box" pixels around each selected pixel

228def smooth(y: np.ndarray, box_pts: int): 

229 '''Smooth using boxcar convolution. 

230 

231 To mitigate boundary effect, this function extrapolates 

232 the data by box_pts // 2 in both edges with "nearest" 

233 method. This will be harmless in the context of this 

234 heuristics since we are interested in the gradient 

235 (diff) between adjuscent channels. 

236 

237 Args: 

238 y: 1D array to be smoothed. 

239 box_pts: Width of the boxcar kernel. Must be a positive integer. 

240 

241 Returns: 

242 Smoothed 1D array of the same length as y. 

243 ''' 

244 if box_pts < 2: 

245 # no smoothing is applied 

246 return y 

247 

248 box = np.ma.ones(box_pts)/box_pts 

249 extra_chans = (box_pts + 1) // 2 

250 y_expand = np.pad(y, extra_chans, mode='edge') 

251 print(f"box_pts {box_pts} extra_chans {extra_chans}") 

252 y_smooth = np.ma.convolve(y_expand, box, mode='same') 

253 return y_smooth[extra_chans:-extra_chans]

256def getskylines(tauspec, spw, spwsetup, fraclevel = 0.5, minpeaklevel = 0.0, spwbordertoavoid = 0.025, taudeg = 2): 

257 '''Get position of sky lines and line widths from the optical depth. 

258 fraclevel: fraction of the maximum of the optical depth to look for. 

259 If fraclevel=0.5 (default), the width results will be the HWHM. 

260 minpeaklevel: Minimum relative opacity at a given peak (relative to median opacity) 

261 to consider a line. Lines smaller than this level are ignored. (default 0.05) 

262 spwbordertoavoid: Fractional border of the SPW to avoid for peak detection. 

263 Default is 0.025 (2.5% of SPW BW) 

264 ''' 

265 nch = len(tauspec) 

266 borderpix = int(max(1.0,np.round(nch*spwbordertoavoid))) 

267 noborder = np.ones(nch, dtype=bool) 

268 noborder[0:borderpix] = False 

269 noborder[nch-borderpix:nch] = False 

270 mintau = np.min(tauspec) 

271 mediantau = np.median(tauspec) 

272 tauzeroD = np.array(np.ediff1d(1.0*(np.ediff1d(tauspec) > 0.0), to_begin=0, to_end=0), dtype=bool) 

273 taunegDD = np.array(np.ediff1d(np.ediff1d(tauspec), to_begin=0, to_end=0) < 0.0, dtype=bool) 

274 taugtminlevel = (tauspec/mediantau >= (1.0 + minpeaklevel)) 

275 taupeak = np.all([tauzeroD, taunegDD, taugtminlevel, noborder], axis=0) 

276 tauvalley = np.all([tauzeroD, ~taunegDD], axis=0) 

277 idxpeak = np.arange(nch, dtype=int)[taupeak] 

278 idxvalley = np.arange(nch, dtype=int)[tauvalley] 

279 npeak = len(idxpeak) 

280 peaktau = tauspec[idxpeak] 

281 output = {'nch': nch, 'npeak': npeak, 'taumin': mintau, 'fraclevel': fraclevel, 'valley': idxvalley, 'peaksinfo': {}} 

282 hwhmleft = np.zeros(npeak) 

283 hwhmright = np.zeros(npeak) 

284 widthratio = np.zeros(npeak) 

285 #Fit "baseline" to tau 

286 nu = spwsetup[spw]['chanfreqs']/1.e09 

287 taufit = sigclipfit(nu, np.ma.MaskedArray(tauspec, mask=np.zeros(nch)), 

288 np.ma.MaskedArray(np.sqrt(tauspec), mask=np.zeros(nch)), 2, 3, 5.0) 

289 taubline = np.ma.sum([taufit['coefs'][deg]*(nu**(taudeg-deg)) for deg in range(taudeg + 1)], axis = 0) 

290 subtau = tauspec - taubline 

291 output['taufit'] = taufit 

292 for k, idx in enumerate(idxpeak): 

293 #calculate the decrease from the tau depth relative to the minimum tau in the SPW 

294 peakratioleft = np.array([subtau[idx - i]/subtau[idx] for i in range(idx)]) 

295 peakratioright = np.array([subtau[idx + i]/subtau[idx] for i in range(nch-idx)]) 

296 posrightneg = peakratioright < fraclevel 

297 if np.sum(posrightneg) > 0: 

298 hwhmright[k] = np.where(posrightneg)[0][0] 

299 else: 

300 #search of peak to the right of this peak, select the border if none 

301 if k < npeak-1: 

302 hwhmright[k] = 0.5*(idxpeak[k+1] - idx) 

303 else: 

304 hwhmright[k] = nch - idx - 1 

305 posleftneg = peakratioleft < fraclevel 

306 if np.sum(posleftneg) > 0: 

307 hwhmleft[k] = np.where(posleftneg)[0][0] 

308 else: 

309 #search of peak to the left of this peak, select the border if none 

310 if k > 0: 

311 hwhmleft[k] = 0.5*(idx - idxpeak[k-1]) 

312 else: 

313 hwhmleft[k] = idx 

314 

315 #If the range of the skylines goes beyond a valley, shrink range around line using those valleys 

316 for k in range(npeak): 

317 valleypointsleft = np.sort([point for point in idxvalley if (point < idxpeak[k])]) 

318 if (len(valleypointsleft) > 0) and (idxpeak[k]-hwhmleft[k] < valleypointsleft[-1]): 

319 hwhmleft[k] = idxpeak[k] - valleypointsleft[-1] 

320 valleypointsright = np.sort([point for point in idxvalley if (point > idxpeak[k])]) 

321 if (len(valleypointsright) > 0) and (idxpeak[k]+hwhmright[k] > valleypointsright[0]): 

322 hwhmright[k] = valleypointsright[0] - idxpeak[k] 

323 

324 #Check if there is more than one peak and if so check if ranges overlap and correct in that case 

325 if npeak > 1: 

326 for k in range(npeak - 1): 

327 if (idxpeak[k]+hwhmright[k] > idxpeak[k+1]-hwhmleft[k+1]): 

328 #If there is an valley point detected between the peaks, choose that one. 

329 #If not, just pick the middle point between peaks 

330 valleypoints = [point for point in idxvalley if (point > idxpeak[k]) and (point < idxpeak[k+1])] 

331 if len(valleypoints) > 0: 

332 hwhmright[k] = valleypoints[0]-idxpeak[k] 

333 hwhmleft[k+1] = idxpeak[k+1]-valleypoints[0] 

334 else: 

335 hwhmright[k] = 0.5*(idxpeak[k+1]-idxpeak[k]) 

336 hwhmleft[k+1] = 0.5*(idxpeak[k+1]-idxpeak[k]) 

337 

338 #Create output dictionary 

339 for k, idx in enumerate(idxpeak): 

340 output['peaksinfo'][k] = {} 

341 output['peaksinfo'][k]['peakpos'] = int(idx) 

342 output['peaksinfo'][k]['taupeak'] = tauspec[idx] 

343 output['peaksinfo'][k]['hwhmleft'] = int(hwhmleft[k]) 

344 output['peaksinfo'][k]['hwhmright'] = int(hwhmright[k]) 

345 output['peaksinfo'][k]['minrange'] = int(idx - hwhmleft[k]) 

346 output['peaksinfo'][k]['maxrange'] = int(idx + hwhmright[k]) 

347 return output

349def gradeskylines(skylines: dict, cntrweight: float = 1.0): 

350 '''Function to calculate a "grade" for each skyline detected by the getskylines() function. 

351 These "grades" are based on a normalized value of the opacity at the skyline peak and the position of 

352 the line in the SPW, and is aimed at providing a decision on which skyline to used for model evaluation. 

353 param: 

354 skylines: Dictionary of detected skylines indexed by SPW, where each sub-dictionary is as obtained 

355 from getskylines(). 

356 cntrweight: Relative weight given to the position of the skyline relative to the center of the SPW. 

357 Must be a positive float number between 0 and 1, a value of 0.0 means no weight to position, 

358 a value >0 gives the line a linearly decreasing grade the further it is away from the center of the SPW. 

359 returns: 

360 A tuple of (spw, peak) that identifies the SPW and id number of the skyline in that SPW that has the best grade. 

361 The skyline id number is the same as in the dictionary output from getskylines(). 

362 Additionally, the grades calculated are saved in the input skylines dictionary, modifying the original variable. 

363 ''' 

364 

365 spwlist = np.sort(list(skylines.keys())) 

366 idxspw = [] 

367 idxpeak = [] 

368 taupeak = [] 

369 centerdist = [] 

370 #Read values and calculate distance to SPW centers 

371 for spw in spwlist: 

372 for i in range(skylines[spw]['npeak']): 

373 taupeak.append(skylines[spw]['peaksinfo'][i]['taupeak']) 

374 centerdist.append(2.0*(skylines[spw]['peaksinfo'][i]['peakpos'] - 0.5*skylines[spw]['nch'])/skylines[spw]['nch']) 

375 idxspw.append(spw) 

376 idxpeak.append(i) 

377 taupeak = np.array(taupeak) 

378 centerdist = np.array(centerdist) 

379 #Calculate grade 

380 normtau = (taupeak - np.min(taupeak))/(np.max(taupeak) - np.min(taupeak)) 

381 grade = normtau*(1.0 - cntrweight*np.abs(centerdist)) 

382 bestgrade = np.argsort(grade)[-1] 

383 #Put grade back into skylines dictionary 

384 idx = 0 

385 for spw in spwlist: 

386 for i in range(skylines[spw]['npeak']): 

387 skylines[spw]['peaksinfo'][i]['grade'] = grade[idx] 

388 idx += 1 

389 #Return SPW and index of higher grade peak 

390 return (idxspw[bestgrade], idxpeak[bestgrade])

408def calcmetric(rawsample: np.ndarray, rawsigmasample: np.ndarray, metrictype_list: str | list[str] = 'intsqdiff', smoothbox: int = 1) -> dict[str, tuple[float, float]]: 

409 '''Function that calculates a metric value for a given piece of spectrum (time-averaged data), typically 

410 a section around the skyline selected by its "grade". 

411 param: 

412 rawsample: (numpy array) Array with the data to be used to compute the metric. 

413 rawsigmasample: (numpy array) Noise array related to the data to be used to compute the metric. Noise 

414 estimation typically used is the standard deviation along the time dimension. 

415 smoothbox: (int) Run a boxcar smoothing on the data piece before calculating the metric. Value in number of channels. 

416 metrictype: (str) String that determines which algorithm to use to calculate the metric value. Options are: 

417 - "intabs": Integral of the absolute value of spectrum piece. (deprecated) 

418 - "maxabs": Maximum of the absolute value of spectrum piece. (deprecated) 

419 - "maxabsdiff": Maximum of the absolute value of the derivative of the spectrum piece. 

420 - "intabsdiff": Integral of the absolute value of the derivative of the spectrum piece. 

421 - "intsqdiff": Integral of the square value of the derivative of the spectrum piece. 

422 ''' 

423 

424 (npols, nch) = np.shape(rawsample) 

425 # limit smoothing box size to 20% of nch 

426 smoothbox = max(1, min(nch // 5, smoothbox)) 

427 #Smooth if requested 

428 if (smoothbox > 1) and (type(smoothbox) == int): 

429 sample = 0.0*rawsample 

430 for i in range(npols): 

431 sample[i,:] = smooth(rawsample[i,:], smoothbox) 

432 sigmasample = rawsigmasample/np.sqrt(smoothbox) 

433 else: 

434 sample = rawsample 

435 sigmasample = rawsigmasample 

436 #Transform arrays to ma if not like that already 

437 if not type(sample) == np.ma.core.MaskedArray: 

438 sample = np.ma.MaskedArray(sample) 

439 sigmasample = np.ma.MaskedArray(sigmasample) 

440 

441 resultdict = {} 

442 

443 if isinstance(metrictype_list, str): 

444 metrictype_list = [metrictype_list] 

445 

446 for metrictype in metrictype_list: 

447 if metrictype == 'intabs': 

448 auxval = [] 

449 auxerr = [] 

450 for i in range(npols): 

451 goodsample = (sample.data[i])[~sample.mask[i]] 

452 sigmagoodsample = (sigmasample[i])[~sample.mask[i]] 

453 if len(goodsample) > 0: 

454 auxval.append(np.ma.sum(np.ma.abs(sample[i,:]))) 

455 auxerr.append(np.ma.sqrt(np.ma.sum(sigmasample[i,:]*sigmasample[i,:]))) 

456 if len(auxval) > 0: 

457 value = np.sum(auxval) 

458 error = np.sqrt(np.sum(np.array(auxerr)*np.array(auxerr))) 

459 else: 

460 value = np.nan 

461 error = np.nan 

462 resultdict[metrictype] = (value, error) 

463 elif metrictype == 'maxabs': 

464 auxval = [] 

465 auxerr = [] 

466 for i in range(npols): 

467 goodsample = (sample.data[i])[~sample.mask[i]] 

468 sigmagoodsample = (sigmasample[i])[~sample.mask[i]] 

469 if len(goodsample) > 0: 

470 idx = np.argsort(np.abs(goodsample))[-1] 

471 auxval.append(np.abs(goodsample[idx])) 

472 auxerr.append(sigmagoodsample[idx]) 

473 if len(auxval) > 0: 

474 idx = np.argsort(np.abs(auxval))[-1] 

475 value = auxval[idx] 

476 error = auxerr[idx] 

477 else: 

478 value = np.nan 

479 error = np.nan 

480 resultdict[metrictype] = (value, error) 

481 elif metrictype == 'maxabsdiff': 

482 auxval = [] 

483 auxerr = [] 

484 for i in range(npols): 

485 goodsample = (sample.data[i])[~sample.mask[i]] 

486 sigmagoodsample = (sigmasample[i])[~sample.mask[i]] 

487 if len(goodsample) > 0: 

488 absdiff = np.abs(np.diff(goodsample)) 

489 idx = np.argsort(absdiff)[-1] 

490 auxval.append(absdiff[idx]) 

491 auxerr.append(np.sqrt(sigmagoodsample[idx]**2+sigmagoodsample[idx+1]**2)) 

492 if len(auxval) > 0: 

493 idx = np.argsort(np.abs(auxval))[-1] 

494 value = auxval[idx] 

495 error = auxerr[idx] 

496 else: 

497 value = np.nan 

498 error = np.nan 

499 resultdict[metrictype] = (value, error) 

500 elif metrictype == 'intabsdiff': 

501 auxval = [] 

502 auxerr = [] 

503 for i in range(npols): 

504 goodsample = (sample.data[i])[~sample.mask[i]] 

505 sigmasqgoodsample = np.square((sigmasample[i])[~sample.mask[i]]) 

506 if len(goodsample) > 0: 

507 absdiff = np.abs(np.diff(goodsample)) 

508 absdiffsqerr = sigmasqgoodsample + np.roll(sigmasqgoodsample, 1) 

509 auxval.append(np.sum(absdiff)) 

510 auxerr.append(np.sqrt(np.sum(absdiffsqerr))) 

511 if len(auxval) > 0: 

512 value = np.sum(auxval) 

513 error = np.sqrt(np.sum(np.array(auxerr)*np.array(auxerr))) 

514 else: 

515 value = np.nan 

516 error = np.nan 

517 resultdict[metrictype] = (value, error) 

518 elif metrictype == 'intsqdiff': 

519 auxval = [] 

520 auxerr = [] 

521 for i in range(npols): 

522 goodsample = (sample.data[i])[~sample.mask[i]] 

523 sigmasqgoodsample = np.square((sigmasample[i])[~sample.mask[i]]) 

524 if len(goodsample) > 0: 

525 absdiff = np.abs(np.diff(goodsample)) 

526 sqdiff = np.square(absdiff) 

527 absdiffsqerr = sigmasqgoodsample + np.roll(sigmasqgoodsample, 1) 

528 sqdifferr = 4*absdiff*absdiff*(absdiffsqerr[0:-1]) 

529 auxval.append(np.sum(sqdiff)) 

530 auxerr.append(np.sqrt(np.sum(sqdifferr))) 

531 if len(auxval) > 0: 

532 value = np.sum(auxval) 

533 error = np.sqrt(np.sum(np.array(auxerr)*np.array(auxerr))) 

534 else: 

535 value = np.nan 

536 error = np.nan 

537 resultdict[metrictype] = (value, error) 

538 else: 

539 resultdict[metrictype] = (-1.0, -1.0) 

540 

541 return resultdict

559def onlineChannelAveraging(msmd, spws=None): 

560 """ 

561 For Cycle 3-onward data, determines the channel averaging factor from 

562 the ratio of the effective channel bandwidth to the channel width. 

563 spw: a single value, or a list; if None, then uses science spws 

564 Returns: single value for a single spw, or a list for a list of spws 

565 -Todd Hunter 

566 """ 

567 hanning_effBw = {1: 2.667, 2: 3.200, 4: 4.923, 8: 8.828, 16: 16.787} 

568 if spws is None: 

569 spws = getSpwList(msmd) 

570 if type(spws) != list: 

571 spws = [spws] 

572 Ns = list(hanning_effBw.keys()) 

573 ratios = [hanning_effBw[i]/i for i in Ns] 

574 Nvalues = [] 

575 for spw in spws: 

576 chanwidths = msmd.chanwidths(spw) 

577 nchan = len(chanwidths) 

578 if (nchan < 5): 

579 return 1 

580 chanwidth = abs(chanwidths[0]) 

581 chaneffwidth = msmd.chaneffbws(spw)[0] 

582 ratio = chaneffwidth/chanwidth 

583 Nvalues.append(Ns[np.argmin(abs(ratios - ratio))]) 

584 if (len(spws) == 1): 

585 return Nvalues[0] 

586 else: 

587 return Nvalues

694def getCalAtmData(ms: str, spws: list, spwsetup: dict): 

695 '''Funtion to extract Tsys, Trec, Tatm and tau data from the ASDM's CALATMOSPHERE table. 

696 param: 

697 ms: MS filename 

698 spws: List of SPWs to load 

699 spwsetup: Dictionary of metadata obtained from getSpecSetup() 

700 ''' 

701 

702 tb = casa_tools.table 

703 #Open CALATMOSPHERE table 

704 tb.open(os.path.join(ms, 'ASDM_CALATMOSPHERE')) 

705 #Get weather parameters 

706 LOG.debug("Start reading weather parameters from ASDM_CALATMOSPHERE table") 

707 tground_all = tb.getcol('groundTemperature') 

708 pground_all = tb.getcol('groundPressure') 

709 hground_all = tb.getcol('groundRelHumidity') 

710 LOG.debug("Done reading weather parameters from ASDM_CALATMOSPHERE table") 

711 #Get the spectra of the ATM measurements 

712 tmatm_all = {} 

713 tsys = {} 

714 trec = {} 

715 tau = {} 

716 antatm = {} 

717 for spwid in spws: 

718 minfreq = np.min(spwsetup[spwid]['chanfreqs']) 

719 maxfreq = np.max(spwsetup[spwid]['chanfreqs']) 

720 midfreq = 0.5*(minfreq+maxfreq) 

721 samebbspw = [s for s in spwsetup['spwlist'] if spwsetup[s]['BBname'] == spwsetup[spwid]['BBname']] 

722 subtb = tb.query('basebandName=="{0:s}" && syscalType=="TEMPERATURE_SCALE"'.format(str(spwsetup[spwid]['BBname']))) 

723 tmatm_all[spwid] = tb.getcol('startValidTime') 

724 #Get frequency vector and find section belonging to this SPW 

725 fullfreq = subtb.getcell('frequencySpectrum', 0) 

726 sectionmid = [] 

727 freqsection = [] 

728 chansec = [] 

729 chanidx = 0 

730 for s in samebbspw: 

731 chansec.append([chanidx,chanidx+spwsetup[s]['nchan']]) 

732 section = fullfreq[chanidx:chanidx+spwsetup[s]['nchan']] 

733 chanidx += spwsetup[s]['nchan'] 

734 sectionmid.append(np.mean(section)) 

735 freqsection.append(section) 

736 thissection = np.argsort(np.abs(np.array(sectionmid) - midfreq))[0] 

737 freq = freqsection[thissection] 

738 order = np.argsort(freq) 

739 startchan = chansec[thissection][0] 

740 endchan = chansec[thissection][1] 

741 #Load Tsys and Trx tables 

742 auxtsys = [subtb.getcell('tSysSpectrum', i) for i in range(subtb.nrows())] 

743 auxtrec = [subtb.getcell('tRecSpectrum', i) for i in range(subtb.nrows())] 

744 npols = auxtsys[0].shape[0] 

745 nrowstsys = len(auxtsys) 

746 tsys[spwid] = np.zeros((npols, spwsetup[spwid]['nchan'], nrowstsys)) 

747 trec[spwid] = np.zeros((npols, spwsetup[spwid]['nchan'], nrowstsys)) 

748 #Resample the curves to match the frequency of the data SPWs 

749 for pol in range(npols): 

750 for row in range(nrowstsys): 

751 tsysfit = CubicSpline(freq[order], auxtsys[row][pol][startchan:endchan][order], bc_type='not-a-knot') 

752 trecfit = CubicSpline(freq[order], auxtrec[row][pol][startchan:endchan][order], bc_type='not-a-knot') 

753 tsys[spwid][pol,:,row] = tsysfit(spwsetup[spwid]['chanfreqs']) 

754 trec[spwid][pol,:,row] = trecfit(spwsetup[spwid]['chanfreqs']) 

755 antatm[spwid] = subtb.getcol('antennaName') 

756 #To get the skyline list, pick index of first antenna, first entry of tau, for the first polarization 

757 #Resample according to the same procedure used for Tsys and Trx 

758 auxtau = subtb.getcell('tauSpectrum', 0)[0] 

759 taufit = CubicSpline(freq[order], auxtau[startchan:endchan][order], bc_type='not-a-knot') 

760 tau[spwid] = taufit(spwsetup[spwid]['chanfreqs']) 

761 tb.close() 

762 

763 return (tground_all, pground_all, hground_all, tmatm_all, tsys, trec, tau, antatm)

980def get_stats_and_shape( 

981 msname: str, 

982 datacolumn: str, 

983 data_desc_id: int, 

984 field_id: int, 

985 state_id_list: NDArray[np.int_] 

986) -> tuple[NDArray[floating], float, NDArray[np.int_]]: 

987 """Compute metrics from selected data in the given MS. 

988 

989 Core part of this function computes the mean and standard deviation 

990 for three-dimensional data (npol, nchan, nrow) by iterating over 

991 the data in the Measurement Set (MS) row-by-row to save memory usage. 

992 The implementation emulates the behavior of np.ma.mean and np.ma.std. 

993 

994 Args: 

995 msname: Name of the MS 

996 datacolumn: Name of the data column to read (e.g., 'DATA', 

997 'CORRECTED_DATA') 

998 data_desc_id: Data description id for data selection 

999 field_id: Field id for data selection 

1000 state_id_list: List of state ids for data selection 

1001 

1002 Returns: 

1003 A tuple containing: 

1004 - time averaged data (masked array) 

1005 - Normalization value for metrics 

1006 - Updated boolean mask array indicating sky lines that are not masked 

1007 """ 

1008 LOG.debug("get_stats_and_shape: Reading data row-by-row to save memory usage") 

1009 it = select_and_yield(msname, datacolumn, data_desc_id, field_id, state_id_list) 

1010 data = next(it) 

1011 npol, nchan, nrow = data.shape 

1012 # The following code emulates the behavior of np.ma.mean and np.ma.std 

1013 data_data = np.ma.filled(data, 0.0) 

1014 data_sum = np.sum(data_data, axis=2) 

1015 data_sqsum = np.sum(data_data * data_data, axis=2) 

1016 num_data = np.sum(np.logical_not(data.mask), axis=2) 

1017 for data in it: 

1018 nrow += data.shape[2] 

1019 data_data = np.ma.filled(data, 0.0) 

1020 data_sum += np.sum(data_data, axis=2) 

1021 data_sqsum += np.sum(data_data * data_data, axis=2) 

1022 num_data += np.sum(np.logical_not(data.mask), axis=2) 

1023 

1024 data_mean = np.ma.masked_array(data_sum, num_data == 0, fill_value=0.0) / num_data 

1025 data_sqmean = np.ma.masked_array(data_sqsum, num_data == 0, fill_value=0.0) / num_data 

1026 data_std = np.sqrt(data_sqmean - (data_mean * data_mean)) / np.sqrt(nrow) 

1027 

1028 data_shape = (npol, nchan, nrow) 

1029 

1030 return data_mean, data_std, data_shape

1033def get_metric(

1040 ) -> tuple[NDArray[floating], float, NDArray[np.bool_]]: 

1041 """Compute metrics from selected data in the given MS. 

1042 

1043 Core part of this function computes the mean and maximum 

1044 for three-dimensional data (npol, nchan, nrow) by iterating over 

1045 the data in the Measurement Set (MS) row-by-row to save memory usage. 

1046 The implementation emulates the behavior of np.ma.mean and np.ma.max. 

1047 

1048 Args: 

1049 msname: Name of the MS 

1050 datacolumn: Name of the data column to read (e.g., 'DATA', 

1051 'CORRECTED_DATA') 

1052 data_desc_id: Data description id for data selection 

1053 field_id: Field id for data selection 

1054 state_id_list: List of state ids for data selection 

1055 skychansel: Boolean mask array indicating sky lines 

1056 

1057 Returns: 

1058 A tuple containing: 

1059 - time averaged data (masked array) 

1060 - Normalization value for metrics 

1061 - Updated boolean mask array indicating sky lines that are not masked 

1062 """ 

1063 LOG.debug("get_metric: Reading data row-by-row to save memory usage") 

1064 it = select_and_yield(msname, datacolumn, data_desc_id, field_id, state_id_list) 

1065 data = next(it) 

1066 npol, nchan, nrow = data.shape 

1067 # The following code emulates the behavior of np.ma.mean and np.ma.max 

1068 data_data = np.ma.filled(data, 0.0) 

1069 data_sum = np.sum(data_data, axis=2) 

1070 data_absmax = np.max(np.abs(data_data), axis=2) 

1071 num_data = np.sum(np.logical_not(data.mask), axis=2) 

1072 for data in it: 

1073 nrow += data.shape[2] 

1074 data_data = np.ma.filled(data, 0.0) 

1075 data_sum += np.sum(data_data, axis=2) 

1076 num_data += np.sum(np.logical_not(data.mask), axis=2) 

1077 _absmax = np.max(np.abs(data_data), axis=2) 

1078 data_absmax = np.maximum(data_absmax, _absmax) 

1079 

1080 # Pre-correction average 

1081 precorravedataon = np.ma.masked_array(data_sum, num_data == 0, fill_value=0.0) / num_data 

1082 maskedchans = np.any(precorravedataon.mask, axis=0) 

1083 

1084 skychansel[maskedchans] = False 

1085 

1086 # Try to calculate the normalizing value for the metrics 

1087 # If it cannot calculate it, fill default value of 1 

1088 # similar thing for plot ranges 

1089 try: 

1090 # metricnorm = np.ma.max(np.ma.abs(normsample)) 

1091 metricnorm = np.ma.max(data_absmax[:, skychansel]) 

1092 except Exception: 

1093 LOG.warning('Could not determine normalization of data! Are all of them flagged??') 

1094 metricnorm = 1.0 

1095 

1096 if metricnorm is np.ma.masked: 

1097 metricnorm = 1.0 

1098 

1099 return precorravedataon, metricnorm, skychansel

1482def getJyperKfromCaltable(mslist, context): 

1483 '''Get Jy/K factors from pipeline caltables. Will determine the MS list and tables from pipeline context.

1495 

1496 Raises:
