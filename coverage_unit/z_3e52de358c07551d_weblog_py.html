72def merge_td_columns(rows, num_to_merge=None, vertical_align=False): 

73 """Merge HTML TD columns with identical values using rowspan. 

74 

75 Arguments: 

76 rows -- a list of tuples, one tuple per row, containing n elements for the 

77 n columns. 

78 num_to_merge -- the number of columns to merge, starting from the left 

79 hand column. Leave as None to merge all columns. 

80 vertical_align -- Set to True to vertically centre any merged/unmerged cells. 

81 

82 Output: 

83 A list of strings, one string per row, containing TD elements. 

84 """ 

85 transposed = list(zip(*rows)) 

86 if num_to_merge is None: 

87 num_to_merge = len(transposed) 

88 valign = ' style="vertical-align:middle;"' if vertical_align else '' 

89 

90 new_cols = [] 

91 for col_idx, col in enumerate(transposed): 

92 if col_idx > num_to_merge - 1: 

93 new_cols.append(['<td>%s</td>' % v for v in col]) 

94 continue 

95 

96 merged = [] 

97 start = 0 

98 while start < len(col): 

99 l = col[start:] 

100 same_vals = list(itertools.takewhile(lambda x: x == col[start], l)) 

101 rowspan = len(same_vals) 

102 start += rowspan 

103 

104 if rowspan > 1: 

105 new_td = [f'<td rowspan="{rowspan}"{valign}>{same_vals[0]}</td>'] 

106 blanks = [''] * (rowspan - 1) 

107 merged.extend(new_td + blanks) 

108 else: 

109 td = f'<td{valign}>{same_vals[0]}</td>' 

110 merged.append(td) 

111 

112 new_cols.append(merged) 

113 

114 return list(zip(*new_cols))

117def merge_td_rows(table): 

118 """ 

119 Merge HTML TD rows with identical values using colspan. 

120 

121 Arguments: 

122 table -- a list of tuples, one tuple per row, containing n elements for the n columns. 

123 

124 Output: 

125 A list of tuples with adjusted idential values merged with colspan. 

126 """ 

127 new_table = [] 

128 for row in table: 

129 row_list = list(row) 

130 start = 0 

131 while start < len(row): 

132 start_cell = row[start] 

133 merge_count = 0 

134 end = start+1 

135 

136 while end < len(row): 

137 if start_cell == row[end]: 

138 row_list[end] = '' 

139 merge_count += 1 

140 end += 1 

141 else: 

142 break 

143 if merge_count > 0: 

144 row_list[start] = row_list[start].replace('<td', fr'<td colspan="{merge_count+1}"') 

145 start += 1 

146 

147 new_table.append(tuple(row_list)) 

148 

149 return new_table

164def total_time_on_target_on_source(ms, autocorr_only=False): 

165 """ 

166 Return the nominal total time on target source for the given MeasurementSet 

167 excluding OFF-source integrations (REFERENCE). The flag is not taken into account. 

168 

169 Background of development: ALMA-TP observations have integrations of both TARGET 

170 and REFERENCE intents in one scan. Scan.time_on_source does not return appropriate 

171 exposure time in the case. 

172 

173 :param ms: MeasurementSet domain object to examine 

174 :param autocorr_only: 

175 :return: a datetime.timedelta object set to the total time on source 

176 """ 

177 science_spws = ms.get_spectral_windows(science_windows_only=True) 

178 state_ids = [s.id for s in ms.states if 'TARGET' in s.intents] 

179 max_time = 0.0 

180 ant_ids = [a.id for a in ms.antennas] 

181 dds = [ms.get_data_description(spw=spw) for spw in science_spws] 

182 science_dds = np.unique([dd.id for dd in dds]) 

183 with casa_tools.TableReader(ms.name) as tb: 

184 for dd in science_dds: 

185 for a1 in ant_ids: 

186 for a2 in ant_ids: 

187 if autocorr_only and a1 != a2: 

188 continue 

189 seltb = tb.query('DATA_DESC_ID == %d AND ANTENNA1 == %d AND ANTENNA2 == %d AND STATE_ID IN %s' % ( 

190 dd, a1, a2, utils.list_to_str(state_ids))) 

191 try: 

192 if seltb.nrows() == 0: 

193 continue 

194 target_exposures = seltb.getcol('EXPOSURE').sum() 

195 LOG.debug( 

196 "Selected %d ON-source rows for DD=%d, Ant1=%d, Ant2=%d: total exposure time = %f sec" % ( 

197 seltb.nrows(), dd, a1, a2, target_exposures)) 

198 max_time = max(max_time, target_exposures) 

199 finally: 

200 seltb.close() 

201 LOG.debug("Max ON-source exposure time = %f sec" % max_time) 

202 return datetime.timedelta(int(max_time / 86400), int(max_time % 86400), int((max_time % 1) * 1e6))

429def split_spw(spw_string: str) -> str: 

430 """ 

431 Splits an SPW string on '#' characters while retaining them, and inserts a <br/> 

432 before the 'ALMA' portion to improve readability. Returns the original string 

433 if 'ALMA' is not found. 

434 

435 Example: 

436 Input: "X100001#X900000004#ALMA_RB_06#BB_1#SW-01#FULL_RES" 

437 Output: "X100001#<br/>X900000004#<br/>ALMA_RB_06#BB_1#SW-01#FULL_RES" 

438 

439 Args: 

440 spw_string: The original SPW string containing '#' delimiters. 

441 

442 Returns: 

443 The reformatted SPW string with <br/> inserted before 'ALMA'. 

444 """ 

445 parts = [] 

446 current = "" 

447 

448 for segment in spw_string.split('#'): 

449 if current: 

450 parts.append(current + '#') 

451 current = segment 

452 parts.append(current) 

453 

454 for i, part in enumerate(parts): 

455 if 'ALMA' in part: 

456 return "<br/>".join(parts[:i] + ["".join(parts[i:])]) 

457 

458 return spw_string

461def get_directory_size(directory): 

462 """ 

463 Calculate the total size of a directory in megabytes (MB). 

464 

465 This includes all files in the directory and its subdirectories. 

466 Symlinks are ignored to avoid counting linked files or causing errors. 

467 

468 Parameters: 

469 directory (str): Path to the target directory. 

470 

471 Returns: 

472 float: Total size of the directory in megabytes. 

473 """ 

474 total_size = 0 

475 for dirpath, dirnames, filenames in os.walk(directory): 

476 for filename in filenames: 

477 filepath = os.path.join(dirpath, filename) 

478 if not os.path.islink(filepath): 

479 try: 

480 total_size += os.path.getsize(filepath) 

481 except OSError as e: 

482 # make it log.warning 

483 print(f"Error accessing {filepath}: {e}") 

484 return total_size / (1024 * 1024)
