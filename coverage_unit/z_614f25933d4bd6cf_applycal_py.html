342def set_unit(ms: MeasurementSet, calapp: list[CalApplication]): 

343 """Set unit to MS data column according to applied calibrations. 

344 

345 Args: 

346 ms: MeasurementSet domain object. 

347 calapp: List of CalApplication objects. 

348 """ 

349 target_fields = ms.get_fields(intent='TARGET') 

350 data_units = dict((f.id, '') for f in target_fields) 

351 for a in calapp: 

352 calto = a.calto 

353 field_name = calto.field 

354 if len(field_name) == 0: 

355 field = ms.get_fields(intent='TARGET') 

356 else: 

357 field = [f for f in ms.get_fields(name=field_name) if 'TARGET' in f.intents] 

358 if len(field) == 0: 

359 continue 

360 

361 assert len(field) == 1 

362 field_id = field[0].id 

363 caltypes = [cf.caltype for cf in a.calfrom] 

364 if ('ps' in caltypes) and ('tsys' in caltypes): 

365 data_units[field_id] = 'K' 

366 else: 

367 LOG.warning( 

368 f'Calibration of {ms.basename} (field {field_name}) is not ' 

369 'correct. Missing pscal and/or tsyscal.' 

370 ) 

371 if 'amp' in caltypes or 'gaincal' in caltypes: 

372 if data_units[field_id] == 'K': 

373 data_units[field_id] = 'Jy' 

374 

375 unit_list = numpy.asarray(list(data_units.values())) 

376 if numpy.all(unit_list == 'K'): 

377 data_unit = 'K' 

378 elif numpy.all(unit_list == 'Jy'): 

379 data_unit = 'Jy' 

380 elif numpy.any(unit_list == 'Jy'): 

381 LOG.warning( 

382 f'Calibration of {ms.basename} is not correct. ' 

383 'Some of calibrations (pscal, tsyscal, ampcal) are missing.' 

384 ) 

385 data_unit = '' 

386 else: 

387 data_unit = '' 

388 

389 if data_unit != '': 

390 with casa_tools.TableReader(ms.name, nomodify=False) as tb: 

391 colnames = tb.colnames() 

392 target_columns = set(colnames) & set(['DATA', 'FLOAT_DATA', 'CORRECTED_DATA']) 

393 for col in target_columns: 

394 tb.putcolkeyword(col, 'UNIT', data_unit)
