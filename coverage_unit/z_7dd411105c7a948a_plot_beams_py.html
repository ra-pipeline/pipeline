1#!/usr/bin/env python 

2 

3import os 

4import numpy as np 

5import matplotlib.pyplot as plt 

6 

7import pipeline.infrastructure as infrastructure 

8from pipeline.infrastructure import casa_tools 

9 

10LOG = infrastructure.get_logger(__name__) 

11 

12def plot_beams(psf_name, plotfile): 

13 """ 

14 Method to plot per channel beams from a PSF cube. 

15 """ 

16 

17 qaTool = casa_tools.quanta 

18 

19 with casa_tools.ImageReader(psf_name) as image: 

20 im_info = image.summary() 

21 

22 f_index = np.where(im_info['axisnames'] == 'Frequency')[0][0] 

23 f_unit = im_info['axisunits'][f_index] 

24 

25 num_chan = im_info['shape'][f_index] 

26 

27 channels = np.arange(num_chan) 

28 freq_refval = float(qaTool.getvalue(qaTool.convert(qaTool.quantity(im_info['refval'][f_index], f_unit), 'GHz'))[0]) 

29 freq_refchan = float(qaTool.getvalue(qaTool.convert(qaTool.quantity(im_info['refpix'][f_index], f_unit), 'GHz'))[0]) 

30 freq_step = float(qaTool.getvalue(qaTool.convert(qaTool.quantity(im_info['incr'][f_index], f_unit), 'GHz'))[0]) 

31 freqs = np.array([freq_refval+freq_step*(c-freq_refchan) for c in channels]) 

32 

33 rb = im_info['perplanebeams'] 

34 major = np.array([float(qaTool.getvalue(qaTool.convert(rb['beams'][f'*{c}']['*0']['major'], 'arcsec'))[0]) for c in channels]) 

35 minor = np.array([float(qaTool.getvalue(qaTool.convert(rb['beams'][f'*{c}']['*0']['minor'], 'arcsec'))[0]) for c in channels]) 

36 

37 plt.close('all') 

38 fig, ax = plt.subplots(figsize=(6.4, 4.8), sharex=True) 

39 plt.text(0.5, 1.18, 'Beam size per channel', transform=ax.transAxes, fontsize=12, ha='center') 

40 plt.text(0.5, 1.14, os.path.basename(psf_name), transform=ax.transAxes, fontsize=9, ha='center') 

41 ax.plot(freqs, major, label='Major axis', c='darkblue') 

42 ax.plot(freqs, minor, label='Minor axis', c='orange') 

43 ax.legend(loc='center left') 

44 ax.set_xlabel('Frequency [GHz]') 

45 ax.set_ylabel('Axis size [arcsec]') 

46 

47 # Alternative x axis 

48 ax2y = ax.twiny() 

49 ax2y.set_xlim(map(lambda f: (f-freqs[0])/freq_step, ax.get_xlim())) 

50 ax2y.set_xlabel(f'{num_chan} Channels') 

51 

52 # Alternative y axis 

53 ax2x = ax.twinx() 

54 ax2x.set_ylim(ax.get_ylim() / max(major)) 

55 ax2x.set_ylabel('Axis size / max(Major axis)') 

56 

57 fig.savefig(plotfile, bbox_inches='tight') 

58 plt.clf() 

59 plt.close(fig)
