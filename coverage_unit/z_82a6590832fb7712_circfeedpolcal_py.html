554 'rotmeas': 0, 

555 'scalebychan': True, 

556 'usescratch': True} 

557 else: 

558 LOG.error("No known flux calibrator found - please check the data.") 

559 

560 # PIPE-1750, added an option to disable setjy call

568 except Exception as ex: 

569 LOG.warning("Exception: Problem with circfeedpolcal setjy. {!s}".format(str(ex)))

572 return fluxcalfieldname, fluxcalfieldid, fluxcal

575 """ 

576 Returns: spwmap for use with gaintable in callibrary (polcal and applycal)

578 m = self.inputs.context.observing_run.get_ms(self.inputs.vis)
