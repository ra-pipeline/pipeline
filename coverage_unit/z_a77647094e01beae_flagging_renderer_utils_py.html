48def flags_by_intent(ms: MeasurementSet, 

49 summaries: list[dict], 

50 intents: Sequence[str] = None) -> dict[str, dict[str, FlagTotal]]: 

51 """ 

52 Arguments: 

53 ms: an instance of MeasurementSet. 

54 summaries: a list of summaries created by flagging tasks. 

55 intents: List of intents for which the total flagging percentage should be reported. 

56 Returns: 

57 a dictionary indexed by summary name (e.g. 'before', '...'), 

58 where each element is itself a dictionary containing FlagTotal tuples for each intent. 

59 """ 

60 # create a dictionary of scans per observing intent, eg. 'PHASE':['1','2','7'] 

61 intent_scans = { 

62 # convert IDs to strings as they're used as summary dictionary keys 

63 intent: [str(scan.id) for scan in ms.scans if intent in scan.intents] 

64 for intent in intents 

65 } 

66 

67 # while we're looping, get the total flagged by looking in all scans 

68 intent_scans['TOTAL'] = [str(s.id) for s in ms.scans] 

69 

70 total = collections.defaultdict(dict) 

71 

72 previous_summary = None 

73 for summary in summaries: 

74 

75 for intent, scan_ids in intent_scans.items(): 

76 flagcount = 0 

77 totalcount = 0 

78 

79 for i in scan_ids: 

80 # workaround for KeyError exception when summary 

81 # dictionary doesn't contain the scan 

82 if i not in summary['scan']: 

83 continue 

84 

85 flagcount += int(summary['scan'][i]['flagged']) 

86 totalcount += int(summary['scan'][i]['total']) 

87 

88 if previous_summary: 

89 flagcount -= int(previous_summary['scan'][i]['flagged']) 

90 

91 ft = FlagTotal(flagcount, totalcount) 

92 total[summary['name']][intent] = ft 

93 

94 previous_summary = summary 

95 

96 return total

99def flags_by_science_spws(ms: MeasurementSet, summaries: list[dict]) -> dict[str, dict[str, FlagTotal]]: 

100 """ 

101 Returns: 

102 a dictionary indexed by summary name, where each element is a dict 

103 with a single key 'SCIENCE_SPWS' and a FlagTotal value 

104 """ 

105 science_spws = ms.get_spectral_windows(science_windows_only=True) 

106 

107 total = collections.defaultdict(dict) 

108 

109 previous_summary = None 

110 for summary in summaries: 

111 

112 flagcount = 0 

113 totalcount = 0 

114 

115 for spw in science_spws: 

116 spw_id = str(spw.id) 

117 # workaround for KeyError exception when summary 

118 # dictionary doesn't contain the spw 

119 if spw_id not in summary['spw']:

121 flagcount += int(summary['spw'][spw_id]['flagged']) 

122 totalcount += int(summary['spw'][spw_id]['total'])

125 flagcount -= int(previous_summary['spw'][spw_id]['flagged']) 

126 

127 ft = FlagTotal(flagcount, totalcount) 

128 total[summary['name']]['SCIENCE SPWS'] = ft 

129 

130 previous_summary = summary 

131 

132 return total

135def adjust_non_science_totals(flagtotals, non_science_agents=None): 

136 """ 

137 Return a copy of the FlagSummary dictionaries, with totals reduced to 

138 account for flagging performed by non-science flagging agents. 

139 

140 The incoming flagtotals report how much data was flagged per agent per 

141 data selection. These flagtotals are divided into two groups: those whose 

142 agent should be considered 'non-science' (and are indicated as such in the 

143 non_science_agents argument) and the remainder. The total number of rows 

144 flagged due to non-science agents is calculated and subtracted from the 

145 total for each of the remainder agents. 

146 """ 

147 if not non_science_agents: 

148 return flagtotals 

149 

150 agents_to_copy = set(non_science_agents) 

151 agents_to_adjust = set(flagtotals.keys()) - agents_to_copy 

152 data_selections = set() 

153 for result in flagtotals.values(): 

154 data_selections.update(set(result.keys())) 

155 

156 # copy agents that use the total number of visibilities across to new 

157 # results 

158 adjusted_results = dict((agent, flagtotals[agent]) 

159 for agent in agents_to_copy 

160 if agent in flagtotals) 

161 

162 # tot up how much data was flagged by each agent per data selection 

163 flagged_non_science = {} 

164 for data_selection in data_selections: 

165 flagged_non_science[data_selection] = sum([v[data_selection].flagged 

166 for v in adjusted_results.values()]) 

167 

168 # subtract this 'number of rows flagged per data selection' from the total 

169 # for the remaining agents 

170 for agent in agents_to_adjust: 

171 for data_selection in flagtotals[agent]: 

172 unadjusted = flagtotals[agent][data_selection] 

173 adjusted = FlagTotal(unadjusted.flagged, 

174 unadjusted.total - flagged_non_science[data_selection]) 

175 if agent not in adjusted_results: 

176 adjusted_results[agent] = {} 

177 adjusted_results[agent][data_selection] = adjusted 

178 

179 return adjusted_results

182def intents_to_summarise(context: Context, all_flag_summary_intents: Sequence[str] | None = None) -> list[str]: 

183 """Find out which intents to list in the flagging table. 

184 

185 Arguments: 

186 context: Pipeline context object containing state information. 

187 all_flag_summary_intents: a list of intents to summarise; 

188 if None, the default list of all relevant intents (calibration and target) is used. 

189 Returns: 

190 the subset of intents from the input list all_flag_summary_intents that are actually present in at least one MS 

191 (a list of strings in the same order as input, but omitting missing ones). 

192 """ 

193 # First get all intents across all MSes in context 

194 context_intents = functools.reduce(lambda x, m: x.union(m.intents), 

195 context.observing_run.measurement_sets, 

196 set()) 

197 # then match intents against those we want in the table, removing those not present. 

198 # List order is preserved in the table. 

199 if all_flag_summary_intents is None: 

200 all_flag_summary_intents = [ 

201 'AMPLITUDE', 'BANDPASS', 'CHECK', 'DIFFGAINREF', 'DIFFGAINSRC', 'PHASE', 'POLANGLE', 'POLARIZATION', 

202 'POLLEAKAGE', 'TARGET'] 

203 intents_to_summarise = [i for i in all_flag_summary_intents 

204 if i in context_intents.intersection(set(all_flag_summary_intents))] 

205 return intents_to_summarise
